56|def _db_path(database: str) -> str | None:
57|    if not re.fullmatch(r"[A-Za-z0-9_]+", database or ""):
58|        return None

63|def _run_query(path: str, query: str, time_limit: float, max_rows: int | None):
64|    """Execute on a read-only connection; raises sqlite3.OperationalError('interrupted') past the time limit."""
65|    con = sqlite3.connect(f"file:{path}?mode=ro", uri=True, check_same_thread=False)
66|    con.text_factory = lambda b: b.decode(errors="replace")
67|    deadline = time.monotonic() + time_limit
68|    con.set_progress_handler(lambda: 1 if time.monotonic() > deadline else 0, 10_000)
69|    try:
70|        cur = con.execute(query)
71|        cols = [d[0] for d in (cur.description or [])]
72|        rows = cur.fetchall() if max_rows is None else cur.fetchmany(max_rows + 1)
73|        return cols, rows
74|    finally:
75|        con.close()

78|def _format(cols: list[str], rows: list[tuple]) -> str:
79|    max_rows, max_chars = TOOL_CONFIGS["max_rows"], TOOL_CONFIGS["max_chars"]
80|    if not cols:
81|        return "OK (no rows returned)"
82|    lines = [" | ".join(cols)] + [" | ".join("NULL" if v is None else str(v) for v in r) for r in rows[:max_rows]]
83|    text = "\n".join(lines)
84|    if len(rows) > max_rows:
85|        text += f"\n... (more than {max_rows} rows)"
86|    if len(text) > max_chars:

91|async def execute_tool(name: str, params: dict) -> str:
92|    if name != "execute_sql":
93|        return f"Error: Tool '{name}' not found"
94|    database, query = str(params.get("database", "")), str(params.get("query", "")).strip()
95|    path = _db_path(database)
96|    if path is None:
97|        return f"Error: unknown database '{database}'"
98|    if not query:
99|        return "Error: empty query"
100|    limit = TOOL_CONFIGS["query_time_limit"]
101|    async with SEMAPHORE:
102|        try:
103|            cols, rows = await asyncio.to_thread(_run_query, path, query, limit, TOOL_CONFIGS["max_rows"])
104|        except sqlite3.OperationalError as e:
105|            if "interrupted" in str(e):
106|                return f"Error: query timed out after {limit:.0f}s"
107|            return f"Error: {e}"
108|        except (sqlite3.Error, ValueError) as e:  # ValueError: e.g. more than one statement
109|            return f"Error: {e}"
110|    return _format(cols, rows)

133|async def reward_func(args, sample: Sample, **kwargs):
134|    """Execution match against the gold query: +1 if the final query returns the gold result, -1 otherwise."""
135|    label = json.loads(sample.label)
136|    db_path = _db_path(label["db_id"])
137|    # a response cut off at the length limit never gave its answer, whatever SQL it had written so far
138|    truncated = sample.status == Sample.Status.TRUNCATED
139|    pred_sql = None if truncated else extract_final_sql(sample.response or "")
140|    result = {"score": -1.0, "pred": pred_sql or "", "acc": False}
141|    if db_path is not None and pred_sql:
142|        key = (label["db_id"], label["query"])
143|        try:
144|            if key not in _GOLD_CACHE:
145|                _GOLD_CACHE[key] = await asyncio.to_thread(_execute_for_reward, db_path, label["query"])
146|            gold = _GOLD_CACHE[key]
147|            pred = await asyncio.to_thread(_execute_for_reward, db_path, pred_sql)
148|            ordered = "order by" in label["query"].lower()
149|            result["acc"] = pred == gold if ordered else Counter(pred) == Counter(gold)
150|        except (sqlite3.Error, ValueError):
151|            result["acc"] = False
152|    if result["acc"]:
153|        result["score"] = 1.0
154|    return result

161|def clip_to_context(args, data: list[list[Sample]]):
162|    """Clip trajectories that ran past the context budget. The engine stops generating at --rollout-max-context-len,

164|    trainer's log-prob pass). Used as --rollout-sample-filter-path and after the rescue."""
165|    global _TOKENIZER

176|            sample.status = Sample.Status.TRUNCATED
