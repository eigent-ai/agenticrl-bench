29|    prompt_len = len(sample.tokens) - sample.response_length
30|    if prompt_len <= 0:

36|    payload = {

67|def _build_batch_prefill_scoring_payload(
68|    args: Any,
69|    samples: list[Sample],
70|    sampling_params: Mapping[str, Any],
71|) -> dict[str, Any]:
72|    payloads = [_build_prefill_scoring_payload(args, sample, sampling_params) for sample in samples]
73|    logprob_start_len = payloads[0]["logprob_start_len"]
74|    if any(payload["logprob_start_len"] != logprob_start_len for payload in payloads):
75|        raise ValueError("Batched SGLang prefill scoring requires a shared logprob_start_len")
76|
77|    lora_paths = {payload.get("lora_path") for payload in payloads}
78|    if len(lora_paths) > 1:
79|        # A batch payload carries a single lora_path; callers must group samples by adapter first.
80|        raise ValueError("Batched SGLang prefill scoring requires a shared lora_path")
81|
82|    batch_payload: dict[str, Any] = {
83|        "input_ids": [payload["input_ids"] for payload in payloads],
84|        "sampling_params": payloads[0]["sampling_params"],
85|        "return_logprob": True,
86|        "logprob_start_len": logprob_start_len,
87|    }
88|    if (lora_path := next(iter(lora_paths))) is not None:
89|        batch_payload["lora_path"] = lora_path
90|    return batch_payload

93|def _extract_response_logprobs(sample: Sample, meta_info: Mapping[str, Any]) -> list[float]:
94|    input_token_logprobs = meta_info.get("input_token_logprobs")
95|    if not input_token_logprobs:
96|        raise ValueError("SGLang prefill scoring response did not include input_token_logprobs")
97|
98|    response_items = input_token_logprobs[-sample.response_length :]
99|    response_tokens = sample.tokens[-sample.response_length :]
100|    scored_tokens = [item[1] for item in response_items]
101|    if scored_tokens != response_tokens:
102|        raise ValueError(
103|            "SGLang prefill scoring token alignment mismatch: "
104|            f"expected response tail {response_tokens[:8]}... len={len(response_tokens)}, "
105|            f"got {scored_tokens[:8]}... len={len(scored_tokens)}"
106|        )
107|
108|    response_logprobs = [item[0] for item in response_items]
109|    if any(logprob is None for logprob in response_logprobs):
110|        raise ValueError("SGLang prefill scoring returned None for a response-token logprob")
111|
112|    return response_logprobs
