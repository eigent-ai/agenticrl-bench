41|    # tool-call markup stays inline in the assistant text (what the model emitted)
42|    record_trajectory = args.save_debug_trajectory_data is not None

44|

46|
47|    prompt_tokens_ids = compute_prompt_ids_from_sample(input.state, sample, tools=tool_specs)

81|        if sink is not None:

83|        update_sample_with_tool_responses(sample, tool_messages, tokenizer=tokenizer)
84|        if record_trajectory:
