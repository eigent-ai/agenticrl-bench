115|def _decode_topk_buffer(info: str, num_tokens: int, num_layers: int, topk: int) -> np.ndarray:
116|    x = np.frombuffer(pybase64.b64decode(info.encode("ascii")), dtype=np.int32)
117|    if num_tokens <= 0:
118|        return np.empty((0, num_layers, max(0, topk)), dtype=np.int32)
119|    if topk == -1:  # indexer: topk dim recovered from buffer length
120|        topk = len(x) // (num_tokens * num_layers)
121|    return x.reshape(num_tokens, num_layers, topk)
