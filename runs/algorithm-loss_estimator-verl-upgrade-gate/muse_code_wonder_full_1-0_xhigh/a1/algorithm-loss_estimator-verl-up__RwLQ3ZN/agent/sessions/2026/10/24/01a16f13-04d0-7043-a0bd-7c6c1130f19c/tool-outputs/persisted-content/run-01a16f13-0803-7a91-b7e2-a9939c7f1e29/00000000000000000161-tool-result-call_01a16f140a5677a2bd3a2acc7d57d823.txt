871|@register_adv_est(AdvantageEstimator.OPTIMAL_TOKEN_BASELINE)
872|def compute_optimal_token_baseline_advantage(
873|    token_level_rewards: torch.Tensor,
874|    response_mask: torch.Tensor,
875|    index: np.ndarray,
876|    old_log_probs: torch.Tensor,
877|    sum_pi_squared: torch.Tensor,
878|    rollout_is_weights: torch.Tensor = None,
879|    handle_zero_tail: bool = True,
880|    epsilon: float = 1e-8,
881|    **kwargs,
882|) -> tuple[torch.Tensor, torch.Tensor]:
883|    """
884|    Compute advantages using Optimal Token Baseline (OTB).
885|
886|    Unlike the group mean based baseline which uses a single baseline per trajectory,
887|    this computes a unique baseline for each timestep using cumulative path variance.
888|
889|    Theory:
890|        For each timestep t in each prompt group:
891|            B_t* = E[G_t × W_t] / E[W_t]
892|        where W_t = Σ_{j=1}^t ||s_j||² (cumulative path-variance proxy)
893|        and ||s_j||² = 1 - 2π_j + Σπ²
894|
895|    The cumulative sum W_t captures the "realized energy" of trajectory has been up to timestep t,
896|    giving higher weight to predicting rewards on high-variance paths.
897|
898|    Args:
899|        token_level_rewards: Rewards at each token position [shape: (bs, response_length)]
900|        response_mask: Binary mask for valid tokens (1) vs padding (0) [shape: (bs, response_length)]
901|        index: Prompt indices for grouping trajectories from same prompt [shape: (bs,)]
902|        old_log_probs: Log probabilities from training policy during generation [shape: (bs, response_length)]
903|        sum_pi_squared: Sum of squared probabilities over vocabulary Σπ² [shape: (bs, response_length)]
904|        rollout_is_weights: Pre-computed IS weights for W correction [shape: (bs, response_length)],
905|            None if not using IS
906|        handle_zero_tail: If True, zero baselines will be set in the portion of the longest trajectory
907|            that extends beyond the second-longest trajectory in the prompt group.
908|            Default: True
909|        epsilon: Small constant for numerical stability (default: 1e-8)
910|
911|    Returns:
912|        advantages: OTB advantage estimates [shape: (bs, response_length)]
913|        returns: Cumulative rewards (returns) from each position [shape: (bs, response_length)]
914|
915|    Note on Rollout Importance Sampling:
916|        When rollout_is_weights is provided, W_t is scaled by ρ̄²(t) to minimize MSE under truncated IS:
917|            B_t* = Σ[G_t × ρ̄²(t) × W_t] / Σ[ρ̄²(t) × W_t]
918|    """
919|    with torch.no_grad():
920|        batch_size, seq_len = token_level_rewards.shape
921|        device = token_level_rewards.device
922|
923|        # Compute returns (reward-to-go) for each timestep
924|        returns = (token_level_rewards * response_mask).flip(dims=[-1]).cumsum(dim=-1).flip(dims=[-1])
925|
926|        # Step 1: Compute w_per_timestep = 1 - 2π_t + Σπ²)
927|        pi_t = torch.exp(old_log_probs)
928|        w_per_timestep = 1 - 2 * pi_t + sum_pi_squared
929|
930|        # Step 2: Apply rollout importance sampling correction (if enabled)
931|        if rollout_is_weights is not None:
932|            # Scale W by ρ̄² to minimize MSE under truncated IS
933|            w_per_timestep = w_per_timestep * (rollout_is_weights**2)
934|
935|        # Step 3: Compute cumulative path-variance proxy: W_t = Σ_{j=1}^t w_j
936|        # This measures accumulated variance from the start of the trajectory up to timestep t
937|        w_cumulative = (w_per_timestep * response_mask).cumsum(dim=-1)
938|
939|        # Group trajectories by prompt
940|        prompt_groups = defaultdict(list)
941|        for i in range(batch_size):
942|            prompt_groups[index[i]].append(i)
943|
944|        # Initialize baselines tensor [batch_size, seq_len]
945|        baselines = torch.zeros_like(returns)
946|
947|        # Compute per-step baseline for each prompt group
948|        for _, trajectory_indices in prompt_groups.items():
949|            N = len(trajectory_indices)
950|            if N == 1:
951|                # Single trajectory - no baseline (advantage = return)
952|                continue
953|
954|            traj_idx = torch.tensor(trajectory_indices, device=device)
955|
956|            # Extract group data [N, seq_len]
957|            returns_group = returns[traj_idx]
958|            w_cumulative_group = w_cumulative[traj_idx]
959|            mask_group = response_mask[traj_idx]
960|
961|            # Compute per-timestep baseline: B_t = Σ[G_t × W_t] / Σ[W_t]
962|            # where W_t = Σ_{j=1}^t ||s_j||² (cumulative path variance)
963|            # Shape: [seq_len]
964|            numerator = (returns_group * w_cumulative_group * mask_group).sum(dim=0)  # Sum over trajectories
965|            denominator = (w_cumulative_group * mask_group).sum(dim=0) + epsilon
966|
967|            baseline_per_step = numerator / denominator  # [seq_len]
968|
969|            # Assign to all trajectories in this group
970|            baselines[traj_idx] = baseline_per_step.unsqueeze(0).expand(N, -1)
971|
972|            if handle_zero_tail:
973|                # Optionally zero out the portion of the longest trajectory that extends
974|                # beyond the second-longest trajectory in the prompt group.
975|                response_lengths = mask_group.sum(dim=-1)
976|                sorted_lengths, _ = torch.sort(response_lengths)
977|                max_length = int(sorted_lengths[-1].item())
978|                second_max_length = int(sorted_lengths[-2].item())
979|                max_length_idx = (response_lengths == max_length).nonzero(as_tuple=True)[0]
980|                if max_length_idx.numel() == 1 and max_length > second_max_length:
981|                    max_length_traj_idx = trajectory_indices[int(max_length_idx[0])]
982|                    baselines[max_length_traj_idx, second_max_length:] = 0.0
983|
984|        # Compute advantages: A_t = G_t - B_t
985|        advantages = (returns - baselines) * response_mask
986|
987|    return advantages, returns

1140|def agg_loss(
1141|    loss_mat: torch.Tensor,
1142|    loss_mask: torch.Tensor,
1143|    loss_agg_mode: str,
1144|    dp_size: int = 1,
1145|    batch_num_tokens: Optional[int] = None,
1146|    global_batch_size: Optional[int] = None,
1147|    loss_scale_factor: Optional[int] = None,
1148|):
1149|    """
1150|    Aggregate the loss across global batch to ensure the loss is invariant to fsdp/megatron parallelism.
1151|
1152|    NOTE: The returned loss has different behaviors for different backend:
1153|    - FSDP: the loss is directly used for backward.
1154|    - Megatron: the loss should be scaled by `num_microbatches` and `cp_size` for pp schedule.
1155|
1156|    Args:
1157|        loss_mat: micro batch loss matrix, (bs, response_length)
1158|        loss_mask: micro batch loss mask, (bs, response_length)
1159|        loss_agg_mode: method to aggregate the loss matrix into a scalar
1160|        dp_size: data parallel size
1161|        batch_num_tokens: number of valid tokens in global batch
1162|        global_batch_size: global batch size
1163|        loss_scale_factor: scale factor for "seq-mean-token-sum-norm" mode. If None, uses loss_mask.shape[-1].
1164|            Set this to a constant value to ensure consistent normalization throughout training.
1165|
1166|    Returns:
1167|        loss: `a scalar torch.Tensor`
1168|            aggregated loss
1169|    """
1170|    if loss_agg_mode == "token-mean":
1171|        if batch_num_tokens is None:
1172|            if dp_size > 1:
1173|                raise ValueError("(global) batch_num_tokens is required when dp_size > 1")
1174|            batch_num_tokens = loss_mask.sum()
1175|        loss = verl_F.masked_sum(loss_mat, loss_mask) / batch_num_tokens * dp_size
1176|    elif loss_agg_mode == "token-sum":
1177|        # DDP/FSDP average gradients across data-parallel ranks. Scaling each
1178|        # rank's local token sum by dp_size makes the reduced gradient equal to
1179|        # the sum over all valid tokens in the global batch.
1180|        loss = verl_F.masked_sum(loss_mat, loss_mask) * dp_size
1181|    elif loss_agg_mode in ["seq-mean-token-sum", "seq-mean-token-sum-norm"]:
1182|        seq_losses = torch.sum(loss_mat * loss_mask, dim=-1)  # token-sum
1183|        seq_mask = (torch.sum(loss_mask, dim=-1) > 0).float()  # exclude fully masked sequences
1184|        if global_batch_size is None:
1185|            if dp_size > 1:
1186|                raise ValueError("global_batch_size is required when dp_size > 1")
1187|            global_batch_size = seq_mask.sum()
1188|        loss = verl_F.masked_sum(seq_losses, seq_mask) / global_batch_size * dp_size  # seq-mean
1189|        if loss_agg_mode == "seq-mean-token-sum-norm":
1190|            if loss_scale_factor is None:
1191|                horizon = loss_mask.shape[-1]
1192|                loss_scale_factor = horizon
1193|            loss /= loss_scale_factor
1194|    elif loss_agg_mode == "seq-mean-token-mean":
1195|        seq_mask = torch.sum(loss_mask, dim=-1)  # per-sequence token count
1196|        seq_losses = torch.sum(loss_mat * loss_mask, dim=-1) / (seq_mask + 1e-8)  # token-mean
1197|        seq_mask = (seq_mask > 0).float()  # exclude fully masked sequences
1198|        if global_batch_size is None:
1199|            if dp_size > 1:
1200|                raise ValueError("global_batch_size is required when dp_size > 1")
1201|            global_batch_size = seq_mask.sum()
1202|        loss = verl_F.masked_sum(seq_losses, seq_mask) / global_batch_size * dp_size  # seq-mean
1203|    else:
1204|        raise ValueError(f"Invalid loss_agg_mode: {loss_agg_mode}")
1205|
1206|    return loss

1211|    old_log_prob,
1212|    log_prob,
1213|    advantages,

1219|    loss_agg_mode: str = "token-mean",
1220|):
1221|    """
1222|    Compute the clipped policy objective and related metrics for PPO.
1223|
1224|    Adapted from
1225|    https://github.com/huggingface/trl/blob/main/trl/trainer/ppo_trainer.py#L1122
1226|
1227|    Args:
1228|        old_log_prob (torch.Tensor):
1229|            Log-probabilities of actions under the old policy, shape (batch_size, response_length).
1230|        log_prob (torch.Tensor):
1231|            Log-probabilities of actions under the current policy, shape (batch_size, response_length).
1232|        advantages (torch.Tensor):
1233|            Advantage estimates for each action, shape (batch_size, response_length).
1234|        response_mask (torch.Tensor):
1235|            Mask indicating which tokens to include in the loss, shape (batch_size, response_length).
1236|        cliprange (float, optional):
1237|            Clipping parameter ε for standard PPO. See https://arxiv.org/abs/1707.06347.
1238|            Defaults to None (must be provided).
1239|        cliprange_low (float, optional):
1240|            Lower clip range for dual-clip PPO. Defaults to same as `cliprange`.
1241|        cliprange_high (float, optional):
1242|            Upper clip range for dual-clip PPO. Defaults to same as `cliprange`.
1243|        clip_ratio_c (float, optional):
1244|            Lower bound of the ratio for dual-clip PPO. See https://arxiv.org/pdf/1912.09729.
1245|            Defaults to 3.0.
1246|        loss_agg_mode (str, optional):
1247|            Aggregation mode for `agg_loss`. Defaults to "token-mean".
1248|    """
1249|    assert clip_ratio_c > 1.0, (

1251|        + f" but get the value: {clip_ratio_c}."
1252|    )
1253|
1254|    negative_approx_kl = log_prob - old_log_prob
1255|    # Clamp negative_approx_kl for stability
1256|    negative_approx_kl = torch.clamp(negative_approx_kl, min=-20.0, max=20.0)
1257|    ratio = torch.exp(negative_approx_kl)
1258|    ppo_kl = verl_F.masked_mean(-negative_approx_kl, response_mask)
1259|
1260|    pg_losses1 = -advantages * ratio
1261|    if cliprange_low is None:
1262|        cliprange_low = cliprange
1263|    if cliprange_high is None:
1264|        cliprange_high = cliprange
1265|    pg_losses2 = -advantages * torch.clamp(
1266|        ratio, 1 - cliprange_low, 1 + cliprange_high
1267|    )  # - clip(ratio, 1-cliprange, 1+cliprange) * A
1268|    clip_pg_losses1 = torch.maximum(
1269|        pg_losses1, pg_losses2
1270|    )  # max(-ratio * A, -clip(ratio, 1-cliprange, 1+cliprange) * A)

1272|

1274|    clip_pg_losses2 = torch.min(pg_losses3, clip_pg_losses1)

1277|    )
