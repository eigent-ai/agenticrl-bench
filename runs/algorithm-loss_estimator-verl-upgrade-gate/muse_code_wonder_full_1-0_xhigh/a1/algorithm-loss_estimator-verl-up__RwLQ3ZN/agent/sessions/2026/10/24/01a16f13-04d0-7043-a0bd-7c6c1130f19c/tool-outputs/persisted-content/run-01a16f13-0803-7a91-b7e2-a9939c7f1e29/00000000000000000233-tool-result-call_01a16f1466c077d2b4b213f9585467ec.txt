946|    # where |T| is the number of tokens generated by the model
947|    mean_log_prob_training = verl_F.masked_mean(old_log_prob, response_mask, axis=-1)  # (batch_size,)
948|    training_ppl = torch.exp(-mean_log_prob_training).mean()  # Batch mean of per-sequence PPL
949|    metrics["training_ppl"] = training_ppl.detach().item()
950|
951|    # Also log log-ppl for easier analysis (avoids exponential scale)
952|    metrics["training_log_ppl"] = (-mean_log_prob_training).mean().detach().item()
953|
954|    # 2. Compute rollout off-policy metrics (only if rollout_log_probs available)
955|    if rollout_log_prob is not None:
956|        # 2a. kl: Direct estimator for KL(π_rollout || π_training)
957|        # This is the standard KL divergence: E[log(π_rollout) - log(π_training)]
958|        # Positive value means rollout policy is more confident than training policy
959|        metrics["kl"] = verl_F.masked_mean(rollout_log_prob - old_log_prob, response_mask).detach().item()
960|
961|        # 2b. k3_kl: K3 estimator for KL(π_rollout || π_training)
962|        # More stable for small KL values using: E[exp(log_ratio) - log_ratio - 1]
963|        # Formula: KL ≈ E[r - log(r) - 1] where r = π_training/π_rollout
964|        log_ratio = old_log_prob - rollout_log_prob
965|        k3_kl_matrix = torch.exp(log_ratio) - log_ratio - 1
966|        metrics["k3_kl"] = verl_F.masked_mean(k3_kl_matrix, response_mask).detach().item()
967|
968|        # 2c. Rollout policy perplexity
969|        mean_log_prob_rollout = verl_F.masked_mean(rollout_log_prob, response_mask, axis=-1)  # (batch_size,)
970|        rollout_ppl = torch.exp(-mean_log_prob_rollout).mean()  # Batch mean of per-sequence PPL
971|        metrics["rollout_ppl"] = rollout_ppl.detach().item()
972|        metrics["rollout_log_ppl"] = (-mean_log_prob_rollout).mean().detach().item()
973|
974|        # 2d. Log PPL difference (sequence-level perplexity difference)
975|        # log_ppl_diff = mean_log_prob_rollout - mean_log_prob_training
976|        # Since ppl = exp(-log_prob), we have:
977|        #   log(ppl_ratio) = log(training_ppl/rollout_ppl) = log_ppl_diff
978|        # Positive value means training assigns lower probability (higher PPL) than rollout
979|        log_ppl_diff = mean_log_prob_rollout - mean_log_prob_training
980|        metrics["log_ppl_diff"] = log_ppl_diff.mean().detach().item()
981|        metrics["log_ppl_abs_diff"] = log_ppl_diff.abs().mean().detach().item()
982|        metrics["log_ppl_diff_max"] = log_ppl_diff.max().detach().item()
983|        metrics["log_ppl_diff_min"] = log_ppl_diff.min().detach().item()
984|
985|        # 2e. PPL ratio (how much higher is training PPL vs rollout PPL)
986|        # Compute per-sequence ratio first, then average
987|        # For numerical stability, compute in log space using log_ppl_diff
988|        # Note: log_ppl_diff = log(ppl_ratio), so ppl_ratio = exp(log_ppl_diff)
989|        # This is the inverse of geometric IS: ppl_ratio_i = 1 / geometric_is_i for each sequence
990|        ppl_ratio = torch.exp(log_ppl_diff).mean()  # mean(exp(log_ppl_diff)) = mean(ppl_ratio_i)
991|        metrics["ppl_ratio"] = ppl_ratio.detach().item()
992|
993|        # 2f. Chi-squared divergence: χ²(π_training || π_rollout) = E_μ[ρ²] - 1
994|        # where ρ = π_training / π_rollout and μ = π_rollout (rollout distribution)
995|        # This measures the variance of importance sampling weights
996|        # Token-level: E_token[ρ²] - 1 (averaged over all tokens)
997|        log_ratio_safe = torch.clamp(log_ratio, min=-SAFETY_BOUND, max=SAFETY_BOUND)
998|        rho_token = torch.exp(log_ratio_safe)  # ρ = π_training / π_rollout (token-level)
999|        rho_squared_token = rho_token.square()
1000|        chi2_token = verl_F.masked_mean(rho_squared_token, response_mask) - 1.0
1001|        metrics["chi2_token"] = chi2_token.detach().item()
1002|
1003|        # Sequence-level: E_seq[(Π ρ_t)²] - 1 = E_seq[exp(2 * Σ log ρ_t)] - 1
1004|        log_ratio_sum = verl_F.masked_sum(log_ratio, response_mask, axis=-1)  # Σ log ρ_t per sequence
1005|        log_ratio_sum_safe = torch.clamp(log_ratio_sum, min=-SAFETY_BOUND, max=SAFETY_BOUND)
1006|        rho_squared_seq = torch.exp(2.0 * log_ratio_sum_safe)  # (Π ρ_t)²
1007|        chi2_seq = rho_squared_seq.mean() - 1.0
1008|        metrics["chi2_seq"] = chi2_seq.detach().item()
1009|
1010|    return metrics
