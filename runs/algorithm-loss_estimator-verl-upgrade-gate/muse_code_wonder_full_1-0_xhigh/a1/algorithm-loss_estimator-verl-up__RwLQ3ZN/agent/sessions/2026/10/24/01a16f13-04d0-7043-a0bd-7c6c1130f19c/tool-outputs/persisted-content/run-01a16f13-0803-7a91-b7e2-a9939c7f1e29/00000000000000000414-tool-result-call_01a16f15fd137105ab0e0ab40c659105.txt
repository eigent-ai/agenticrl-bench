181|    def _postprocess_output(self, output, *, global_token_num, delta_time, forward_only, images_seqlens):
182|        """
183|
184|        Args:
185|            output: a dictionary containing loss, model_outputs and metrics
186|
187|        Returns:
188|
189|        """
190|
191|        metrics: dict = output.pop("metrics")
192|        # perform all gather in dp group to ensure that it's correct.
193|        # Here each metric in metrics can be a list (micro-batch metrics) or a singleton
194|        # we should always sum the loss of each micro-batch as we scale by global_bsz/global_token
195|        loss = torch.sum(torch.tensor(output.pop("loss"), device=self.device_name))
196|        dp_group = self.engine.get_data_parallel_group()
197|        if dp_group is not None:
198|            torch.distributed.all_reduce(loss, op=torch.distributed.ReduceOp.AVG, group=dp_group)
199|        loss = loss.item()
200|
201|        # For grad_norm, we do not perform all reduce because it is already been done when clipping grad
202|        grad_norm = metrics.pop("grad_norm", None)
203|        if isinstance(grad_norm, torch.Tensor):
204|            grad_norm = grad_norm.detach().item()
205|        lr = metrics.pop("lr", None)
206|
207|        # For other metrics, we perform all gather in dp group (only if DP > 1)
208|        if dp_group is not None:
209|            final_metrics = allgather_dict_into_dict(data=metrics, group=dp_group)
210|        else:
211|            final_metrics = metrics
212|        final_metrics["loss"] = loss
213|        if grad_norm is not None:
214|            final_metrics["grad_norm"] = grad_norm
215|        if lr is not None:
216|            final_metrics["lr"] = lr
217|
218|        # log memory
219|        final_metrics["perf/max_memory_allocated_gb"] = get_torch_device().max_memory_allocated() / (1024**3)
220|        final_metrics["perf/max_memory_reserved_gb"] = get_torch_device().max_memory_reserved() / (1024**3)
221|        final_metrics["perf/cpu_memory_used_gb"] = psutil.virtual_memory().used / (1024**3)
222|
223|        # TODO: confirm the mtp loss IS same across dp
224|        for k, v in final_metrics.items():
225|            if k.startswith("mtp_losses"):
226|                flatten_v = [sublist[0] for sublist in v]  # sublist should be single element
227|                final_metrics[k] = sum(flatten_v) / len(flatten_v)
228|        # compute mfu
229|        if global_token_num is not None and self.flops_counter is not None:
230|            estimated_flops, promised_flops = self.flops_counter.estimate_flops(
231|                global_token_num, delta_time, images_seqlens=images_seqlens
232|            )
233|            final_metrics["mfu"] = estimated_flops / promised_flops / torch.distributed.get_world_size()
234|            if forward_only:
235|                final_metrics["mfu"] /= 3.0
236|        # model outputs
237|        model_output = output.pop("model_output", {})
238|        # We only return final_metrics
239|        final_output = tu.get_tensordict(tensor_dict=model_output, non_tensor_dict={"metrics": final_metrics})
240|        return final_output

242|    @register(dispatch_mode=make_nd_compute_dataproto_dispatch_fn(mesh_name="train"), blocking=False)
243|    def train_mini_batch(self, data: TensorDict) -> TensorDict:
244|        """Split a batch into N mini-batches run for multiple epochs
245|
246|        Args:
247|            data:
248|
249|        Returns:
250|
251|        """
252|        maybe_fix_3d_position_ids(data)
253|        batch_size_per_dp = data.shape[0]
254|        disable_auto_offload = tu.pop(data, key="disable_auto_offload", default=False)
255|        mini_batch_size = tu.pop(data, key="mini_batch_size", default=None)
256|        num_mini_batch = tu.pop(data, key="num_mini_batch", default=None)
257|        epochs = tu.pop(data, key="epochs", default=1)
258|        seed = tu.pop(data, key="seed", default=42)
259|        dataloader_kwargs = tu.pop(data, key="dataloader_kwargs", default={})
260|
261|        assert mini_batch_size is not None or num_mini_batch is not None
262|
263|        if mini_batch_size is None:
264|            assert batch_size_per_dp % num_mini_batch == 0, f"Got {batch_size_per_dp=} and {num_mini_batch=}"
265|            mini_batch_size_per_gpu = batch_size_per_dp // num_mini_batch
266|        else:
267|            assert mini_batch_size % self.engine.get_data_parallel_size() == 0, (
268|                f"Got {mini_batch_size=} and {self.engine.get_data_parallel_size()=}"
269|            )
270|            mini_batch_size_per_gpu = mini_batch_size // self.engine.get_data_parallel_size()
271|
272|        # make iterator
273|        dataloader = tu.make_iterator(
274|            data,
275|            mini_batch_size=mini_batch_size_per_gpu,
276|            epochs=epochs,
277|            seed=seed + self.engine.get_data_parallel_rank(),
278|            dataloader_kwargs=dataloader_kwargs,
279|        )
280|
281|        with (
282|            self.engine.train_mode(disable_auto_offload=disable_auto_offload),
283|            Timer(name="train_batch", logger=None),
284|        ):
285|            # update
286|            output_lst = []
287|            total_num_iterations = data.shape[0] // mini_batch_size_per_gpu * epochs
288|
289|            for batch_idx, mini_batch_td in enumerate(dataloader):
290|                maybe_fix_3d_position_ids(mini_batch_td)
291|                # add global token num
292|                if "input_ids" in mini_batch_td:
293|                    global_token_num = mini_batch_td["input_ids"].offsets().diff().tolist()  # (total_nnz,)
294|                    # allgather from dp rank
295|                    global_token_num_output = [None] * torch.distributed.get_world_size(
296|                        self.engine.get_data_parallel_group()
297|                    )
298|                    torch.distributed.all_gather_object(
299|                        global_token_num_output, global_token_num, self.engine.get_data_parallel_group()
300|                    )
301|                    global_token_num = [x for xs in global_token_num_output for x in xs]
302|                else:
303|                    global_token_num = None
304|
305|                tu.assign_non_tensor(
306|                    mini_batch_td,
307|                    global_token_num=NonTensorData(global_token_num),
308|                    update_lr_scheduler=batch_idx == total_num_iterations - 1,
309|                    disable_auto_offload=True,
310|                )
311|                # Name the mini-batch in the trace so each iteration of this otherwise identical
312|                # loop is distinguishable when the whole step is recorded (continuous mode).
313|                with torch.profiler.record_function(f"mini_batch{batch_idx}"):
314|                    actor_output = self.train_batch(mini_batch_td)
315|                # Advance the profiler once per mini-batch. This drives any
316|                # torch.profiler.schedule (the schedule's unit is one update mini-batch) and,
317|                # with no schedule, only labels torch's ProfilerStep#<n> boundary. start()/stop()
318|                # happen on the outer worker, but the profiler is process-global so this reaches it.
319|                self.profiler.step()
320|                output_lst.append(actor_output)
321|
322|            if self.engine.is_mp_src_rank_with_outputs():
323|                actor_output = [tu.get(output, "metrics") for output in output_lst]
324|                metrics = {}
325|                for output in actor_output:
326|                    for key, val in output.items():
327|                        # flattn dp and micro batch
328|                        if isinstance(val, list):
329|                            output[key] = (
330|                                Metric.aggregate_dp(val)
331|                                if isinstance(val[0], Metric)
332|                                else list(chain.from_iterable(val))
333|                            )
334|                    append_to_dict(metrics, output)
335|
336|                output = tu.get_tensordict(tensor_dict={}, non_tensor_dict={"metrics": metrics}).cpu()
337|            else:
338|                output = None
339|        return output

342|    @DistProfiler.annotate(color="red", role="train_batch")
343|    def train_batch(self, data: TensorDict) -> TensorDict:
344|        assert self.loss_fn is not None, "loss function can't be None when calling train_batch"
345|        assert not self.engine_config.forward_only, "Can't run `train_batch` when forward_only is in the engine config."

347|        global_token_num = tu.get(data, key="global_token_num")
348|        disable_auto_offload = tu.get(data, key="disable_auto_offload", default=False)
349|        images_seqlens = tu.get(data, key="images_seqlens", default=None)
350|
351|        # inject engineering parameters if not specified
352|        default_keys = dict(
353|            use_remove_padding=self.model_config.get("use_remove_padding", False),
354|            use_dynamic_bsz=self.engine_config.use_dynamic_bsz,
355|            max_token_len_per_gpu=self.engine_config.max_token_len_per_gpu,

358|        )
359|
360|        for key, val in default_keys.items():

362|                tu.assign_non_tensor(data, **{key: val})

366|            Timer(name="train_batch", logger=None) as timer,

370|            # for training, we only care about loss and metrics

372|

386|                output,
387|                global_token_num=global_token_num,
388|                delta_time=delta_time,
389|                forward_only=False,
390|                images_seqlens=images_seqlens,
391|            ).cpu()
392|        else:
393|            final_output = None
394|
395|        return final_output
