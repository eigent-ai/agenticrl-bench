1802|
1803|    if cliprange_low is None:
1804|        cliprange_low = cliprange
1805|    if cliprange_high is None:
1806|        cliprange_high = cliprange

1869|        advantages (torch.Tensor):

1929|def compute_policy_loss_geo_mean(
1930|    old_log_prob: torch.Tensor,
1931|    log_prob: torch.Tensor,
1932|    advantages: torch.Tensor,
1933|    response_mask: torch.Tensor,

1938|    """

1940|
1941|    Adapted from paper https://arxiv.org/abs/2507.20673
1942|    https://github.com/callsys/GMPO/blob/main/train_zero_math_gmpo.py
1943|
1944|    Args:
1945|        old_log_prob (torch.Tensor):
1946|            Log-probabilities of actions under the old policy, shape (batch_size, response_length).
1947|        log_prob (torch.Tensor):
1948|            Log-probabilities of actions under the current policy, shape (batch_size, response_length).
1949|        advantages (torch.Tensor):
1950|            Advantage estimates for each action, shape (batch_size, response_length).
1951|        response_mask (torch.Tensor):
1952|            Mask indicating which tokens to include in the loss, shape (batch_size, response_length).
1953|        loss_agg_mode (str, optional):
1954|            not used
1955|    """
1956|
1957|    assert config is not None
1958|    assert not isinstance(config, AlgoConfig)
1959|    clip_ratio = config.clip_ratio  # Clipping parameter. See https://arxiv.org/abs/1707.06347.
1960|    clip_ratio_low = config.clip_ratio_low if config.clip_ratio_low is not None else clip_ratio
1961|    clip_ratio_high = config.clip_ratio_high if config.clip_ratio_high is not None else clip_ratio
1962|
1963|    cliprange = clip_ratio
1964|    cliprange_low = clip_ratio_low
1965|    cliprange_high = clip_ratio_high
1966|    if cliprange_low is None:
1967|        cliprange_low = cliprange
1968|    if cliprange_high is None:
1969|        cliprange_high = cliprange
1970|

1972|    # Clamp negative_approx_kl for stability (uncomment it if you like)

1974|    ppo_kl = verl_F.masked_mean(-negative_approx_kl, response_mask)
1975|
1976|    # Clipping at token-level & Clipping wider
1977|    sgn_advantage = torch.sign(advantages)
1978|    negative_approx_kl_clamp = torch.clamp(negative_approx_kl, -cliprange_low, cliprange_high)
1979|    negative_approx_kl_min = torch.min(sgn_advantage * negative_approx_kl, sgn_advantage * negative_approx_kl_clamp)

1982|    # Geometric-Mean Policy Optimization

2008|        "actor/ppo_kl": ppo_kl.detach().item(),
