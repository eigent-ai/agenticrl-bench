47|@dataclass
48|class RouterReplayConfig(BaseConfig):
49|    """Configuration for router replay in MoE models.
50|
51|    This configuration controls the routing behavior for Mixture of Experts (MoE) models,
52|    allowing for deterministic training through route recording and replay.
53|
54|    Args:
55|        mode (str): Router replay mode. Options: 'disabled', 'R2', 'R3'.
56|            - 'disabled': No router replay functionality
57|            - 'R2': Use Router Replay routing strategy
58|            - 'R3': Use Rollout Router Replay routing strategy
59|        record_file (Optional[str]): File path to save recorded routing decisions.
60|            Required when mode is 'record', 'R2', or 'R3'.
61|        replay_file (Optional[str]): File path to load recorded routing decisions for replay.
62|            Required when mode is 'replay'.
63|    """
64|
65|    mode: str = "disabled"
66|    record_file: Optional[str] = None
67|    replay_file: Optional[str] = None
68|
69|    def __post_init__(self):
70|        """Validate router replay configuration."""
71|        valid_modes = ["disabled", "R2", "R3"]
72|        if self.mode not in valid_modes:
73|            raise ValueError(f"Invalid router_replay mode: {self.mode}. Must be one of {valid_modes}")

104|@dataclass
105|class ActorConfig(BaseConfig):
106|    """Configuration for actor model training.
107|
108|    The inheritance from BaseConfig provides omegaconf.DictConfig-like interface for a dataclass config.
109|
110|    Args:
111|        strategy (str): Training strategy. Must be specified.
112|        ppo_mini_batch_size (int): Mini-batch size for PPO training.
113|        ppo_micro_batch_size (Optional[int]): Micro-batch size for PPO training.
114|            If None, uses ppo_micro_batch_size_per_gpu.
115|        ppo_micro_batch_size_per_gpu (Optional[int]): Micro-batch size per GPU for PPO training.
116|        use_dynamic_bsz (bool): Whether to use dynamic batch sizing.
117|        ppo_max_token_len_per_gpu (int): Maximum token length per GPU for PPO training.
118|        clip_ratio (float): PPO clipping ratio for policy loss.
119|        clip_ratio_low (float): Lower bound for PPO clipping ratio.
120|        clip_ratio_high (float): Upper bound for PPO clipping ratio.
121|        policy_loss (PolicyLossConfig): Configuration for policy loss computation.
122|        clip_ratio_c (float): Clipping ratio for critic loss.
123|        loss_agg_mode (str): Loss aggregation mode, including 'token-mean', 'token-sum', and sequence modes.
124|        loss_scale_factor (Optional[int]): Scale factor for 'seq-mean-token-sum-norm' loss aggregation mode.
125|            If None, uses response_length. Set to a constant to ensure consistent normalization.
126|        entropy_coeff (float): Entropy coefficient for regularization.
127|        tau_pos (float): Positive tau for SAPO smoothing (>= 1.0 keeps rewards stable).
128|        tau_neg (float): Negative tau for SAPO smoothing (> tau_pos for asymmetry).
129|        use_kl_loss (bool): Whether to use KL divergence loss.
130|        use_torch_compile (bool): Whether to use torch.compile for optimization.
131|        kl_loss_coef (float): KL divergence loss coefficient.
132|        kl_loss_type (str): Type of KL loss to use.
133|        ppo_epochs (int): Number of PPO epochs per training step.
134|        shuffle (bool): Whether to shuffle data during training.
135|        checkpoint (CheckpointConfig): Configuration for checkpointing.
136|        optim (OptimizerConfig): Configuration for optimizer.
137|        use_fused_kernels (bool): Whether to use custom fused kernels (e.g., FlashAttention, fused MLP).
138|        data_loader_seed (int): Seed for data loader. If None, uses global seed.
139|    """
140|
141|    _mutable_fields = BaseConfig._mutable_fields | {
142|        "ppo_mini_batch_size",
143|        "ppo_micro_batch_size",
144|        "ppo_micro_batch_size_per_gpu",
145|        "ppo_infer_micro_batch_size_per_gpu",
146|        "engine",
147|        "model_config",
148|    }
149|
150|    strategy: str = MISSING
151|    ppo_mini_batch_size: int = 256
152|    ppo_micro_batch_size: Optional[int] = None  # deprecate
153|    ppo_micro_batch_size_per_gpu: Optional[int] = None
154|    ppo_infer_micro_batch_size_per_gpu: Optional[int] = None
155|    use_dynamic_bsz: bool = False
156|    ppo_max_token_len_per_gpu: int = 16384
157|    ppo_infer_max_token_len_per_gpu: int = 16384
158|    clip_ratio: float = 0.2
159|    clip_ratio_low: float = 0.2
160|    clip_ratio_high: float = 0.2
161|    freeze_vision_tower: bool = False
162|    policy_loss: PolicyLossConfig = field(default_factory=PolicyLossConfig)
163|    clip_ratio_c: float = 3.0
164|    loss_agg_mode: str = "token-mean"
165|    loss_scale_factor: Optional[int] = None
166|    entropy_coeff: float = 0
167|    tau_pos: float = 1.0
168|    tau_neg: float = 1.05
169|    calculate_entropy: bool = False
170|    calculate_sum_pi_squared: bool = False
171|    use_kl_loss: bool = False
172|    # Whether to enable PrefixGrouper-based shared-prefix forward
173|    use_prefix_grouper: bool = False
174|    use_torch_compile: bool = True
175|    kl_loss_coef: float = 0.001
176|    kl_loss_type: str = "low_var_kl"
177|    ppo_epochs: int = 1
178|    shuffle: bool = False
179|    data_loader_seed: int = 42
180|    checkpoint: CheckpointConfig = field(default_factory=CheckpointConfig)
181|    optim: OptimizerConfig = field(default_factory=OptimizerConfig)
182|    use_fused_kernels: bool = False
183|    profiler: ProfilerConfig = field(default_factory=ProfilerConfig)
184|    engine: BaseConfig = field(default_factory=BaseConfig)
185|    rollout_n: int = MISSING  # must be override by sampling config
186|    model_config: HFModelConfig = field(default_factory=BaseConfig)
187|
188|    # Store global batch info for loss aggregation:
189|    # dp_size: data parallel size
190|    # batch_num_tokens: number of valid tokens in global batch
191|    # global_batch_size: global batch size
192|    global_batch_info: dict = field(default_factory=dict)
193|    qat: QATConfig = field(default_factory=QATConfig)
194|
195|    def __post_init__(self):
196|        """Validate actor configuration parameters."""
197|        assert self.strategy != MISSING
198|        assert self.rollout_n != MISSING
199|        if not self.use_dynamic_bsz:
200|            if self.ppo_micro_batch_size is not None and self.ppo_micro_batch_size_per_gpu is not None:
201|                raise ValueError(
202|                    "[actor] You have set both 'actor.ppo_micro_batch_size' AND 'actor.ppo_micro_batch_size_per_gpu'. "
203|                    "Please remove 'actor.ppo_micro_batch_size' because only '*_ppo_micro_batch_size_per_gpu' is "
204|                    "supported (the former is deprecated)."
205|                )
206|            else:
207|                assert not (self.ppo_micro_batch_size is None and self.ppo_micro_batch_size_per_gpu is None), (
208|                    "[actor] Please set at least one of 'actor.ppo_micro_batch_size' or "
209|                    "'actor.ppo_micro_batch_size_per_gpu' if use_dynamic_bsz is not enabled."
210|                )
211|
212|        valid_loss_agg_modes = [
213|            "token-mean",
214|            "token-sum",
215|            "seq-mean-token-sum",
216|            "seq-mean-token-mean",
217|            "seq-mean-token-sum-norm",
218|        ]
219|        if self.loss_agg_mode not in valid_loss_agg_modes:
220|            raise ValueError(f"Invalid loss_agg_mode: {self.loss_agg_mode}")
221|
222|    def validate(self, n_gpus: int, train_batch_size: int, model_config: dict = None):
223|        """Validate actor configuration with runtime parameters."""
224|        if not self.use_dynamic_bsz:
225|            if train_batch_size < self.ppo_mini_batch_size:
226|                raise ValueError(
227|                    f"train_batch_size ({train_batch_size}) must be >= "
228|                    f"actor.ppo_mini_batch_size ({self.ppo_mini_batch_size})"
229|                )
230|
231|            sp_size = getattr(self, "ulysses_sequence_parallel_size", 1)
232|            if self.ppo_micro_batch_size is not None:
233|                if self.ppo_mini_batch_size % self.ppo_micro_batch_size != 0:
234|                    raise ValueError(
235|                        f"ppo_mini_batch_size ({self.ppo_mini_batch_size}) must be divisible by "
236|                        f"ppo_micro_batch_size ({self.ppo_micro_batch_size})"
237|                    )
238|                if self.ppo_micro_batch_size * sp_size < n_gpus:
239|                    raise ValueError(
240|                        f"ppo_micro_batch_size ({self.ppo_micro_batch_size}) * "
241|                        f"ulysses_sequence_parallel_size ({sp_size}) must be >= n_gpus ({n_gpus})"
242|                    )
243|
244|    @staticmethod
245|    def _check_mutually_exclusive(mbs, mbs_per_gpu, name: str):
246|        """Validate mutually exclusive micro batch size configuration options."""
247|        param = "ppo_micro_batch_size"
248|        param_per_gpu = f"{param}_per_gpu"
249|
250|        if mbs is None and mbs_per_gpu is None:
251|            raise ValueError(f"[{name}] Please set at least one of '{name}.{param}' or '{name}.{param_per_gpu}'.")
252|
253|        if mbs is not None and mbs_per_gpu is not None:
254|            raise ValueError(
255|                f"[{name}] You have set both '{name}.{param}' AND '{name}.{param_per_gpu}'. Please remove "
256|                f"'{name}.{param}' because only '*_{param_per_gpu}' is supported (the former is deprecated)."
257|            )

288|@dataclass
289|class FSDPActorConfig(ActorConfig):
290|    """Configuration for FSDP actor models.
291|
292|    The inheritance from BaseConfig provides omegaconf.DictConfig-like interface for a dataclass config.
293|
294|    Args:
295|        strategy (str): Training strategy set to 'fsdp' for Fully Sharded Data Parallel.
296|        grad_clip (float): Gradient clipping threshold.
297|        ulysses_sequence_parallel_size (int): [DEPRECATED] Ulysses sequence parallel size for long sequences.
298|        entropy_from_logits_with_chunking (bool): Whether to compute entropy from logits
299|            with chunking for memory efficiency.
300|        entropy_checkpointing (bool): Whether to use gradient checkpointing for entropy computation.
301|        pad_to_length (bool): Whether to pad every packed micro-batch to a static token count.
302|            Forwarded to ``fsdp_config.pad_to_length``, which is what the engine reads.
303|        fsdp_config (dict[str, Any]): Configuration for FSDP settings.
304|        use_remove_padding (bool): Whether to remove padding tokens in inputs during training
305|    """
306|
307|    strategy: str = "fsdp"
308|    grad_clip: float = 1.0
309|    ulysses_sequence_parallel_size: int = 1
310|    entropy_from_logits_with_chunking: bool = False
311|    entropy_from_logits_chunk_size: int = 2048
312|    entropy_checkpointing: bool = False
313|    pad_to_length: bool = False
314|    fsdp_config: FSDPEngineConfig = field(default_factory=FSDPEngineConfig)
315|    use_remove_padding: bool = False
316|    use_rollout_log_probs: bool = False
317|
318|    def __post_init__(self):
319|        """Validate FSDP actor configuration parameters."""
320|        super().__post_init__()
321|        self.engine = self.fsdp_config
322|        # Sync strategy to engine config so engine_workers can pick the right FSDP version.
323|        # EngineConfig.strategy defaults to None, so without this, engine_workers.py always
324|        # falls back to FSDP1 even when actor.strategy="fsdp2".
325|        object.__setattr__(self.engine, "strategy", self.strategy)
326|
327|        # backward compatibility
328|        if self.ulysses_sequence_parallel_size > 1:
329|            self.fsdp_config.ulysses_sequence_parallel_size = self.ulysses_sequence_parallel_size
330|
331|    def validate(self, n_gpus: int, train_batch_size: int, model_config: dict = None):
332|        """Validate FSDP actor configuration with runtime parameters."""
333|        super().validate(n_gpus, train_batch_size, model_config)
334|        if (
335|            self.ulysses_sequence_parallel_size > 1
336|            and model_config
337|            and not model_config.get("use_remove_padding", False)
338|        ):
339|            raise ValueError(
340|                "When using sequence parallelism for actor/ref policy, you must enable `use_remove_padding`."
341|            )
