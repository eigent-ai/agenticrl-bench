122|        output["responses"] = torch.tensor(output.pop("response_ids"), dtype=torch.int64)
123|        output["response_mask"] = torch.tensor(output.pop("response_mask"), dtype=torch.int64)
124|
125|        response_logprobs = output.pop("response_logprobs", None)
126|        if response_logprobs is not None:
127|            output["rollout_log_probs"] = torch.tensor(response_logprobs, dtype=torch.float32)
128|
129|        routed_experts = output.pop("routed_experts", None)
130|        if routed_experts is not None:
131|            routed_experts = torch.tensor(routed_experts, dtype=torch.int16)
132|            # Router replay indexes this field by absolute token position, so it must
133|            # span the whole sequence. The rollout engine records fewer rows than that:
134|            # it only sees tokens fed through the model, and multi-turn loops stop
135|            # recording at the last generation. Trailing rows stay zero; replay masks
136|            # them out instead of consuming them.
137|            total_length = output["prompts"].size(0) + output["responses"].size(0)
138|            aligned = routed_experts.new_zeros((total_length, *routed_experts.shape[1:]))
139|            num_rows = min(routed_experts.size(0), total_length)
140|            aligned[:num_rows] = routed_experts[:num_rows]
141|            output["routed_experts"] = aligned
142|
143|        # rm_scores: reward score for each token
144|        reward_score = output.pop("reward_score", None)
145|        if reward_score is not None:
146|            rm_scores = torch.zeros_like(output["response_mask"], dtype=torch.float32)
147|            rm_scores[-1] = reward_score
148|            output["rm_scores"] = rm_scores

150|        # Mutating a default dict does not add the field to Pydantic's fields-set,
151|        # so model_dump(exclude_unset=True) can omit populated extra fields.
152|        extra_fields = output.setdefault("extra_fields", self.extra_fields.copy())
153|        teacher_ids, teacher_logprobs = (
