28|def sft_loss(config: ActorConfig, model_output, data: TensorDict, dp_group=None):
29|    pad_mode = tu.get_non_tensor_data(data=data, key="pad_mode", default=DatasetPadMode.NO_PADDING)
30|    dp_size = data["dp_size"]
31|    batch_num_tokens = data["batch_num_tokens"]
32|
33|    log_prob = model_output["log_probs"]
34|
35|    if pad_mode == DatasetPadMode.NO_PADDING:
36|        # log_prob and loss mask are nested tensors of shape [bsz, j1]
37|        # for each sample, loss mask shape is [1, prompt_length + response_length]
38|        loss_mask = data["loss_mask"]
39|
40|        log_prob_flatten = log_prob.values()
41|        loss_mask_flatten = loss_mask.values()
42|
43|        # left-shift the loss mask by one token to align with log_prob
44|        loss_mask_flatten = torch.roll(loss_mask_flatten, shifts=-1, dims=0)
45|
46|        # NOTE: loss is averaged over all tokens in the batch across all data parallel groups,
47|        # For FSDP backend, the loss is directly used for backward; while for Megatron backend,
48|        # the loss should be scaled by `num_microbatches` for pp schedule.
49|        loss = -masked_sum(log_prob_flatten, loss_mask_flatten) / batch_num_tokens * dp_size
50|    else:
51|        response_mask = data["response_mask"].to(bool)
52|        loss = -masked_sum(log_prob, response_mask) / batch_num_tokens * dp_size
53|
54|    return loss, {}

57|def ppo_loss(config: ActorConfig, model_output, data: TensorDict, dp_group=None):
58|    """Computes ppo loss from model output (log_prob, entropy, values, etc. ) and old_log_probs from data."""
59|    log_prob = no_padding_2_padding(model_output["log_probs"], data)
60|    entropy = model_output.get("entropy", None)
61|    if entropy is not None:
62|        entropy = no_padding_2_padding(entropy, data)
63|
64|    # global batch info for loss aggregation
65|    config.global_batch_info["dp_size"] = data["dp_size"]
66|    config.global_batch_info["batch_num_tokens"] = data["batch_num_tokens"]
67|    config.global_batch_info["global_batch_size"] = data["global_batch_size"]
68|    config.global_batch_info["loss_scale_factor"] = config.loss_scale_factor
69|
70|    # assumes that if any of the global batch info is set, the policy_loss_fn will
71|    # normalize using dp_size/global_bsz/global_token; in this case, metric aggregation should be SUM
72|    # to reflect the mean loss over the global batch
73|    if (
74|        data["dp_size"] > 1
75|        or data["batch_num_tokens"] is not None
76|        or data["global_batch_size"] is not None
77|        or config.loss_scale_factor is not None
78|    ):
79|        metric_aggregation = AggregationType.SUM
80|    else:
81|        metric_aggregation = AggregationType.MEAN
82|
83|    metrics = {}
84|
85|    # select fields and convert to padded tensor
86|    fields = ["response_mask", "old_log_probs", "advantages"]
87|    if "rollout_is_weights" in data:
88|        fields.append("rollout_is_weights")
89|    if "ref_log_prob" in data:
90|        fields.append("ref_log_prob")
91|    data = data.select(*fields).to_padded_tensor()
92|
93|    response_mask = data["response_mask"].to(bool)
94|    # compute policy loss
95|    old_log_prob = data["old_log_probs"]
96|    advantages = data["advantages"]
97|    rollout_is_weights = data.get("rollout_is_weights", None)
98|
99|    loss_agg_mode = config.loss_agg_mode
100|
101|    loss_mode = config.policy_loss.get("loss_mode", "vanilla")
102|
103|    policy_loss_fn = get_policy_loss_fn(loss_mode)
104|    pg_loss, pg_metrics = policy_loss_fn(
105|        old_log_prob=old_log_prob,
106|        log_prob=log_prob,
107|        advantages=advantages,
108|        response_mask=response_mask,
109|        loss_agg_mode=loss_agg_mode,
110|        config=config,
111|        rollout_is_weights=rollout_is_weights,
112|    )
113|
114|    # AggregationType.MEAN for pg metrics: assumes policy_loss_fn normalizes by local_bsz/local_tokens
115|    # Ex: in compute_policy_loss_vanilla, pg_metrics are pg_clipfrac, ppo_kl, pg_clipfrac_lower
116|    pg_metrics = Metric.from_dict(pg_metrics, aggregation=AggregationType.MEAN)
117|
118|    metrics.update(pg_metrics)
119|    metrics["actor/pg_loss"] = Metric(value=pg_loss, aggregation=metric_aggregation)
120|    policy_loss = pg_loss
121|
122|    # add entropy loss
123|    if entropy is not None:
124|        entropy_loss = agg_loss(
125|            loss_mat=entropy, loss_mask=response_mask, loss_agg_mode=loss_agg_mode, **config.global_batch_info
126|        )
127|        entropy_coeff = config.entropy_coeff
128|        policy_loss -= entropy_coeff * entropy_loss
129|        metrics["actor/entropy_loss"] = Metric(value=entropy_loss, aggregation=metric_aggregation)
130|
131|    # add kl loss
132|    if config.use_kl_loss:
133|        ref_log_prob = data["ref_log_prob"]
134|        # compute kl loss
135|        kld = kl_penalty(logprob=log_prob, ref_logprob=ref_log_prob, kl_penalty=config.kl_loss_type)
136|        kl_loss = agg_loss(
137|            loss_mat=kld, loss_mask=response_mask, loss_agg_mode=config.loss_agg_mode, **config.global_batch_info
138|        )
139|
140|        policy_loss += kl_loss * config.kl_loss_coef
141|        metrics["kl_loss"] = Metric(value=kl_loss, aggregation=metric_aggregation)
142|        metrics["kl_coef"] = config.kl_loss_coef
143|
144|    return policy_loss, metrics

147|def value_loss(config: CriticConfig, model_output, data: TensorDict, dp_group=None):
148|    """value loss
149|
150|    Args:
151|        config: CriticConfig
152|        model_output: model output from the model
153|        data: the input to the model
154|        dp_group: data paralle group
155|
156|    Returns:
157|        value loss
158|    """
159|    vpreds = no_padding_2_padding(model_output["values"], data)  # (bsz, response_length)
160|
161|    # Normalize the value loss over the global mini-batch (dp_size / batch_num_tokens /
162|    # global_batch_size) instead of the local micro-batch, so the accumulated critic gradient is
163|    # invariant to how the mini-batch is split into micro-batches (as the actor's ppo_loss does).
164|    dp_size = data["dp_size"]
165|    batch_num_tokens = data["batch_num_tokens"]
166|    global_batch_size = data["global_batch_size"]
167|
168|    # When the loss is normalized over the global batch, each micro-batch contributes a partial sum,
169|    # so the loss metric must be aggregated with SUM to reflect the global-batch mean.
170|    if (
171|        dp_size > 1
172|        or batch_num_tokens is not None
173|        or global_batch_size is not None
174|        or config.loss_scale_factor is not None
175|    ):
176|        metric_aggregation = AggregationType.SUM
177|    else:
178|        metric_aggregation = AggregationType.MEAN
179|
180|    # select fields and convert to padded tensor
181|    data = data.select("values", "returns", "response_mask").to_padded_tensor()
182|    values = data["values"]
183|    returns = data["returns"]
184|    response_mask = data["response_mask"].to(bool)
185|
186|    vf_loss, vf_clipfrac = compute_value_loss(
187|        vpreds=vpreds,
188|        values=values,
189|        returns=returns,
190|        response_mask=response_mask,
191|        cliprange_value=config.cliprange_value,
192|        loss_agg_mode=config.loss_agg_mode,
193|        dp_size=dp_size,
194|        batch_num_tokens=batch_num_tokens,
195|        global_batch_size=global_batch_size,
196|        loss_scale_factor=config.loss_scale_factor,
197|    )
198|
199|    metrics = {
200|        "critic/vf_loss": Metric(value=vf_loss, aggregation=metric_aggregation),
201|        "critic/vf_clipfrac": vf_clipfrac.detach().item(),
202|        "critic/vpred_mean": masked_mean(vpreds, response_mask).detach().item(),
203|    }
204|
205|    return vf_loss, metrics
