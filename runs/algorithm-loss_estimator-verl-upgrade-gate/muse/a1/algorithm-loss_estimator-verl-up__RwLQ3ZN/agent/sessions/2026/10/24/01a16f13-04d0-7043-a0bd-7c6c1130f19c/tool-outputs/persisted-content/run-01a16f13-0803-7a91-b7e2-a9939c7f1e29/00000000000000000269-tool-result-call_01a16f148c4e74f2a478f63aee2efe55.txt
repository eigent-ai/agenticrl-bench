47|class DistillationLossSettings(BaseConfig):
48|    """
49|    Settings for a distillation loss function to be registered.
50|
51|    Args:
52|        names (str | list[str]): Name(s) to register the distillation loss function under.
53|        use_topk (bool): Whether the loss function uses top-k log probabilities.

65|        if sum([self.use_topk, self.use_estimator]) != 1:
66|            raise ValueError(

75|def register_distillation_loss(
76|    loss_settings: DistillationLossSettings,
77|) -> Callable[[DistillationLossFn], DistillationLossFn]:
78|    """Register a distillation loss function with the given name."""
79|
80|    def decorator(func: DistillationLossFn) -> DistillationLossFn:
81|        for name in loss_settings.names:
82|            if name in DISTILLATION_LOSS_REGISTRY:
83|                raise ValueError(f"Distillation loss function with name '{name}' is already registered.")
84|            DISTILLATION_LOSS_REGISTRY[name] = func
85|            DISTILLATION_SETTINGS_REGISTRY[name] = loss_settings
86|        return func
87|
88|    return decorator

109|def compute_distillation_loss_range(
110|    distillation_losses: torch.Tensor, response_mask: torch.Tensor
111|) -> dict[str, Metric]:
112|    """Compute min and max distillation loss over valid response tokens."""
113|    if response_mask.is_nested:
114|        distillation_losses_response = distillation_losses[response_mask.bool().to_padded_tensor(False)]
115|    else:
116|        distillation_losses_response = distillation_losses[response_mask.bool()]
117|    return {
118|        "distillation/loss_min": Metric(AggregationType.MIN, distillation_losses_response.min()),
119|        "distillation/loss_max": Metric(AggregationType.MAX, distillation_losses_response.max()),
120|    }

123|def compute_topk_loss(
124|    config: ActorConfig,
125|    distillation_config: DistillationConfig,
126|    data: TensorDict,
127|    student_logits: torch.Tensor,
128|    data_format: str,
129|) -> torch.Tensor:
130|    """Compute the topk loss in logit processor.
131|
132|    Returns:
133|    - distillation_losses: (bsz, seqlen/cp_size)
134|    - student_mass: (bsz, seqlen/cp_size)
135|    - teacher_mass: (bsz, seqlen/cp_size)
136|    """
137|    match config.strategy:
138|        # VeOmni uses FSDP2 internally, so its loss computation is identical to FSDP.
139|        case "fsdp" | "veomni" | "fsdp2":
140|            import verl.trainer.distillation.fsdp.losses as fsdp_losses
141|
142|            distillation_loss_fn = fsdp_losses.compute_forward_kl_topk
143|        case "megatron":
144|            import verl.trainer.distillation.megatron.losses as megatron_losses
145|
146|            distillation_loss_fn = megatron_losses.compute_forward_kl_topk
147|        case _:
148|            raise NotImplementedError(f"Unsupported strategy: {config.strategy=}")
149|
150|    outputs = distillation_loss_fn(
151|        student_logits=student_logits,
152|        teacher_topk_log_probs=data["teacher_logprobs"],
153|        teacher_topk_ids=data["teacher_ids"],
154|        config=distillation_config,
155|        data_format=data_format,
156|    )
157|
158|    expected_shape = student_logits.shape[:2]
159|    for k, v in outputs.items():
160|        assert v.shape == expected_shape, f"Expected shape {expected_shape}, but got {v.shape} for {k=}."
161|
162|    return outputs

165|def distillation_ppo_loss(
166|    config: ActorConfig,
167|    distillation_config: Optional[DistillationConfig],
168|    model_output: dict = None,
169|    data: TensorDict = None,
170|    dp_group=None,
171|    student_logits: torch.Tensor = None,
172|    data_format: str = "thd",
173|):
174|    """Loss function used both for logit processor and final policy loss.
175|    - student_logits is not None, compute the topk loss in logit processor.
176|    - student_logits is None, compute final policy loss.
177|
178|    [split sequence across sp/cp groups]
179|                   |
180|    [model forward and output logits: (bsz, seqlen/cp_size, vocab_size/tp_size)]
181|                   |
182|    [logits processor compute topk loss: (bsz, seqlen/cp_size)]
183|                   |
184|    [all gather topk loss across sp/cp groups: (bsz, seqlen)]
185|                   |
186|    [combine topk loss with policy loss]
187|
188|    Args:
189|        config: Actor configuration.
190|        distillation_config: Distillation configuration.
191|        model_output: Model output, including log_probs, entropy.
192|        data: Micro input batch, contains
193|          - teacher_logprobs: (bsz, seqlen, topk)
194|          - teacher_ids: (bsz, seqlen, topk)
195|        student_logits: (bsz, seqlen/cp_size, vocab_size/tp_size).
196|        data_format: "thd" or "bshd", models not support THD format, e.g GPT-OSS, Qwen3.5
197|
198|    Returns:
199|    - student_logits is not None, return the topk loss tensor (bsz, seqlen/cp_size).
200|    - student_logits is None, return the final policy loss scalar and metrics.
201|    """
202|
203|    # Called as logits processor
204|    if student_logits is not None:
205|        return compute_topk_loss(config, distillation_config, data, student_logits, data_format)
206|
207|    # Called as final policy loss
208|    distillation_loss_config = distillation_config.distillation_loss
209|    distill_loss, distill_metrics = distillation_loss(config, distillation_config, model_output, data)
210|    if not distillation_loss_config.use_task_rewards and not distillation_loss_config.use_policy_gradient:
211|        # no need to compute policy loss
212|        policy_loss = 0.0
213|        policy_metrics = {}
214|    else:
215|        policy_loss, policy_metrics = ppo_loss(config, model_output, data, dp_group)
216|        if not distillation_loss_config.use_task_rewards:
217|            policy_loss = 0.0
218|
219|    # Combine distillation with policy loss
220|    policy_metrics.update(distill_metrics)
221|    distillation_loss_coef = (
222|        distillation_loss_config.distillation_loss_coef if distillation_loss_config.use_task_rewards else 1.0
223|    )
224|    policy_loss += distill_loss * distillation_loss_coef
225|    policy_metrics["distillation/loss"] = Metric(value=distill_loss, aggregation=AggregationType.SUM)
226|
227|    return policy_loss, policy_metrics

230|def distillation_loss(
231|    config: ActorConfig,
232|    distillation_config: DistillationConfig,
233|    model_output: dict,
234|    data: TensorDict,
235|) -> tuple[torch.Tensor, dict[str, Any]]:
236|    """
237|    Compute the distillation loss and related metrics.
238|
239|    Returns:
240|    - distillation_loss: Aggregated distillation loss scalar.
241|    - distillation_metrics: Dictionary of metrics.
242|    """
243|    assert distillation_config is not None
244|    loss_config: DistillationLossConfig = distillation_config.distillation_loss
245|    distillation_loss_fn = get_distillation_loss_fn(loss_config.loss_mode)
246|    distillation_losses, distillation_metrics = distillation_loss_fn(
247|        config=config,
248|        distillation_config=distillation_config,
249|        model_output=model_output,
250|        data=data,
251|    )
252|    response_mask = data["response_mask"]
253|    loss_agg_mode = config.loss_agg_mode
254|
255|    distillation_metrics.update(
256|        compute_distillation_loss_range(distillation_losses=distillation_losses, response_mask=response_mask)
257|    )
258|    if loss_config.loss_max_clamp is not None:
259|        # clamping min is for k1 loss which can be negative
260|        distillation_losses = distillation_losses.clamp(min=-loss_config.loss_max_clamp, max=loss_config.loss_max_clamp)
261|
262|    # global batch info for loss aggregation
263|    config.global_batch_info["dp_size"] = data["dp_size"]
264|    config.global_batch_info["batch_num_tokens"] = data["batch_num_tokens"]
265|    config.global_batch_info["global_batch_size"] = data["global_batch_size"]
266|    config.global_batch_info["loss_scale_factor"] = config.loss_scale_factor
267|
268|    if loss_config.use_policy_gradient:
269|        # Use negative distillation loss as reward, as done by https://thinkingmachines.ai/blog/on-policy-distillation/.
270|        policy_loss_fn = get_policy_loss_fn(loss_config.policy_loss_mode)
271|        for k, v in config.global_batch_info.items():
272|            loss_config.global_batch_info[k] = v
273|        log_prob = no_padding_2_padding(model_output["log_probs"], data)
274|        old_log_prob = data["old_log_probs"]
275|        if old_log_prob.is_nested:
276|            old_log_prob = data["old_log_probs"].to_padded_tensor(0.0)
277|        if response_mask.is_nested:
278|            response_mask = response_mask.to_padded_tensor(False)
279|        rollout_is_weights = data.get("rollout_is_weights", None)
280|        distillation_loss, pg_metrics = policy_loss_fn(
281|            old_log_prob=old_log_prob,
282|            log_prob=log_prob,
283|            advantages=-distillation_losses.detach(),
284|            response_mask=response_mask,
285|            loss_agg_mode=loss_agg_mode,
286|            config=loss_config,
287|            rollout_is_weights=rollout_is_weights,
288|        )
289|        pg_metrics = {f"distillation/{k[len('actor/') :]}": v for k, v in pg_metrics.items()}
290|        distillation_metrics.update(pg_metrics)
291|    else:
292|        # Directly backpropagate distillation loss as a supervised loss, as in https://arxiv.org/abs/2306.13649.
293|        if response_mask.is_nested:
294|            response_mask = response_mask.to_padded_tensor(False)
295|        distillation_loss = agg_loss(
296|            loss_mat=distillation_losses,
297|            loss_mask=response_mask,
298|            loss_agg_mode=loss_agg_mode,
299|            **config.global_batch_info,
300|        )
301|
302|    return distillation_loss, distillation_metrics

305|@register_distillation_loss(DistillationLossSettings(names=["forward_kl_topk"], use_topk=True))  # type: ignore[arg-type]
306|def compute_forward_kl_topk(
307|    config: ActorConfig,
308|    distillation_config: DistillationConfig,
309|    model_output: dict,
310|    data: TensorDict,
311|) -> tuple[torch.Tensor, dict[str, Any]]:
312|    """Compute forward KL distillation loss and related metrics using top-k log probabilities.
313|
314|    Returns:
315|    - distillation_losses: (bsz, resp_len)
316|    - distillation_metrics: Dictionary of metrics.
317|    """
318|    # topk loss has been computed in logits processor
319|    distillation_losses = no_padding_2_padding(model_output["distillation_losses"], data)
320|    student_mass = no_padding_2_padding(model_output["student_mass"], data)
321|    teacher_mass = no_padding_2_padding(model_output["teacher_mass"], data)
322|    overlap_count = model_output.get("overlap_count")
323|    overlap_token_advantage = model_output.get("overlap_token_advantage")
324|    if overlap_count is not None and overlap_token_advantage is not None:
325|        overlap_count = no_padding_2_padding(overlap_count, data)
326|        overlap_token_advantage = no_padding_2_padding(overlap_token_advantage, data)
327|    if data["response_mask"].is_nested:
328|        response_mask_bool = data["response_mask"].bool().to_padded_tensor(False)
329|    else:
330|        response_mask_bool = data["response_mask"].bool()
331|    assert distillation_losses.shape == student_mass.shape == teacher_mass.shape == response_mask_bool.shape
332|
333|    overlap_metrics = {}
334|    if overlap_count is not None and overlap_token_advantage is not None:
335|        assert overlap_count.shape == overlap_token_advantage.shape == response_mask_bool.shape
336|        valid_overlap_count = overlap_count[response_mask_bool]
337|        k = distillation_config.distillation_loss.topk
338|        assert k is not None
339|        # Diagnostics for tracking teacher/student top-k overlap in OPD, following
340|        # "Rethinking On-Policy Distillation of Large Language Models" (arXiv:2604.13016):
341|        # overlap ratio and average teacher-token KL contribution on overlapped tokens.
342|        overlap_metrics["distillation/overlap_ratio"] = (valid_overlap_count.float().mean() / k).item()
343|        overlap_position_mask = response_mask_bool & (overlap_count > 0)
344|        if overlap_position_mask.any():
345|            overlap_metrics["distillation/overlap_token_advantage"] = (
346|                overlap_token_advantage[overlap_position_mask].mean().item()
347|            )
348|        else:
349|            overlap_metrics["distillation/overlap_token_advantage"] = 0.0
350|
351|    # Log amount of mass in the top-k log probabilities for both student and teacher.
352|    student_mass = student_mass[response_mask_bool]
353|    teacher_mass = teacher_mass[response_mask_bool]
354|    distillation_metrics = {
355|        "distillation/student_mass": student_mass.mean().item(),
356|        "distillation/student_mass_min": Metric(AggregationType.MIN, student_mass.min()),
357|        "distillation/student_mass_max": Metric(AggregationType.MAX, student_mass.max()),
358|        "distillation/teacher_mass": teacher_mass.mean().item(),
359|        "distillation/teacher_mass_min": Metric(AggregationType.MIN, teacher_mass.min()),
360|        "distillation/teacher_mass_max": Metric(AggregationType.MAX, teacher_mass.max()),
361|        **overlap_metrics,
362|    }
363|
364|    # Due to use of top-k, student and teacher distributions don't sum to 1 -> divergences can be negative.
365|    distillation_losses = distillation_losses.clamp_min(0.0)
366|
367|    return distillation_losses, distillation_metrics

370|@register_distillation_loss(
371|    DistillationLossSettings(names=["kl", "k1", "abs", "mse", "k2", "low_var_kl", "k3"], use_estimator=True)
372|)  # type: ignore[arg-type]
373|def compute_distillation_loss_reverse_kl_estimator(
374|    config: ActorConfig,
375|    distillation_config: DistillationConfig,
376|    model_output,
377|    data: TensorDict,
378|) -> tuple[torch.Tensor, dict[str, Any]]:
379|    """
380|    Compute the distillation loss and related metrics using single-sample KL estimators.
381|
382|    Uses the kl_penalty function from core_algos which supports various KL divergence
383|    estimators: "kl", "k1", "abs", "mse", "k2", "low_var_kl", "k3".
384|
385|    Returns:
386|    - distillation_losses: (bsz, resp_len)
387|    - distillation_metrics: Dictionary of metrics.
388|    """
389|    student_log_probs = no_padding_2_padding(model_output["log_probs"], data)
390|    teacher_log_probs = no_padding_2_padding(data["teacher_logprobs"], data).squeeze(-1)
391|    if data["response_mask"].is_nested:
392|        response_mask_bool = data["response_mask"].bool().to_padded_tensor(False)
393|    else:
394|        response_mask_bool = data["response_mask"].bool()
395|    assert teacher_log_probs.shape == student_log_probs.shape == response_mask_bool.shape
396|
397|    loss_config: DistillationLossConfig = distillation_config.distillation_loss
398|    distillation_losses = kl_penalty(
399|        logprob=student_log_probs, ref_logprob=teacher_log_probs, kl_penalty=loss_config.loss_mode
400|    )
401|    # Since k1 can be negative, log the mean absolute loss.
402|    metrics = {
403|        "distillation/abs_loss": Metric(AggregationType.MEAN, distillation_losses[response_mask_bool].abs().mean()),
404|    }
405|    return distillation_losses, metrics
