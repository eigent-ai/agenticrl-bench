35|def _get_teacher_sampling_params(
36|    teacher_model_config: DistillationTeacherModelConfig,
37|    distillation_loss_config: DistillationLossConfig,
38|) -> dict[str, Any]:
39|    """Get sampling parameters for teacher model when computing log probabilities for distillation."""
40|    # Temperature has no effect on prompt_logprobs: the teacher performs a forward pass over
41|    # existing tokens (no sampling). Always use temperature=1.0 regardless of the config value.
42|    # The default distillation.yaml copies the student rollout temperature via Hydra interpolation
43|    # (temperature: ${oc.select:actor_rollout_ref.rollout.temperature}), which causes a spurious
44|    # crash when rollout.temperature != 1.0.
45|    if teacher_model_config.inference.temperature != 1.0:
46|        logger.warning(
47|            "Teacher inference temperature is set to %.1f, but temperature has no effect "
48|            "on prompt_logprobs (forward pass only). Using temperature=1.0.",
49|            teacher_model_config.inference.temperature,
50|        )
51|    num_logprobs = distillation_loss_config.topk if distillation_loss_config.loss_settings.use_topk else 0
52|    return {
53|        "max_tokens": 1,
54|        "temperature": 1.0,
55|        "prompt_logprobs": num_logprobs,
56|        "detokenize": False,
57|    }

60|def _pad_teacher_outputs(
61|    teacher_ids: torch.Tensor,
62|    teacher_logprobs: torch.Tensor,
63|    prompt_width: int,
64|    response_width: int,
65|    prompt_length: int,
66|    response_length: int,
67|    pad_token_id: int,
68|) -> tuple[torch.Tensor, torch.Tensor]:
69|    # TODO(wuxibin): remove padding and use tensordict.
70|    left_pad_size = prompt_width - prompt_length
71|    right_pad_size = response_width - response_length
72|    padding = (0, 0, left_pad_size, right_pad_size)
73|    return (
74|        F.pad(teacher_ids, padding, value=pad_token_id).unsqueeze(0),
75|        F.pad(teacher_logprobs, padding, value=0.0).unsqueeze(0),
76|    )

79|class AsyncTeacherLLMServerManager:
80|    """Teacher-specific async client used for distillation logprob computation."""
81|
82|    def __init__(
83|        self,
84|        config: DictConfig,
85|        teacher_client: dict[str, LLMServerClient],
86|    ):
87|        self.distillation_config: DistillationConfig = omega_conf_to_dataclass(config.distillation)
88|        self.distillation_loss_config: DistillationLossConfig = self.distillation_config.distillation_loss
89|        self.teacher_key: str = self.distillation_config.teacher_key
90|
91|        self.teacher_model_configs: dict[str, DistillationTeacherModelConfig] = self.distillation_config.teacher_models
92|        expected = set(self.teacher_model_configs)
93|        if set(teacher_client.keys()) != expected:
94|            raise ValueError(
95|                f"teacher client keys {sorted(teacher_client.keys())} "
96|                f"do not match teacher routing keys {sorted(expected)}."
97|            )
98|        self.teacher_client: dict[str, LLMServerClient] = teacher_client
99|
100|    def _resolve_teacher_key(self, routing_key: Optional[str]) -> str:
101|        if len(self.teacher_model_configs) == 1:
102|            # Single-teacher path: route everything to the one teacher regardless of the sample's key.
103|            return next(iter(self.teacher_model_configs))
104|        if routing_key is None:
105|            raise ValueError(
106|                f"Routing key is required for multi-teacher distillation "
107|                f"(configured via distillation.teacher_key={self.teacher_key!r})."
108|            )
109|        if routing_key not in self.teacher_model_configs:
110|            raise ValueError(
111|                f"No teacher configured for routing key {routing_key!r}. "
112|                f"Configured teachers: {sorted(self.teacher_model_configs)}."
113|            )
114|        return routing_key
115|
116|    async def compute_teacher_logprobs_single(
117|        self,
118|        sequence_ids: list[int],
119|        multi_modal_data: Optional[dict[str, Any]] = None,
120|        mm_processor_kwargs: Optional[dict[str, Any]] = None,
121|        mm_processor_output: Optional[list[dict[str, Any]]] = None,
122|        routing_key: Optional[str] = None,
123|    ) -> tuple[torch.Tensor, torch.Tensor]:
124|        """Compute teacher log probabilities for a single unpadded sequence."""
125|        multi_modal_data = multi_modal_data or {}
126|        teacher_key = self._resolve_teacher_key(routing_key)
127|        teacher_model_config = self.teacher_model_configs[teacher_key]
128|        client = self.teacher_client[teacher_key]
129|        teacher_output = await client.generate(
130|            request_id=uuid4().hex,
131|            prompt_ids=sequence_ids,
132|            sampling_params=_get_teacher_sampling_params(teacher_model_config, self.distillation_loss_config),
133|            image_data=multi_modal_data.get("images"),
134|            video_data=multi_modal_data.get("videos"),
135|            audio_data=multi_modal_data.get("audios"),
136|            mm_processor_output=mm_processor_output,
137|            mm_processor_kwargs=mm_processor_kwargs,
138|        )
139|        # Shapes: # S, (1 or K), where S is the response length, K is either 1 or topk depending on
140|        # the distillation loss settings.
141|        teacher_ids = torch.tensor(teacher_output.extra_fields["prompt_ids"], dtype=torch.int32)
142|        teacher_logprobs = torch.tensor(teacher_output.extra_fields["prompt_logprobs"])
143|        assert teacher_ids.shape[0] == teacher_logprobs.shape[0] == len(sequence_ids)
144|        return teacher_ids, teacher_logprobs
