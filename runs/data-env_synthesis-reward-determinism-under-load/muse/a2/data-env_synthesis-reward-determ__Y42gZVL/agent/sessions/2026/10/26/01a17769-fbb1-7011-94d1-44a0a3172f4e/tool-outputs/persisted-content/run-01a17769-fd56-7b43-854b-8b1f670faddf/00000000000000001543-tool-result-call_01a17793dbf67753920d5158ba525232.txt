+def _wait_for(predicate, timeout=10.0, interval=0.2):
+    """Poll *predicate* until truthy or deadline; return its last value.
+
+    Clipboard ownership becomes visible asynchronously (xclip forks a
+    holder daemon) and Xvfb startup is not instantaneous; under host
+    contention the old fixed sleep(0.3)/sleep(1.5) could expire before
+    either was ready — and a stale clipboard from a previous test
+    could be read instead of this test's content.
+    """
+    deadline = time.monotonic() + timeout
+    while True:
+        value = predicate()
+        if value:
+            return value
+        if time.monotonic() >= deadline:
+            return value
+        time.sleep(interval)
+
