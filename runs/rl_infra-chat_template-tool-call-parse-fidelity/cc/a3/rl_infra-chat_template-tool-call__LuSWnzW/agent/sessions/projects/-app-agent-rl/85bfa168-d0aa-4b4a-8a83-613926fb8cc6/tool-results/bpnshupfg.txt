    50	def get_argument_type(
    51	    func_name: str, arg_key: str, defined_tools: List[Tool]
    52	) -> Optional[str]:
    53	    """Get the expected type of a function argument from tool definitions.
    54	
    55	    Supports complex JSON Schema definitions including:
    56	    - Direct type field (including type arrays)
    57	    - anyOf/oneOf: parameter can be any of multiple types
    58	    - enum: parameter must be one of enum values
    59	    - allOf: parameter must satisfy all type definitions
    60	    - properties: inferred as object type
    61	    - items: inferred as array type
    62	
    63	    Args:
    64	        func_name: Name of the function/tool
    65	        arg_key: Name of the argument
    66	        defined_tools: List of available tools
    67	
    68	    Returns:
    69	        The type string (e.g., 'string', 'number', 'object') or None if not found
    70	    """
    71	    name2tool = {tool.function.name: tool for tool in defined_tools}
    72	
    73	    # Check if function exists
    74	    tool = name2tool.get(func_name)
    75	    if not tool:
    76	        return None
    77	
    78	    # Get parameters safely using getattr
    79	    params = getattr(tool.function, "parameters", None)
    80	    if not isinstance(params, dict):
    81	        return None
    82	
    83	    # Navigate to the type using dict.get() for safe access
    84	    properties = params.get("properties")
    85	    if not isinstance(properties, dict):
    86	        return None
    87	
    88	    arg_spec = properties.get(arg_key)
    89	    if isinstance(arg_spec, dict):
    90	        # Use the new type inference function for complex JSON Schema support
    91	        return infer_type_from_json_schema(arg_spec)
    92	
    93	    return None

    96	def _convert_to_number(value: str) -> Any:
    97	    """Convert string to appropriate number type (int or float).
    98	
    99	    Args:
   100	        value: String value to convert
   101	
   102	    Returns:
   103	        Converted number or original string if conversion fails
   104	    """
   105	    try:
   106	        if "." in value or "e" in value.lower():
   107	            return float(value)
   108	        else:
   109	            return int(value)
   110	    except (ValueError, AttributeError):
   111	        return value

   114	def parse_arguments(
   115	    json_value: str, arg_type: Optional[str] = None
   116	) -> Tuple[Any, bool]:
   117	    """Parse argument value with multiple fallback strategies.
   118	
   119	    Args:
   120	        json_value: Raw string value to parse
   121	        arg_type: Expected type hint ('string', 'number', 'object', etc.)
   122	
   123	    Returns:
   124	        Tuple of (parsed_value, is_valid_json)
   125	    """
   126	    # Strategy 1: Direct JSON parsing
   127	    try:
   128	        parsed_value = json.loads(json_value)
   129	
   130	        # Type coercion for number type
   131	        if arg_type == "number" and isinstance(parsed_value, str):
   132	            parsed_value = _convert_to_number(parsed_value)
   133	
   134	        return parsed_value, True
   135	    except (json.JSONDecodeError, ValueError):
   136	        pass
   137	
   138	    # Strategy 2: Unescape and parse
   139	    try:
   140	        wrapped = json.loads('{"tmp": "' + json_value + '"}')
   141	        parsed_value = json.loads(wrapped["tmp"])
   142	
   143	        if arg_type == "number" and isinstance(parsed_value, str):
   144	            parsed_value = _convert_to_number(parsed_value)
   145	
   146	        return parsed_value, True
   147	    except (json.JSONDecodeError, ValueError, KeyError):
   148	        pass
   149	
   150	    # Strategy 3: ast.literal_eval
   151	    try:
   152	        parsed_value = ast.literal_eval(json_value)
   153	        return parsed_value, True
   154	    except (ValueError, SyntaxError):
   155	        pass
   156	
   157	    # Strategy 4: Treat as string
   158	    try:
   159	        quoted_value = json.dumps(str(json_value))
   160	        return json.loads(quoted_value), True
   161	    except (json.JSONDecodeError, ValueError):
   162	        return json_value, False

   165	class Glm47MoeDetector(BaseFormatDetector):
   166	    """
   167	    Detector for GLM-4.7 and GLM-5 models.
   168	    Assumes function call format:
   169	      <tool_call>get_weather<arg_key>city</arg_key><arg_value>北京</arg_value><arg_key>date</arg_key><arg_value>2024-06-27</arg_value></tool_call><tool_call>get_weather<arg_key>city</arg_key><arg_value>上海</arg_value><arg_key>date</arg_key><arg_value>2024-06-27</arg_value></tool_call>
   170	    """
   171	
   172	    def __init__(self):
   173	        super().__init__()
   174	        self.bot_token = "<tool_call>"
   175	        self.eot_token = "</tool_call>"
   176	        self.func_call_regex = r"<tool_call>.*?</tool_call>"
   177	        self.func_detail_regex = re.compile(
   178	            r"<tool_call>(.*?)(<arg_key>.*?)?</tool_call>", re.DOTALL
   179	        )
   180	        self.func_arg_regex = re.compile(
   181	            r"<arg_key>(.*?)</arg_key>(?:\\n|\s)*<arg_value>(.*?)</arg_value>",
   182	            re.DOTALL,
   183	        )
   184	        self._last_arguments = ""
   185	        self.current_tool_id = -1
   186	        self.current_tool_name_sent = False
   187	        self._streamed_raw_length = 0
   188	        self._tool_call_completed = False  # Track if tool call has been completed
   189	        self._sent_empty_object = (
   190	            False  # Track if empty object has been sent for no-arg functions
   191	        )
   192	        self._reset_streaming_state()

   194	    def _reset_streaming_state(self) -> None:

   196	        self._stream_state = StreamState.INIT

   198	        self._current_value = ""
   199	        self._xml_tag_buffer = ""
   200	        self._is_first_param = True
   201	        self._value_started = False
   202	        self._cached_value_type: Optional[str] = (
   203	            None  # Cache the value type for consistency
   204	        )
   205	        self._tool_call_completed = False  # Reset tool call completion status
   206	        self._sent_empty_object = False  # Reset empty object sent status

   212	    def detect_and_parse(self, text: str, tools: List[Tool]) -> StreamingParseResult:
   213	        """
   214	        One-time parsing: Detects and parses tool calls in the provided text.
   215	
   216	        :param text: The complete text to parse.
   217	        :param tools: List of available tools.
   218	        :return: ParseResult indicating success or failure, consumed text, leftover text, and parsed calls.
   219	        """
   220	        if self.bot_token not in text:
   221	            return StreamingParseResult(normal_text=text, calls=[])
   222	
   223	        # Extract all normal text (before, between, and after tool calls)
   224	        normal_text_parts = []
   225	        last_end = 0
   226	
   227	        # Find all tool call matches
   228	        for match in re.finditer(self.func_call_regex, text, re.DOTALL):
   229	            # Add text before this tool call
   230	            if match.start() > last_end:
   231	                normal_text_parts.append(text[last_end : match.start()])
   232	            last_end = match.end()
   233	
   234	        # Add any remaining text after the last tool call
   235	        if last_end < len(text):
   236	            normal_text_parts.append(text[last_end:])
   237	
   238	        # Combine all normal text parts
   239	        normal_text = "".join(normal_text_parts).strip()
   240	
   241	        # Parse tool calls
   242	        match_result_list = re.findall(self.func_call_regex, text, re.DOTALL)
   243	        calls = []
   244	        try:
   245	            for match_result in match_result_list:
   246	                # Get function name
   247	                func_detail = self.func_detail_regex.search(match_result)
   248	                if func_detail is None:
   249	                    continue
   250	                func_name = func_detail.group(1) if func_detail.group(1) else ""
   251	                func_args = func_detail.group(2) if func_detail.group(2) else ""
   252	                arguments = {}
   253	                if func_args:
   254	                    pairs = self.func_arg_regex.findall(func_args)
   255	                    # Parse arguments using shared method
   256	                    arguments = self._parse_argument_pairs(pairs, func_name, tools)
   257	
   258	                # construct match_result for parse_base_json
   259	                match_result = {"name": func_name, "parameters": arguments}
   260	                calls.extend(self.parse_base_json(match_result, tools))
   261	            return StreamingParseResult(normal_text=normal_text, calls=calls)
   262	        except Exception as e:
   263	            logger.error(f"Error in detect_and_parse: {e}", exc_info=True)
   264	            # return the normal text if parsing fails
   265	            return StreamingParseResult(normal_text=text)

   267	    def _get_value_type(self, func_name: str, key: str, tools: List[Tool]) -> str:
   268	        """Get parameter type from tool definition, with fallback to auto-detection.
   269	
   270	        Args:
   271	            func_name: Name of the function
   272	            key: Parameter name
   273	            tools: List of available tools
   274	
   275	        Returns:
   276	            Type string: 'string', 'number', 'object', 'array', or 'boolean'
   277	        """
   278	        arg_type = get_argument_type(func_name, key, tools)
   279	        if arg_type:
   280	            return arg_type
   281	
   282	        # Improved auto-detection type from value (best effort)
   283	        value_content = self._current_value.strip() if self._current_value else ""
   284	
   285	        if not value_content:
   286	            return "string"
   287	
   288	        # Try to parse as valid JSON first
   289	        try:
   290	            parsed = json.loads(value_content)
   291	            if isinstance(parsed, dict):
   292	                return "object"
   293	            elif isinstance(parsed, list):
   294	                return "array"
   295	            elif isinstance(parsed, bool):
   296	                return "boolean"
   297	            elif isinstance(parsed, (int, float)):
   298	                return "number"
   299	            # For string values, check if they look like numbers
   300	            elif isinstance(parsed, str):
   301	                if parsed.isdigit() or (
   302	                    parsed.startswith("-") and parsed[1:].isdigit()
   303	                ):
   304	                    return "number"
   305	                return "string"
   306	        except json.JSONDecodeError:
   307	            # Not valid JSON, try heuristic detection
   308	            first_char = value_content[0] if value_content else ""
   309	
   310	            if first_char.isdigit() or first_char in ["-", "."]:
   311	                return "number"
   312	            elif first_char in ["{", "["]:
   313	                return "object"
   314	            elif first_char in ['"', "'"]:
   315	                return "string"
   316	
   317	        # Default to string (safest fallback)
   318	        return "string"

   320	    def _format_value_complete(self, value: str, value_type: str) -> str:
   321	        """Format complete value based on type.
   322	
   323	        Args:
   324	            value: Raw value string
   325	            value_type: Expected type ('string', 'number', 'object')
   326	
   327	        Returns:
   328	            Properly formatted JSON value string
   329	        """
   330	        if value_type == "string":
   331	            # Ensure proper JSON string formatting with quotes
   332	            return json.dumps(value, ensure_ascii=False)
   333	        elif value_type == "number":
   334	            try:
   335	                num = _convert_to_number(value.strip() if value else "")
   336	                return str(num)
   337	            except (ValueError, AttributeError):
   338	                # Fallback to string if not a valid number
   339	                logger.warning(
   340	                    f"Failed to parse '{value}' as number, treating as string"
   341	                )
   342	                return json.dumps(str(value) if value else "", ensure_ascii=False)
   343	        else:
   344	            # For object/array types, return as-is (should already be valid JSON)
   345	            return value

   347	    def _process_xml_to_json_streaming(
   348	        self, raw_increment: str, func_name: str, tools: List[Tool]
   349	    ) -> str:
   350	        """Convert XML increment to JSON streaming output using state machine.
   351	
   352	        This method processes XML fragments character by character and converts them
   353	        to JSON format incrementally. It maintains state across calls to handle
   354	        partial XML tags and values.
   355	
   356	        Args:
   357	            raw_increment: New XML content to process
   358	            func_name: Name of the function being called
   359	            tools: List of available tools for type inference
   360	
   361	        Returns:
   362	            JSON string increment to append to the output
   363	        """
   364	        json_output = ""
   365	
   366	        for char in raw_increment:
   367	            self._xml_tag_buffer += char
   368	
   369	            if self._stream_state in [StreamState.INIT, StreamState.BETWEEN]:
   370	                if self._xml_tag_buffer.endswith("<arg_key>"):
   371	                    self._stream_state = StreamState.IN_KEY
   372	                    self._current_key = ""
   373	                    self._xml_tag_buffer = ""
   374	                    json_output += "{" if self._is_first_param else ", "
   375	                    self._is_first_param = False
   376	
   377	            elif self._stream_state == StreamState.IN_KEY:
   378	                if self._xml_tag_buffer.endswith("</arg_key>"):
   379	                    self._current_key = self._xml_tag_buffer[:-10].strip()
   380	                    self._xml_tag_buffer = ""
   381	                    self._stream_state = StreamState.WAITING_VALUE
   382	                    json_output += (
   383	                        json.dumps(self._current_key, ensure_ascii=False) + ": "
   384	                    )
   385	
   386	            elif self._stream_state == StreamState.WAITING_VALUE:
   387	                if self._xml_tag_buffer.endswith("<arg_value>"):
   388	                    self._stream_state = StreamState.IN_VALUE
   389	                    self._current_value = ""
   390	                    self._xml_tag_buffer = ""
   391	                    self._value_started = False
   392	                    # Determine and cache the value type at the start
   393	                    self._cached_value_type = self._get_value_type(
   394	                        func_name, self._current_key, tools
   395	                    )
   396	
   397	            elif self._stream_state == StreamState.IN_VALUE:
   398	                if self._xml_tag_buffer.endswith("</arg_value>"):
   399	                    final_value = self._xml_tag_buffer[:-12]
   400	                    self._current_value += final_value
   401	
   402	                    # Use cached value type for consistency
   403	                    value_type = self._cached_value_type or "string"
   404	
   405	                    if self._value_started:
   406	                        # Output any remaining content
   407	                        if final_value:
   408	                            if value_type == "string":
   409	                                json_output += json.dumps(
   410	                                    final_value, ensure_ascii=False
   411	                                )[1:-1]
   412	                            else:
   413	                                json_output += final_value
   414	                        # Always output closing quote for string type when value was started
   415	                        if value_type == "string":
   416	                            json_output += '"'
   417	                    else:
   418	                        # Value was never started (empty or complete in one chunk)
   419	                        json_output += self._format_value_complete(
   420	                            self._current_value, value_type
   421	                        )
   422	
   423	                    self._xml_tag_buffer = ""
   424	                    self._stream_state = StreamState.BETWEEN
   425	                    self._current_value = ""
   426	                    self._value_started = False
   427	                    self._cached_value_type = None  # Reset cached type
   428	                else:
   429	                    closing_tag = "</arg_value>"
   430	                    is_potential_closing = len(self._xml_tag_buffer) <= len(
   431	                        closing_tag
   432	                    ) and closing_tag.startswith(self._xml_tag_buffer)
   433	
   434	                    if not is_potential_closing:
   435	                        content = self._xml_tag_buffer
   436	                        # Use cached value type for consistency
   437	                        value_type = self._cached_value_type or "string"
   438	
   439	                        if value_type == "string":
   440	                            if not self._value_started:
   441	                                json_output += '"'
   442	                                self._value_started = True
   443	                            if content:
   444	                                json_output += json.dumps(content, ensure_ascii=False)[
   445	                                    1:-1
   446	                                ]
   447	                                self._current_value += content
   448	                                self._xml_tag_buffer = ""
   449	                        elif value_type == "number":
   450	                            if content:
   451	                                if not self._value_started:
   452	                                    self._value_started = True
   453	                                json_output += content
   454	                                self._current_value += content
   455	                                self._xml_tag_buffer = ""
   456	                        else:
   457	                            # For object/array types, output as-is
   458	                            if content:
   459	                                if not self._value_started:
   460	                                    self._value_started = True
   461	                                json_output += content
   462	                                self._current_value += content
   463	                                self._xml_tag_buffer = ""
   464	
   465	        return json_output

   467	    def _extract_match_groups(self, match: re.Match) -> tuple[str, str, str]:
   468	        """Extract function name, arguments and end marker from regex match.
   469	
   470	        Args:
   471	            match: Regex match object
   472	
   473	        Returns:
   474	            (func_name, func_args_raw, is_tool_end)
   475	        """
   476	        func_name = match.group(1).strip()
   477	        func_args_raw = match.group(2).strip() if match.group(2) else ""
   478	        is_tool_end = match.group(3) or ""
   479	        return func_name, func_args_raw, is_tool_end

   481	    def _send_tool_name_if_needed(
   482	        self, func_name: str, has_arg_key: bool, is_tool_end: str
   483	    ) -> Optional[ToolCallItem]:
   484	        """Send tool name if needed.
   485	
   486	        Args:
   487	            func_name: Function name
   488	            has_arg_key: Whether current text contains <arg_key
   489	            is_tool_end: Whether end marker is encountered
   490	
   491	        Returns:
   492	            Tool call item or None
   493	        """
   494	        if self.current_tool_name_sent:
   495	            return None
   496	
   497	        # Function name completeness check
   498	        is_func_name_complete = has_arg_key or is_tool_end == self.eot_token
   499	
   500	        if not is_func_name_complete:
   501	            return None
   502	
   503	        if not func_name:
   504	            logger.warning("Empty function name detected, skipping tool call")
   505	            return None
   506	
   507	        # Send tool name
   508	        self.current_tool_name_sent = True
   509	        self._streamed_raw_length = 0
   510	        self._reset_streaming_state()
   511	
   512	        # Record tool info
   513	        self.prev_tool_call_arr[self.current_tool_id] = {
   514	            "name": func_name,
   515	            "arguments": {},
   516	        }
   517	
   518	        return ToolCallItem(
   519	            tool_index=self.current_tool_id,
   520	            name=func_name,
   521	            parameters="",
   522	        )

   524	    def _process_arguments_streaming(
   525	        self, func_name: str, func_args_raw: str, tools: List[Tool]
   526	    ) -> Optional[ToolCallItem]:
   527	        """Process streaming arguments.
   528	
   529	        Args:
   530	            func_name: Function name
   531	            func_args_raw: Raw argument string
   532	            tools: List of available tools
   533	
   534	        Returns:
   535	            Tool call item with parameter updates or None
   536	        """
   537	        current_raw_length = len(func_args_raw)
   538	
   539	        if current_raw_length <= self._streamed_raw_length:
   540	            return None
   541	
   542	        # Get new raw XML content
   543	        raw_increment = func_args_raw[self._streamed_raw_length :]
   544	
   545	        # Convert XML to JSON using state machine
   546	        json_increment = self._process_xml_to_json_streaming(
   547	            raw_increment, func_name, tools
   548	        )
   549	
   550	        # CRITICAL: Update streamed length BEFORE early return
   551	        # Even if json_increment is empty, the input has been consumed by the state machine
   552	        self._streamed_raw_length = current_raw_length
   553	
   554	        if not json_increment:
   555	            return None
   556	
   557	        # Update state
   558	        self._last_arguments += json_increment
   559	        self.streamed_args_for_tool[self.current_tool_id] += json_increment
   560	
   561	        return ToolCallItem(
   562	            tool_index=self.current_tool_id,
   563	            name=None,
   564	            parameters=json_increment,
   565	        )

   567	    def _finalize_tool_call(
   568	        self,
   569	        func_name: str,
   570	        func_args_raw: str,
   571	        tools: List[Tool],
   572	        match_end_pos: int,
   573	        current_text: str,
   574	    ) -> List[ToolCallItem]:
   575	        """Complete tool call processing.
   576	
   577	        Args:
   578	            func_name: Function name
   579	            func_args_raw: Raw argument string
   580	            tools: List of available tools
   581	            match_end_pos: Match end position
   582	            current_text: Current text
   583	
   584	        Returns:
   585	            List of tool call items to add
   586	        """
   587	        calls = []
   588	
   589	        # Handle no-arg function or need to close braces
   590	        if self._is_first_param and not self._sent_empty_object:
   591	            # No-arg function
   592	            calls.append(
   593	                ToolCallItem(
   594	                    tool_index=self.current_tool_id,
   595	                    name=None,
   596	                    parameters="{}",
   597	                )
   598	            )
   599	            self._last_arguments += "{}"
   600	            self.streamed_args_for_tool[self.current_tool_id] += "{}"
   601	            self._sent_empty_object = True
   602	        elif not self._last_arguments.endswith("}") and not self._sent_empty_object:
   603	            # Need to close brace
   604	            calls.append(
   605	                ToolCallItem(
   606	                    tool_index=self.current_tool_id,
   607	                    name=None,
   608	                    parameters="}",
   609	                )
   610	            )
   611	            self._last_arguments += "}"
   612	            self.streamed_args_for_tool[self.current_tool_id] += "}"
   613	            self._sent_empty_object = True
   614	
   615	        # Parse final arguments
   616	        if func_args_raw:
   617	            try:
   618	                pairs = self.func_arg_regex.findall(func_args_raw)
   619	                if pairs:
   620	                    arguments = self._parse_argument_pairs(pairs, func_name, tools)
   621	                    self.prev_tool_call_arr[self.current_tool_id][
   622	                        "arguments"
   623	                    ] = arguments
   624	            except Exception as e:
   625	                logger.debug(f"Failed to parse arguments: {e}", exc_info=True)
   626	
   627	        # Clean buffer
   628	        self._buffer = current_text[match_end_pos:]
   629	
   630	        # Reset state for next tool call
   631	        self._tool_call_completed = True
   632	        self.current_tool_id += 1
   633	        self._last_arguments = ""
   634	        self.current_tool_name_sent = False
   635	        self._streamed_raw_length = 0
   636	        self._reset_streaming_state()
   637	
   638	        return calls
   639	
   640	    def parse_streaming_increment(
   641	        self, new_text: str, tools: List[Tool]
   642	    ) -> StreamingParseResult:
   643	        """
   644	        Streaming incremental parsing tool calls for GLM-4.5 and GLM-4.6 format.
   645	        Uses a state machine to convert XML to JSON incrementally for true character-by-character streaming.
   646	        Outputs JSON increments immediately as XML data arrives.
   647	        """
   648	        self._buffer += new_text
   649	        current_text = self._buffer
   650	
   651	        # Check if we have a tool call
   652	        has_tool_call = self.bot_token in current_text
   653	
   654	        if not has_tool_call:
   655	            # Check if buffer could be the start of a tool call
   656	            # Keep buffer if it could be a partial match of bot_token
   657	            is_potential_start = any(
   658	                self.bot_token.startswith(current_text[-i:])
   659	                for i in range(1, min(len(current_text), len(self.bot_token)) + 1)
   660	            )
   661	
   662	            if not is_potential_start:
   663	                # Not a potential tool call, return as normal text
   664	                # Must return the entire buffer (current_text), not just new_text,
   665	                # because buffer may contain previously accumulated characters like '<'
   666	                # that turned out not to be part of a tool call
   667	                output_text = current_text
   668	                self._buffer = ""
   669	                if self.eot_token in output_text:
   670	                    output_text = output_text.replace(self.eot_token, "")
   671	                return StreamingParseResult(normal_text=output_text)
   672	            else:
   673	                # Could be start of tool call, keep buffering
   674	                return StreamingParseResult(normal_text="", calls=[])
   675	
   676	        # Extract any text before the first bot_token and return it as normal_text
   677	        normal_text = ""
   678	        first_bot_token_idx = current_text.find(self.bot_token)
   679	        if first_bot_token_idx > 0:
   680	            normal_text = current_text[:first_bot_token_idx]
   681	            current_text = current_text[first_bot_token_idx:]
   682	            # Update buffer to only include from the bot token onwards
   683	            self._buffer = current_text
   684	
   685	        if not hasattr(self, "_tool_indices"):
   686	            self._tool_indices = self._get_tool_indices(tools)
   687	
   688	        calls: list[ToolCallItem] = []
   689	        try:
   690	            # Try to match a partial or complete tool call
   691	            # Use a single flexible regex pattern that handles all cases
   692	            partial_match = re.search(
   693	                r"<tool_call>(.*?)(?:(<arg_key.*?))?(?:(</tool_call>)|$)",
   694	                current_text,
   695	                re.DOTALL,
   696	            )
   697	
   698	            if not partial_match:
   699	                return StreamingParseResult(normal_text=normal_text, calls=[])
   700	
   701	            # Extract match groups using helper method
   702	            func_name, func_args_raw, is_tool_end = self._extract_match_groups(
   703	                partial_match
   704	            )
   705	
   706	            # Initialize tool call state if needed (keeping existing logic)
   707	            if self.current_tool_id == -1:
   708	                self.current_tool_id = 0
   709	                self.prev_tool_call_arr = []
   710	                self.streamed_args_for_tool = [""]
   711	                self._streamed_raw_length = 0
   712	                self.current_tool_name_sent = False  # Reset for new tool call
   713	                self._reset_streaming_state()
   714	            # Check if this is a continuation of an existing tool call or a new one
   715	            elif not self.current_tool_name_sent:
   716	                # Only increment tool_id if we're truly starting a NEW tool call
   717	                # Don't increment if this is just the first time we're processing
   718	                # a tool call that was received in the buffer
   719	                # The key insight: only increment when we've COMPLETED a previous tool call
   720	                # and now see another bot_token in new_text
   721	                pass  # Remove the problematic auto-increment logic
   722	
   723	            # Ensure tracking arrays are large enough (keeping existing logic)
   724	            while len(self.prev_tool_call_arr) <= self.current_tool_id:
   725	                self.prev_tool_call_arr.append({})
   726	            while len(self.streamed_args_for_tool) <= self.current_tool_id:
   727	                self.streamed_args_for_tool.append("")
   728	
   729	            # Determine if function name is complete by checking for <arg_key> in the full text
   730	            # This is important for streaming scenarios where args come in later chunks
   731	            has_arg_key = "<arg_key" in current_text
   732	
   733	            # Send tool name if needed
   734	            tool_name_item = self._send_tool_name_if_needed(
   735	                func_name, has_arg_key, is_tool_end
   736	            )
   737	            if tool_name_item:
   738	                calls.append(tool_name_item)
   739	
   740	            # Process streaming arguments if tool name has been sent
   741	            if self.current_tool_name_sent:
   742	                arg_item = self._process_arguments_streaming(
   743	                    func_name, func_args_raw, tools
   744	                )
   745	                if arg_item:
   746	                    calls.append(arg_item)
   747	
   748	                # Finalize tool call if end token is encountered
   749	                if is_tool_end == self.eot_token and not self._tool_call_completed:
   750	                    finalize_calls = self._finalize_tool_call(
   751	                        func_name,
   752	                        func_args_raw,
   753	                        tools,
   754	                        partial_match.end(),
   755	                        current_text,
   756	                    )
   757	                    calls.extend(finalize_calls)
   758	                    return StreamingParseResult(normal_text=normal_text, calls=calls)
   759	
   760	        except Exception as e:
   761	            logger.error(f"Error in parse_streaming_increment: {e}", exc_info=True)
   762	            return StreamingParseResult(normal_text=current_text)
   763	
   764	        return StreamingParseResult(normal_text=normal_text, calls=calls)

   766	    def _parse_argument_pairs(
   767	        self, pairs: List[Tuple[str, str]], func_name: str, tools: List[Tool]
   768	    ) -> Dict[str, Any]:
   769	        """Parse argument key-value pairs with type coercion.
   770	
   771	        Args:
   772	            pairs: List of (key, value) tuples from regex matching
   773	            func_name: Name of the function
   774	            tools: List of available tools
   775	
   776	        Returns:
   777	            Dictionary of parsed arguments
   778	        """
   779	        arguments = {}
   780	        for arg_key, arg_value in pairs:
   781	            arg_key = arg_key.strip()
   782	            arg_type = get_argument_type(func_name, arg_key, tools)
   783	            parsed_value, is_good_json = parse_arguments(arg_value, arg_type)
   784	
   785	            if arg_type == "string":
   786	                # Only convert to string if explicitly defined as string type
   787	                if isinstance(parsed_value, str):
   788	                    arguments[arg_key] = parsed_value
   789	                elif isinstance(parsed_value, (dict, list)):
   790	                    # If parsed as dict/list but schema says string, convert to JSON string
   791	                    arguments[arg_key] = json.dumps(parsed_value, ensure_ascii=False)
   792	                else:
   793	                    arguments[arg_key] = str(parsed_value)
   794	            elif arg_type is None:
   795	                # If type is not defined, keep the parsed value as-is
   796	                arguments[arg_key] = parsed_value if is_good_json else arg_value
   797	            else:
   798	                # For other types (number, object, array, etc.), use parsed value
   799	                arguments[arg_key] = parsed_value if is_good_json else arg_value
   800	
   801	        return arguments
