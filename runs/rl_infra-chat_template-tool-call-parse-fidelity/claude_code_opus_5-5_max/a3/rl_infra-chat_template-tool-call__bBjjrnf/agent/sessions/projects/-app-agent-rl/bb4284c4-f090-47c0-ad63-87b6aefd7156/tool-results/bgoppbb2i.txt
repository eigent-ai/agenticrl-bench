    32	class BaseReasoningFormatDetector:
    33	    """Base class providing two sets of interfaces: one-time and streaming incremental."""
    34	
    35	    def __init__(
    36	        self,
    37	        think_start_token: str,
    38	        think_end_token: str,
    39	        think_excluded_tokens: Optional[List[str]] = None,
    40	        force_reasoning: bool = False,
    41	        stream_reasoning: bool = True,
    42	        tool_start_token: Optional[str] = None,
    43	        continue_final_message: bool = False,
    44	        previous_content: str = "",
    45	        thinks_internally: bool = False,
    46	        reasoning_default: str = "always",
    47	        force_nonempty_content: bool = False,
    48	    ):
    49	        self.think_start_token = think_start_token
    50	        self.think_end_token = think_end_token
    51	        self.think_excluded_tokens = think_excluded_tokens
    52	        self.tool_start_token = tool_start_token
    53	        self.force_reasoning = force_reasoning
    54	        self._in_reasoning = force_reasoning
    55	        self.stream_reasoning = stream_reasoning
    56	        self.thinks_internally = thinks_internally
    57	        self.reasoning_default = reasoning_default
    58	
    59	        self._buffer = ""
    60	        self.stripped_think_start = False
    61	        self.think_start_self_label = ""
    62	
    63	        self._force_nonempty_content = force_nonempty_content
    64	        self._accumulated_reasoning = ""
    65	
    66	        self.continue_final_message = continue_final_message
    67	        if self.continue_final_message:
    68	            self.previous_content = previous_content
    69	            self.previous_count = len(previous_content)
    70	        else:
    71	            self.previous_content = ""
    72	            self.previous_count = 0
    73	
    74	        if self.think_start_token in self.previous_content:
    75	            self._in_reasoning = True
    76	        if self.think_end_token in self.previous_content:
    77	            self._in_reasoning = False

    79	    def _maybe_apply_force_nonempty_content(
    80	        self, ret: StreamingParseResult
    81	    ) -> StreamingParseResult:
    82	        if self._force_nonempty_content and not ret.normal_text:
    83	            ret.normal_text, ret.reasoning_text = ret.reasoning_text, ret.normal_text
    84	        return ret

    86	    def detect_and_parse(self, text: str) -> StreamingParseResult:
    87	        """
    88	        One-time parsing: Detects and parses reasoning sections in the provided text.
    89	        Returns both reasoning content and normal text separately.
    90	        """
    91	        return self._maybe_apply_force_nonempty_content(
    92	            self._detect_and_parse_impl(text)
    93	        )
    94	
    95	    def _detect_and_parse_impl(self, text: str) -> StreamingParseResult:
    96	        in_reasoning = self._in_reasoning or self.think_start_token in text
    97	
    98	        if not in_reasoning:
    99	            return StreamingParseResult(normal_text=text)
   100	
   101	        # The text is considered to be in a reasoning block.
   102	        think_start_text = self.think_start_token + self.think_start_self_label
   103	        processed_text = text
   104	        while processed_text.startswith(think_start_text):
   105	            processed_text = processed_text[len(think_start_text) :]
   106	
   107	        if (
   108	            self.think_end_token not in processed_text
   109	            and self.think_end_token not in self.previous_content
   110	        ):
   111	            # Check for tool_start_token interruption
   112	            if (
   113	                in_reasoning
   114	                and self.tool_start_token is not None
   115	                and self.tool_start_token in processed_text
   116	            ):
   117	                # Find the first occurrence of tool_start_token and split there
   118	                tool_idx = processed_text.find(self.tool_start_token)
   119	                reasoning_text = processed_text[:tool_idx]
   120	                # Preserve tool_start_token in normal text
   121	                normal_text = processed_text[tool_idx:]
   122	                return StreamingParseResult(
   123	                    normal_text=normal_text, reasoning_text=reasoning_text
   124	                )
   125	            # Assume reasoning was truncated before end token
   126	            return StreamingParseResult(reasoning_text=processed_text)
   127	
   128	        # Extract reasoning content
   129	        if self.think_end_token in processed_text:
   130	            splits = processed_text.split(self.think_end_token, maxsplit=1)
   131	            reasoning_text = splits[0]
   132	            normal_text = splits[1]
   133	
   134	            return StreamingParseResult(
   135	                normal_text=normal_text, reasoning_text=reasoning_text
   136	            )
   137	        else:
   138	            # think_end_token is in self.previous_content for continue_final_message=True case
   139	            return StreamingParseResult(normal_text=processed_text)
   140	
   141	    def parse_streaming_increment(self, new_text: str) -> StreamingParseResult:
   142	        """
   143	        Streaming incremental parsing for reasoning content.
   144	        Handles partial reasoning tags and content.
   145	
   146	        If stream_reasoning is False:
   147	            Accumulates reasoning content until the end tag is found
   148	        If stream_reasoning is True:
   149	            Streams reasoning content as it arrives
   150	        """
   151	        ret = self._parse_streaming_increment_impl(new_text)
   152	        if self._force_nonempty_content:
   153	            if self._in_reasoning:
   154	                self._accumulated_reasoning += ret.reasoning_text
   155	            else:
   156	                self._accumulated_reasoning = ""
   157	        return ret
   158	
   159	    def _parse_streaming_increment_impl(self, new_text: str) -> StreamingParseResult:
   160	        self._buffer += new_text
   161	        current_text = self._buffer
   162	
   163	        think_start_text = self.think_start_token + self.think_start_self_label
   164	
   165	        # If the current text is a prefix of the think token, keep buffering
   166	        tokens_to_check = [think_start_text, self.think_end_token]
   167	        if self.tool_start_token:
   168	            tokens_to_check.append(self.tool_start_token)
   169	        if any(
   170	            token.startswith(current_text) and token != current_text
   171	            for token in tokens_to_check
   172	        ):
   173	            return StreamingParseResult()
   174	
   175	        # Strip `<think>` token if present
   176	        if not self.stripped_think_start and think_start_text in current_text:
   177	            current_text = current_text.replace(think_start_text, "", 1)
   178	            self.stripped_think_start = True
   179	            self._in_reasoning = True
   180	
   181	        # Handle end of reasoning block
   182	        if self._in_reasoning and self.think_end_token in current_text:
   183	            end_idx = current_text.find(self.think_end_token)
   184	
   185	            reasoning_text = current_text[:end_idx]
   186	
   187	            self._buffer = ""
   188	            self._in_reasoning = False
   189	            normal_text = current_text[end_idx + len(self.think_end_token) :]
   190	
   191	            return StreamingParseResult(
   192	                normal_text=normal_text, reasoning_text=reasoning_text
   193	            )
   194	
   195	        # Continue with reasoning content
   196	        if self._in_reasoning:
   197	            # Check for tool_start_token interruption
   198	            if self.tool_start_token and self.tool_start_token in current_text:
   199	                tool_idx = current_text.find(self.tool_start_token)
   200	                reasoning_text = current_text[:tool_idx]
   201	                # Preserve tool_start_token in normal text
   202	                normal_text = current_text[tool_idx:]
   203	                self._buffer = ""
   204	                self._in_reasoning = False
   205	                return StreamingParseResult(
   206	                    normal_text=normal_text, reasoning_text=reasoning_text
   207	                )
   208	            if self.stream_reasoning:
   209	                # Stream the content immediately
   210	                self._buffer = ""
   211	                return StreamingParseResult(reasoning_text=current_text)
   212	            else:
   213	                return StreamingParseResult()
   214	
   215	        # If we're not in a reasoning block return as normal text
   216	        if not self._in_reasoning:
   217	            self._buffer = ""
   218	            return StreamingParseResult(normal_text=current_text)
   219	
   220	        return StreamingParseResult()
   221	
   222	    def finish(self) -> StreamingParseResult:
   223	        """
   224	        Called once when the stream ends. If force_nonempty_content is set
   225	        and the stream ended mid-reasoning, reclassifies the accumulated
   226	        reasoning (plus any partial token still buffered) as normal text.
   227	        """
   228	        if self._force_nonempty_content and self._in_reasoning:
   229	            # stream_reasoning=False never clears _buffer, so the opening think
   230	            # token (stripped only from the base class's local view) survives here.
   231	            buffer = self._buffer
   232	            think_start_text = self.think_start_token + self.think_start_self_label
   233	            if buffer.startswith(think_start_text):
   234	                buffer = buffer[len(think_start_text) :]
   235	            normal_text = self._accumulated_reasoning + buffer
   236	            self._accumulated_reasoning = ""
   237	            self._buffer = ""
   238	            if normal_text:
   239	                return StreamingParseResult(normal_text=normal_text)
   240	        return StreamingParseResult()

   443	class GptOssDetector(BaseReasoningFormatDetector):
   444	    """
   445	    Detector for T4-style reasoning format (GPT-OSS), using the HarmonyParser.
   446	    """
   447	
   448	    def __init__(
   449	        self,
   450	        stream_reasoning: bool = True,
   451	        force_reasoning: bool = True,
   452	        continue_final_message: bool = False,
   453	        previous_content: str = "",
   454	        force_nonempty_content: bool = False,
   455	    ):
   456	        super().__init__(
   457	            "<|channel|>analysis<|message|>",
   458	            "<|end|>",
   459	            force_reasoning=force_reasoning,
   460	            stream_reasoning=stream_reasoning,
   461	            continue_final_message=continue_final_message,
   462	            previous_content=previous_content,
   463	            force_nonempty_content=force_nonempty_content,
   464	        )
   465	        self.parser = HarmonyParser()
   466	
   467	    def detect_and_parse(self, text: str) -> StreamingParseResult:
   468	        events = self.parser.parse(text)
   469	        # Flush the buffer for one-shot parsing
   470	        events += self.parser.parse("")
   471	
   472	        reasoning_text = "".join(
   473	            [e.content for e in events if e.event_type == "reasoning"]
   474	        )
   475	        normal_parts = []
   476	        for e in events:
   477	            if e.event_type == "normal":
   478	                normal_parts.append(e.content)
   479	            elif e.event_type == "tool_call":
   480	                # Use raw_text to preserve structural markers for function call detector
   481	                normal_parts.append(e.raw_text if e.raw_text else e.content)
   482	        normal_text = "".join(normal_parts)
   483	        # Tool call events preserve raw text with structural markers
   484	
   485	        return self._maybe_apply_force_nonempty_content(
   486	            StreamingParseResult(
   487	                normal_text=normal_text,
   488	                reasoning_text=reasoning_text,
   489	            )
   490	        )
   491	
   492	    def parse_streaming_increment(self, new_text: str) -> StreamingParseResult:
   493	        events = self.parser.parse(new_text)
   494	
   495	        reasoning_text = "".join(
   496	            [e.content for e in events if e.event_type == "reasoning"]
   497	        )
   498	        normal_parts = []
   499	        for e in events:
   500	            if e.event_type == "normal":
   501	                normal_parts.append(e.content)
   502	            elif e.event_type == "tool_call":
   503	                # Use raw_text to preserve structural markers for function call detector
   504	                normal_parts.append(e.raw_text if e.raw_text else e.content)
   505	        normal_text = "".join(normal_parts)
   506	
   507	        return StreamingParseResult(
   508	            normal_text=normal_text,
   509	            reasoning_text=reasoning_text,
   510	        )

   578	
   579	    In multi-turn chats M3 prefixes earlier non-thinking turns with a bare
   580	    ``</mm:think>``, so a non-thinking reply may open with one stray closer; drop it unless thinking.
   581	    """
   582	
   583	    def __init__(
   584	        self,
   585	        stream_reasoning: bool = True,
   586	        force_reasoning: bool = False,
   587	        continue_final_message: bool = False,
   588	        previous_content: str = "",
   589	        force_nonempty_content: bool = False,
   590	    ):

   592	            "<mm:think>",
   593	            "</mm:think>",
   594	            force_reasoning=force_reasoning,
   595	            stream_reasoning=stream_reasoning,
   596	            continue_final_message=continue_final_message,
   597	            previous_content=previous_content,
   598	        )
   599	        self._lead_buffer = ""
   600	        self._checked_leading_close = False
   601	        self._force_nonempty_content = force_nonempty_content

   603	    def detect_and_parse(self, text: str) -> StreamingParseResult:
   604	        if not self._in_reasoning and text.lstrip().startswith(self.think_end_token):
   605	            text = text.lstrip()[len(self.think_end_token) :]
   606	        ret = super().detect_and_parse(text)
   607	        if self._force_nonempty_content and not ret.normal_text:
   608	            ret.normal_text, ret.reasoning_text = ret.reasoning_text, ret.normal_text

   611	    def parse_streaming_increment(self, new_text: str) -> StreamingParseResult:
   612	        # ``</mm:think>`` is a single token, so a stray leading closer arrives whole.
   613	        if not self._checked_leading_close and not self._in_reasoning:
   614	            self._lead_buffer += new_text
   615	            stripped = self._lead_buffer.lstrip()
   616	            if not stripped:
   617	                return StreamingParseResult()
   618	            self._checked_leading_close = True
   619	            if stripped.startswith(self.think_end_token):
   620	                new_text = stripped[len(self.think_end_token) :]
   621	            else:
   622	                new_text = self._lead_buffer
   623	            self._lead_buffer = ""
   624	            if not new_text:
   625	                return StreamingParseResult()
   626	        return super().parse_streaming_increment(new_text)

   730	class InklingDetector(BaseReasoningFormatDetector):
   731	    """Detector for Inkling typed content blocks."""
   732	
   733	    # Parse the model's sequence of typed content blocks, for example:
   734	    #   <|message_model|><|content_thinking|>reasoning<|end_message|>
   735	    #   <|message_model|><|content_text|>visible answer<|end_message|>
   736	    #   <|content_model_end_sampling|>
   737	    # Special tokens must decode literally so thinking and visible text can be
   738	    # routed to their respective response fields.
   739	    def __init__(
   740	        self,
   741	        stream_reasoning: bool = True,
   742	        force_reasoning: bool = False,
   743	        continue_final_message: bool = False,
   744	        previous_content: str = "",
   745	        force_nonempty_content: bool = False,
   746	    ):
   747	        del force_nonempty_content
   748	        super().__init__(
   749	            CONTENT_THINKING,
   750	            END_MESSAGE,
   751	            force_reasoning=force_reasoning,
   752	            stream_reasoning=stream_reasoning,
   753	            continue_final_message=continue_final_message,
   754	            previous_content=previous_content,
   755	            thinks_internally=False,
   756	            reasoning_default="always",
   757	        )
   758	
   759	        self._kind: str | None = None
   760	        self._pending_header = ""
   761	        self._pending_reasoning = ""
   762	
   763	    def detect_and_parse(self, text: str) -> StreamingParseResult:
   764	        self._buffer = ""
   765	        self._kind = None
   766	        self._pending_header = ""
   767	        self._pending_reasoning = ""
   768	        ret = self._parse_blocks(text)
   769	        if self._kind == "reasoning" and not self.stream_reasoning:
   770	            ret.reasoning_text += self._pending_reasoning
   771	        self._kind = None
   772	        self._pending_header = ""
   773	        self._pending_reasoning = ""
   774	        return ret
   775	
   776	    def parse_streaming_increment(self, new_text: str) -> StreamingParseResult:
   777	        text = self._buffer + new_text
   778	        partial_len = self._partial_control_length(text)
   779	        if partial_len:
   780	            self._buffer = text[-partial_len:]
   781	            text = text[:-partial_len]
   782	        else:
   783	            self._buffer = ""
   784	        return self._parse_blocks(text)
   785	
   786	    def finish(self) -> StreamingParseResult:
   787	        # Flush reasoning buffered under stream_reasoning=False when the stream
   788	        # ends before a control/end token closes the block (e.g. max_tokens cut
   789	        # a thinking block short). Mirrors the non-streaming flush in
   790	        # detect_and_parse; without it the trailing reasoning trace is dropped.
   791	        reasoning_text = ""
   792	        if self._kind == "reasoning" and not self.stream_reasoning:
   793	            reasoning_text = self._pending_reasoning
   794	        self._buffer = ""
   795	        self._pending_reasoning = ""
   796	        self._pending_header = ""
   797	        self._kind = None
   798	        return StreamingParseResult(reasoning_text=reasoning_text)
   799	
   800	    @staticmethod
   801	    def _partial_control_length(text: str) -> int:
   802	        max_token_len = max(map(len, _INKLING_CONTROL_TOKENS))
   803	        for length in range(min(len(text), max_token_len - 1), 0, -1):
   804	            suffix = text[-length:]
   805	            if any(
   806	                len(suffix) < len(token) and token.startswith(suffix)
   807	                for token in _INKLING_CONTROL_TOKENS
   808	            ):
   809	                return length
   810	        return 0
   811	
   812	    def _parse_blocks(self, text: str) -> StreamingParseResult:
   813	        reasoning: list[str] = []
   814	        content: list[str] = []
   815	        saw_control = False
   816	        pos = 0
   817	
   818	        def emit(text: str) -> None:
   819	            if self._kind == "reasoning":
   820	                if self.stream_reasoning:
   821	                    reasoning.append(text)
   822	                else:
   823	                    self._pending_reasoning += text
   824	            elif self._kind == "content":
   825	                content.append(text)
   826	            elif self._kind == "tool":
   827	                content.append(text)
   828	            elif self._kind == "header":
   829	                self._pending_header += text
   830	            elif text:
   831	                # No open block — e.g. a continue_final_message stream resuming
   832	                # mid text block. Route to visible content, matching the
   833	                # no-control-token path below.
   834	                content.append(text)
   835	
   836	        def flush_reasoning() -> None:
   837	            if self._kind == "reasoning" and not self.stream_reasoning:
   838	                reasoning.append(self._pending_reasoning)
   839	                self._pending_reasoning = ""
   840	
   841	        for match in _INKLING_CONTROL_RE.finditer(text):
   842	            saw_control = True
   843	            emit(text[pos : match.start()])
   844	
   845	            token = match.group(0)
   846	            pos = match.end()
   847	            if token == MESSAGE_MODEL:
   848	                if self._kind in (None, "header"):
   849	                    flush_reasoning()
   850	                    self._pending_header = ""
   851	                    self._kind = "header"
   852	                else:
   853	                    # Inside an open block a decoded <|message_model|> string
   854	                    # is payload the model wrote (e.g. quoting the protocol) —
   855	                    # a real header can only follow an end token. Preserve it
   856	                    # instead of rerouting the rest of the block into a header.
   857	                    emit(token)
   858	            elif token in (CONTENT_INVOKE_TOOL_JSON, CONTENT_INVOKE_TOOL_TEXT):
   859	                # Preserve the tool-invocation framing (json and headerless raw
   860	                # text) in content so the tool-call detector receives it.
   861	                flush_reasoning()
   862	                if self._kind == "header":
   863	                    content.extend((MESSAGE_MODEL, self._pending_header, token))
   864	                    self._pending_header = ""
   865	                else:
   866	                    content.append(token)
   867	                self._kind = "tool"
   868	            elif self._kind == "tool":
   869	                content.append(token)
   870	                if token in _INKLING_END_TOKENS:
   871	                    self._kind = None
   872	            elif token in _INKLING_CONTENT_KINDS:
   873	                flush_reasoning()
   874	                self._pending_header = ""
   875	                self._kind = _INKLING_CONTENT_KINDS[token]
   876	            elif token in _INKLING_END_TOKENS:
   877	                flush_reasoning()
   878	                self._pending_header = ""
   879	                self._kind = None
   880	
   881	        tail = text[pos:]
   882	        if saw_control or self._kind is not None:
   883	            emit(tail)
   884	        else:
   885	            content.append(text)
   886	
   887	        return StreamingParseResult(
   888	            normal_text="".join(content),
   889	            reasoning_text="".join(reasoning),
   890	        )

   949	    @staticmethod

   956	    def detect_and_parse(self, text: str) -> StreamingParseResult:
   957	        blocks = self.detect_and_parse_block_sequence(text)
   958	        reasoning_parts = [t for k, t in blocks if k == "reasoning"]
   959	        text_parts = [t for k, t in blocks if k == "text"]

   966	    def detect_and_parse_block_sequence(self, text: str) -> list[tuple[str, str]]:
   967	        """Return an ordered sequence of blocks: [("reasoning"|"text", content), ...]"""
   968	        start_tok = self.think_start_token
   969	        end_tok = self.think_end_token
   970	        blocks: list[tuple[str, str]] = []
   971	        cursor = 0
   972	
   973	        # continue_final_message can resume inside an existing inner
   974	        if self._in_reasoning:
   975	            if (e := text.find(end_tok, cursor)) == -1:
   976	                blocks.extend(self._split_inner_reasoning(text[cursor:]))
   977	                blocks.append(("text", ""))
   978	                return blocks
   979	            blocks.extend(self._split_inner_reasoning(text[cursor:e]))
   980	            cursor = e + len(end_tok)
   981	
   982	        while True:
   983	            if (s := text.find(start_tok, cursor)) == -1:
   984	                # Always include the trailing text block (may be empty)
   985	                blocks.append(("text", text[cursor:]))
   986	                break
   987	            if s > cursor:
   988	                blocks.append(("text", text[cursor:s]))
   989	
   990	            cursor = s + len(start_tok)
   991	            if (e := text.find(end_tok, cursor)) == -1:
   992	                blocks.extend(self._split_inner_reasoning(text[cursor:]))
   993	                blocks.append(("text", ""))
   994	                break
   995	            blocks.extend(self._split_inner_reasoning(text[cursor:e]))
   996	            cursor = e + len(end_tok)
   997	
   998	        last_idx = len(blocks) - 1
   999	        blocks = [
  1000	            (k, t)
  1001	            for i, (k, t) in enumerate(blocks)
  1002	            if not (k == "text" and t == "" and i != last_idx)
  1003	        ]
  1004	
  1005	        return blocks
  1006	
  1007	    def _split_inner_reasoning(self, inner_text: str) -> list[tuple[str, str]]:
  1008	        """
  1009	        Split content inside <|inner_prefix|>...<|inner_suffix|> into:
  1010	          - ("reasoning", <thoughts text>)
  1011	          - ("text", <|tools_prefix|>[...]<|tools_suffix|>) for any tool calls inside reasoning
  1012	        """
  1013	        tool_start = self._tool_start_token
  1014	        tool_end = self._tool_end_token
  1015	        out: list[tuple[str, str]] = []
  1016	        cursor = 0
  1017	
  1018	        while True:
  1019	            if (s := inner_text.find(tool_start, cursor)) == -1:
  1020	                if (tail := inner_text[cursor:]) != "":
  1021	                    out.append(("reasoning", tail))
  1022	                break
  1023	            if s > cursor:
  1024	                out.append(("reasoning", inner_text[cursor:s]))
  1025	
  1026	            if (e := inner_text.find(tool_end, s)) == -1:
  1027	                out.append(("text", inner_text[s:]))
  1028	                break
  1029	
  1030	            out.append(("text", inner_text[s : e + len(tool_end)]))
  1031	            cursor = e + len(tool_end)
  1032	
  1033	        return out
  1034	
  1035	    def parse_streaming_increment(self, new_text: str) -> StreamingParseResult:
  1036	        self._buffer += new_text
  1037	
  1038	        out_reasoning = ""
  1039	        out_normal = ""
  1040	
  1041	        start_tok = self.think_start_token
  1042	        end_tok = self.think_end_token
  1043	        tool_start = self._tool_start_token
  1044	        tool_end = self._tool_end_token
  1045	
  1046	        while True:
  1047	            if not self._in_reasoning:
  1048	                if (s := self._buffer.find(start_tok)) == -1:
  1049	                    if partial := self._ends_with_partial_token(
  1050	                        self._buffer, start_tok
  1051	                    ):
  1052	                        out_normal += self._buffer[:-partial]
  1053	                        self._buffer = self._buffer[-partial:]
  1054	                    else:
  1055	                        out_normal += self._buffer
  1056	                        self._buffer = ""
  1057	                    return StreamingParseResult(
  1058	                        normal_text=out_normal, reasoning_text=out_reasoning
  1059	                    )
  1060	
  1061	                out_normal += self._buffer[:s]
  1062	                self._buffer = self._buffer[s + len(start_tok) :]
  1063	                self._in_reasoning = True
  1064	                self._reasoning_acc = ""
  1065	                self._in_inner_tool = False
  1066	                continue
  1067	
  1068	            if self._in_inner_tool:
  1069	                if (end_pos := self._buffer.find(tool_end)) == -1:
  1070	                    if (
  1071	                        hold := self._ends_with_partial_token(self._buffer, tool_end)
  1072	                    ) != 0:
  1073	                        out_normal += self._buffer[:-hold]
  1074	                        self._buffer = self._buffer[-hold:]
  1075	                    else:
  1076	                        out_normal += self._buffer
  1077	                        self._buffer = ""
  1078	                    return StreamingParseResult(
  1079	                        normal_text=out_normal, reasoning_text=out_reasoning
  1080	                    )
  1081	
  1082	                out_normal += self._buffer[: end_pos + len(tool_end)]
  1083	                self._buffer = self._buffer[end_pos + len(tool_end) :]
  1084	                self._in_inner_tool = False
  1085	                continue
  1086	
  1087	            pos_tool = self._buffer.find(tool_start)
  1088	            pos_end = self._buffer.find(end_tok)
  1089	
  1090	            if pos_tool == -1 and pos_end == -1:
  1091	                if self.stream_reasoning:
  1092	                    if (
  1093	                        hold := max(
  1094	                            self._ends_with_partial_token(self._buffer, end_tok),
  1095	                            self._ends_with_partial_token(self._buffer, tool_start),
  1096	                        )
  1097	                    ) != 0:
  1098	                        out_reasoning += self._buffer[:-hold]
  1099	                        self._buffer = self._buffer[-hold:]
  1100	                    else:
  1101	                        out_reasoning += self._buffer
  1102	                        self._buffer = ""
  1103	                return StreamingParseResult(
  1104	                    normal_text=out_normal, reasoning_text=out_reasoning

  1110	                reasoning_chunk = self._buffer[:pos_end]

  1133	    cohere2_vision Command-A-Plus).
  1134	
  1135	    Generated format (the assistant prefix in the chat template already emits
  1136	    ``<|START_THINKING|>`` when ``reasoning=True``, so the *generated* text
  1137	    typically begins inside the thinking block):
  1138	
  1139	        thinking_content<|END_THINKING|><|START_TEXT|>final_answer<|END_TEXT|>

  1142	    in the prefix and the generated text is just::
  1143	
  1144	        <|START_TEXT|>final_answer<|END_TEXT|>
  1145

  1147	      - ``reasoning_text`` = the thinking block (between START_THINKING and
  1148	        END_THINKING, with the START tag stripped if the model echoed it).
  1149	      - ``normal_text`` = the content between ``<|START_TEXT|>`` and
  1150	        ``<|END_TEXT|>``, with both markers stripped. If no ``<|START_TEXT|>``
  1151	        appears (the model exhausted max_new_tokens still inside thinking),
  1152	        ``normal_text`` is the empty string.
  1153

  1155	    ``special_tokens_map.json`` (``<|START_THINKING|>`` etc.).

  1159	    TEXT_END_TOKEN = "<|END_TEXT|>"

  1170	        continue_final_message: bool = False,
  1171	        previous_content: str = "",
  1172	        force_nonempty_content: bool = False,
  1173	    ):
  1174	        # The chat template puts <|START_THINKING|> in the assistant prefix
  1175	        # when reasoning is enabled, so the *generated* text usually starts
  1176	        # already inside thinking. ``force_reasoning=True`` makes the base
  1177	        # detector treat the leading bytes as reasoning even though the
  1178	        # generated stream typically does not echo <|START_THINKING|>.

  1180	            think_start_token="<|START_THINKING|>",
  1181	            think_end_token="<|END_THINKING|>",

  1183	            stream_reasoning=stream_reasoning,

  1186	            force_nonempty_content=force_nonempty_content,

  1189	        #   1. reasoning  (between START_THINKING [in prefix] and END_THINKING)
  1190	        #   2. either ``<|START_TEXT|>...<|END_TEXT|>`` (final answer) or

  1192	        # When ``reasoning=False`` the chat template emits both START/END
  1193	        # thinking in the prefix and step 1 is empty; the generated stream

  1195	        self._reasoning_done = False
  1196	        self._saw_text_start = False
  1197	        self._saw_text_end = False
  1198	        self._in_action_mode = False

  1200	    @classmethod
  1201	    def _strip_text_markers(cls, raw: str) -> str:
  1202	        """Extract the substring between ``<|START_TEXT|>`` and
  1203	        ``<|END_TEXT|>``. If ``<|START_TEXT|>`` is absent but a
  1204	        ``<|START_ACTION|>`` block is present, the model produced a tool
  1205	        call instead of a text answer -- return the raw text untouched so
  1206	        the downstream tool-call parser can pick up the action block. If
  1207	        neither marker is present (ran out of tokens still inside
  1208	        thinking) return ``""``. If ``<|END_TEXT|>`` is absent (stop token
  1209	        or max_new_tokens cut the stream off inside the text block) return
  1210	        everything after ``<|START_TEXT|>``.
  1211	        """
  1212	        if not raw:
  1213	            return ""
  1214	        s = raw.find(cls.TEXT_START_TOKEN)
  1215	        if s == -1:
  1216	            if cls.ACTION_START_TOKEN in raw:
  1217	                return raw
  1218	            return ""
  1219	        s += len(cls.TEXT_START_TOKEN)
  1220	        tail = raw[s:]
  1221	        e = tail.find(cls.TEXT_END_TOKEN)
  1222	        if e == -1:
  1223	            return tail
  1224	        return tail[:e]

  1226	    def detect_and_parse(self, text: str) -> StreamingParseResult:
  1227	        # Direct parse: split on the (single) ``<|END_THINKING|>`` token if
  1228	        # present. Anything before is reasoning, anything after is the
  1229	        # final-text block. If no END_THINKING but a START_TEXT exists,
  1230	        # we're in the reasoning=False case (chat template emitted both
  1231	        # START/END thinking in the prefix; the model only generated the
  1232	        # text block). Otherwise the model exhausted tokens still thinking
  1233	        # and ``normal_text`` ends up empty -- matching the convention of
  1234	        # the other detectors in this module (DeepSeekR1, Qwen3, ...). The
  1235	        # empty content is propagated as ``message.content = None`` by
  1236	        # serving_chat, and downstream code is expected to treat that as
  1237	        # "no answer" rather than falling back to ``reasoning_content``.
  1238	        end_think_idx = text.find(self.think_end_token)
  1239	        text_start_idx = text.find(self.TEXT_START_TOKEN)
  1240	        action_start_idx = text.find(self.ACTION_START_TOKEN)
  1241	        if end_think_idx != -1:
  1242	            reasoning = text[:end_think_idx]
  1243	            rest = text[end_think_idx + len(self.think_end_token) :]
  1244	        elif text_start_idx != -1:
  1245	            reasoning = text[:text_start_idx]
  1246	            rest = text[text_start_idx:]
  1247	        elif action_start_idx != -1:
  1248	            # reasoning=False + tool call: chat template emitted both
  1249	            # START/END thinking in the prefix, the model only generated
  1250	            # an action block. Treat the prefix before the action block as
  1251	            # (probably empty) reasoning so the action block reaches the
  1252	            # tool-call parser intact.
  1253	            reasoning = text[:action_start_idx]
  1254	            rest = text[action_start_idx:]
  1255	        else:
  1256	            reasoning = text
  1257	            rest = ""
  1258	
  1259	        # Some checkpoints echo the START_THINKING token even though the
  1260	        # chat template put it in the prefix; drop it if so.
  1261	        think_start_text = self.think_start_token + self.think_start_self_label
  1262	        if reasoning.startswith(think_start_text):
  1263	            reasoning = reasoning[len(think_start_text) :]
  1264	
  1265	        return self._maybe_apply_force_nonempty_content(
  1266	            StreamingParseResult(
  1267	                normal_text=self._strip_text_markers(rest),
  1268	                reasoning_text=reasoning,
  1269	            )
  1270	        )

  1272	    def parse_streaming_increment(self, new_text: str) -> StreamingParseResult:
  1273	        """Streaming parse. Custom state machine -- we don't reuse the base
  1274	        class because Cohere's "reasoning=False" path (the model emits no
  1275	        ``<|END_THINKING|>``, just goes straight to a text or action block)
  1276	        is fundamentally incompatible with the base detector's
  1277	        ``force_reasoning`` semantics."""
  1278	        self._buffer += new_text
  1279	        buf = self._buffer
  1280	
  1281	        if not self._reasoning_done:
  1282	            # Look for any marker that ends reasoning: an explicit
  1283	            # END_THINKING, or an implicit transition via the start of the
  1284	            # final-text or action block (reasoning=False case).
  1285	            markers = (
  1286	                (self.think_end_token, "think_end"),
  1287	                (self.TEXT_START_TOKEN, "text"),
  1288	                (self.ACTION_START_TOKEN, "action"),
  1289	            )
  1290	            first_pos = None
  1291	            first_marker = None
  1292	            first_kind = None
  1293	            for marker_text, kind in markers:
  1294	                p = buf.find(marker_text)
  1295	                if p != -1 and (first_pos is None or p < first_pos):
  1296	                    first_pos, first_marker, first_kind = p, marker_text, kind
  1297	            if first_pos is None:
  1298	                # No marker seen yet. Stream the reasoning prefix, but keep
  1299	                # enough tail in the buffer to recognise a marker split
  1300	                # across chunk boundaries.
  1301	                if not self.stream_reasoning:
  1302	                    return StreamingParseResult()
  1303	                max_keep = max(len(m) for m, _ in markers) - 1
  1304	                if len(buf) > max_keep:
  1305	                    head = buf[:-max_keep]
  1306	                    self._buffer = buf[-max_keep:]
  1307	                    return StreamingParseResult(reasoning_text=head)
  1308	                return StreamingParseResult()
  1309	
  1310	            reasoning_chunk = buf[:first_pos]
  1311	            if first_kind == "think_end":
  1312	                self._buffer = buf[first_pos + len(first_marker) :]
  1313	            else:
  1314	                # Implicit reasoning-end: leave the start-of-block marker in
  1315	                # the buffer for the post-thinking branch below to consume.
  1316	                self._buffer = buf[first_pos:]
  1317	            self._reasoning_done = True
  1318	            if reasoning_chunk:
  1319	                return StreamingParseResult(reasoning_text=reasoning_chunk)
  1320	            buf = self._buffer
  1321	
  1322	        # Reasoning is closed. Decide between text-stripping and
  1323	        # action-passthrough on first sight of a marker.
  1324	        if self._in_action_mode:
  1325	            if not buf:
  1326	                return StreamingParseResult()
  1327	            self._buffer = ""
  1328	            return StreamingParseResult(normal_text=buf)
  1329	
  1330	        if not self._saw_text_start:
  1331	            s_text = buf.find(self.TEXT_START_TOKEN)
  1332	            s_action = buf.find(self.ACTION_START_TOKEN)
  1333	            picks = [
  1334	                (p, k) for p, k in ((s_text, "text"), (s_action, "action")) if p != -1
  1335	            ]
  1336	            if not picks:
  1337	                max_keep = (
  1338	                    max(len(self.TEXT_START_TOKEN), len(self.ACTION_START_TOKEN)) - 1
  1339	                )
  1340	                if len(buf) > max_keep:
  1341	                    self._buffer = buf[-max_keep:]
  1342	                return StreamingParseResult()
  1343	            picks.sort()
  1344	            first_pos, first_kind = picks[0]
  1345	            if first_kind == "action":
  1346	                self._in_action_mode = True
  1347	                out_normal = buf[first_pos:]
  1348	                self._buffer = ""
  1349	                return StreamingParseResult(normal_text=out_normal)
  1350	            # Found <|START_TEXT|>. Drop everything up to and including the
  1351	            # marker -- text content streams next.
  1352	            self._buffer = buf[first_pos + len(self.TEXT_START_TOKEN) :]
  1353	            self._saw_text_start = True
  1354	            buf = self._buffer
  1355	
  1356	        if self._saw_text_start and not self._saw_text_end:
  1357	            e = buf.find(self.TEXT_END_TOKEN)
  1358	            if e == -1:
  1359	                # Emit everything except a possible partial END_TEXT tail.
  1360	                keep = len(self.TEXT_END_TOKEN) - 1
  1361	                if len(buf) > keep:
  1362	                    out_normal = buf[:-keep]
  1363	                    self._buffer = buf[-keep:]
  1364	                    return StreamingParseResult(normal_text=out_normal)
  1365	                return StreamingParseResult()
  1366	            out_normal = buf[:e]
  1367	            self._buffer = buf[e + len(self.TEXT_END_TOKEN) :]
  1368	            self._saw_text_end = True
  1369	            return StreamingParseResult(normal_text=out_normal)
  1370	
  1371	        return StreamingParseResult()
