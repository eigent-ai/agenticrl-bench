156|    def _get_local_slice(
157|        self,
158|        forward_batch: ForwardBatch,
159|        can_run_graph: bool,
160|        cuda_graph_batch: Optional[int],
161|    ) -> torch.Tensor:
162|        """Return the device_cache slice for this forward batch, GPU-resident.
163|
164|        Default assumes per-rank-local capture: each rank writes [:local_num_tokens)
165|        to its own device_cache. Subclasses with global-tensor capture semantics
166|        (e.g. shared cuda graph buffer indexed by dp_rank) should override and
167|        consume can_run_graph / cuda_graph_batch.
168|        """
169|        del can_run_graph, cuda_graph_batch  # reserved for subclass override
170|        num_tokens = forward_batch.out_cache_loc.shape[0]
171|        return self.device_cache.buffer[:num_tokens, :, : self.topk_size]

173|    def get_topk(
174|        self,
175|        req_pool_idx: int,
176|        seqlen: int,
177|        req_to_token_pool: ReqToTokenPool,
178|        start_len: int = 0,
179|    ) -> torch.Tensor:
180|        if start_len < 0:
181|            raise ValueError(f"{start_len=} must be non-negative")
182|        start_len = min(start_len, seqlen - 1)
183|        cache_pool_idx = (
184|            req_to_token_pool.req_to_token[req_pool_idx][start_len : seqlen - 1]
185|            .cpu()
186|            .clone()

190|    def on_forward_end(
191|        self,
192|        forward_batch: ForwardBatch,
193|        can_run_graph: bool,
194|        cuda_graph_batch: Optional[int],
195|        no_copy_to_cpu: bool = False,
196|    ) -> Optional[TopkCaptureOutput]:
197|        """If no_copy_to_cpu is True, return a TopkCaptureOutput holding GPU tensors so
198|        the overlap thread can do non-blocking D2H + finalize itself. Otherwise sync
199|        D2H inline and return None (legacy non-overlap path).
200|        """
201|        slice_gpu = self._get_local_slice(
202|            forward_batch, can_run_graph, cuda_graph_batch
203|        )
204|        # get_topk reads req_to_token IDs; attention may have rebound the write
205|        # loc to kernel-facing IDs, which are neither stable nor the same space.
206|        out_cache_loc = forward_batch.out_cache_loc_virtual
207|        if out_cache_loc is None:
208|            out_cache_loc = forward_batch.out_cache_loc
209|        else:
210|            # Kernel batches may be padded; only real request tokens are stored.
211|            slice_gpu = slice_gpu[: out_cache_loc.shape[0]]
212|        out_cache_loc = out_cache_loc[: slice_gpu.shape[0]]
213|        if no_copy_to_cpu:
214|            # Clone before the next overlapping forward reuses these buffers.
215|            return TopkCaptureOutput(

217|                topk=slice_gpu.clone(),
218|                host_cache=self.host_cache,
219|            )
220|        out_cache_loc_cpu = out_cache_loc.cpu()
221|        self.host_cache.buffer[out_cache_loc_cpu] = slice_gpu.cpu()
222|        return None
