147|def _parse_mode(value: Any) -> int:
148|    if isinstance(value, int):
149|        return value
150|    if isinstance(value, str):
151|        return int(value, 8)

155|def _fill(cls, data: dict[str, Any], where: str):
156|    fields = {f.name for f in dataclasses.fields(cls)}
157|    unknown = set(data) - fields
158|    if unknown:
159|        raise PolicyError(f"unknown key(s) in {where}: {', '.join(sorted(unknown))}")
160|    return cls(**data)

172|def check_root_owned(path: str) -> None:
173|    """Refuse files that a non-root user could have written."""
174|    st = os.stat(path)
175|    if st.st_uid != 0 or st.st_mode & (stat.S_IWGRP | stat.S_IWOTH):
176|        raise PolicyError(f"{path} must be owned by root and not writable by group/others")

179|def parse_policy(data: dict[str, Any]) -> Policy:
180|    data = dict(data)
181|    limits = _fill(Limits, dict(data.pop("limits", {}) or {}), "[limits]")
182|    images = _fill(ImageRules, dict(data.pop("images", {}) or {}), "[images]")
183|    network = _fill(NetworkRules, dict(data.pop("network", {}) or {}), "[network]")
184|    for key in ("socket_mode", "audit_log_mode"):
185|        if key in data:
186|            data[key] = _parse_mode(data[key])
187|    policy = _fill(Policy, data, "policy")
188|    policy.limits = limits
189|    policy.images = images
190|    policy.network = network
191|    return policy

194|def load_policy(path: str | None = None, *, strict: bool = False) -> Policy:
195|    path = path or os.environ.get("SANDBOX_BROKER_POLICY") or DEFAULT_POLICY_PATH
196|    if not os.path.exists(path):
197|        raise PolicyError(f"policy file not found: {path}")
198|    if strict:
199|        check_root_owned(path)
200|    raw = Path(path).read_bytes()
201|    if path.endswith(".json"):
202|        data = json.loads(raw)
203|    else:
204|        try:
205|            import tomllib  # type: ignore[import-not-found]
206|        except ModuleNotFoundError:  # pragma: no cover - py<3.11
207|            try:
208|                import tomli as tomllib  # type: ignore[no-redef]
209|            except ModuleNotFoundError as exc:
210|                raise PolicyError("TOML policy needs Python 3.11+ or tomli; use a .json policy") from exc
211|        data = tomllib.loads(raw.decode())
212|    policy = parse_policy(data)
213|    policy.source_path = path
214|    policy.source_sha256 = hashlib.sha256(raw).hexdigest()
215|    base = os.path.dirname(os.path.abspath(path))
216|    rules = policy.images
217|    if rules.pins_file:
218|        pins_path = os.path.join(base, rules.pins_file)
219|        if strict:
220|            check_root_owned(pins_path)
221|        rules.pins.update(json.loads(Path(pins_path).read_text()))
222|    if rules.build_contexts_file:
223|        ctx_path = os.path.join(base, rules.build_contexts_file)
224|        if strict:
225|            check_root_owned(ctx_path)
226|        rules.build_contexts.extend(_read_text_list(ctx_path))
227|    rules.pins = {normalize_ref(k).name_tag: _check_digest(v) for k, v in rules.pins.items()}
228|    rules.build_contexts = sorted({c.lower() for c in rules.build_contexts})
229|    for c in rules.build_contexts:
230|        if not re.fullmatch(r"[0-9a-f]{64}", c):
231|            raise PolicyError(f"build context hash must be 64 hex chars: {c!r}")
232|    return policy

278|def normalize_ref(ref: str) -> ImageRef:
279|    ref = ref.strip()
280|    if not _REF_RE.match(ref) or ".." in ref or "//" in ref:
281|        raise ValueError(f"malformed image reference: {ref!r}")
282|    digest = None
283|    if "@" in ref:
284|        ref, digest = ref.split("@", 1)
285|        _check_digest(digest)
286|    tag = None
287|    slash = ref.rfind("/")
288|    colon = ref.rfind(":")
289|    if colon > slash:
290|        ref, tag = ref[:colon], ref[colon + 1:]
291|        if not re.fullmatch(r"[A-Za-z0-9_][A-Za-z0-9_.\-]{0,127}", tag):
292|            raise ValueError(f"malformed image tag: {tag!r}")
293|    parts = ref.split("/")
294|    if len(parts) > 1 and ("." in parts[0] or ":" in parts[0] or parts[0] == "localhost"):
295|        registry, repo = parts[0], "/".join(parts[1:])
296|    else:
297|        registry, repo = "docker.io", ref
298|    if registry in ("index.docker.io", "registry-1.docker.io"):
299|        registry = "docker.io"
300|    if registry == "docker.io" and "/" not in repo:
301|        repo = f"library/{repo}"
302|    if repo != repo.lower():
303|        raise ValueError(f"repository names must be lowercase: {repo!r}")
304|    if tag is None and digest is None:
305|        tag = "latest"
306|    return ImageRef(registry, repo, tag, digest)

309|def normalize_pattern(pattern: str) -> str:
310|    pattern = pattern.strip()
311|    first = pattern.split("/", 1)[0]
312|    if "/" in pattern and ("." in first or ":" in first or first == "localhost"):
313|        return pattern
314|    if "/" not in pattern.split(":", 1)[0].split("@", 1)[0]:
315|        pattern = f"library/{pattern}"
316|    return f"docker.io/{pattern}"

321|    kind: str  # "registry" | "context"
322|    display: str
323|    pull_ref: str | None = None

327|def resolve_registry_image(rules: ImageRules, ref: str) -> ResolvedImage:
328|    """Map a requested image to the exact reference to pull, or raise PermissionError."""
329|    try:
330|        parsed = normalize_ref(ref)
331|    except ValueError as exc:
332|        raise PermissionError(str(exc)) from exc
333|    patterns = [normalize_pattern(p) for p in rules.allow]
334|    if not any(fnmatch.fnmatchcase(parsed.match_key, p) for p in patterns):
335|        raise PermissionError(f"image {ref!r} is not in the allowed image list")
336|    pin = rules.pins.get(parsed.name_tag) if parsed.tag else None
337|    if pin and parsed.digest and parsed.digest != pin:
338|        raise PermissionError(f"image {ref!r} does not match its pinned digest")
339|    digest = parsed.digest or pin
340|    if rules.require_pin and not digest:
341|        raise PermissionError(f"image {ref!r} has no pinned digest")
342|    return ResolvedImage("registry", parsed.match_key, pull_ref=parsed.pull_ref(digest))

349|def hash_context_dir(root: str | os.PathLike) -> str:
350|    """Content hash of a Docker build context (paths, exec bits and file contents)."""
351|    root = os.fspath(root)
352|    h = hashlib.sha256(b"sandbox-broker-context-v1\n")
353|    entries = []
354|    for dirpath, dirnames, filenames in os.walk(root):
355|        dirnames.sort()
356|        for name in filenames + [d for d in dirnames if os.path.islink(os.path.join(dirpath, d))]:
357|            full = os.path.join(dirpath, name)
358|            rel = os.path.relpath(full, root).replace(os.sep, "/")
359|            entries.append((rel, full))
360|    for rel, full in sorted(entries):
361|        st = os.lstat(full)
362|        if stat.S_ISLNK(st.st_mode):
363|            h.update(f"L {rel} {os.readlink(full)}\n".encode())
364|        elif stat.S_ISREG(st.st_mode):
365|            fh = hashlib.sha256()
366|            with open(full, "rb") as f:
367|                for chunk in iter(lambda: f.read(1 << 20), b""):
368|                    fh.update(chunk)
369|            x = "x" if st.st_mode & 0o111 else "-"
370|            h.update(f"F {rel} {x} {fh.hexdigest()}\n".encode())
371|    return h.hexdigest()
