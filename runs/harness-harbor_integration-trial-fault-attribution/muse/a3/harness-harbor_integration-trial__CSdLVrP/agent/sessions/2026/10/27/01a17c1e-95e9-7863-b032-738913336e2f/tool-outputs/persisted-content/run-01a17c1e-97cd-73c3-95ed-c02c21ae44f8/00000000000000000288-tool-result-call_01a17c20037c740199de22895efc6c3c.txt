90|        """Hostname of the sandbox's command router (``task-<id>.w.modal.host``), known right after create."""
91|        try:
92|            from modal._utils.async_utils import synchronizer

94|            impl = synchronizer._translate_in(sb)
95|            url = getattr(getattr(impl, "_init_command_router_access", None), "url", None)
96|            if not url and getattr(impl, "_command_router_client", None) is not None:
97|                url = impl._command_router_client._server_url
98|            return urlparse(url).hostname if url else None
99|        except Exception:  # noqa: BLE001

102|    async def _find_own_router_host(self) -> str:
103|        """Modal applies a sandbox's network policy through that sandbox's own command router, so
104|        this container must keep its own router host reachable. The container therefore has to
105|        start with ``*.w.modal.host`` allowed: re-applying that starting policy opens the router
106|        connection, which tells us the host; the next update narrows the list."""
107|        from modal._utils.async_utils import synchronizer
108|
109|        starting = sorted(set(self._network.parent_domains) | {"*.w.modal.host"})
110|        try:
111|            await asyncio.wait_for(self._parent._experimental_set_outbound_network_policy.aio(
112|                outbound_domain_allowlist=starting, outbound_cidr_allowlist=list(self._network.parent_cidrs)), 45)
113|        except Exception as exc:  # noqa: BLE001
114|            raise RuntimeError(
115|                "cannot reach this container's own command router; start the container with "
116|                "'*.w.modal.host' in its allowlist (the broker narrows it to exact hosts) "
117|                f"or turn off manage_parent_allowlist ({type(exc).__name__}: {exc})") from exc
118|        router = getattr(synchronizer._translate_in(self._parent), "_command_router_client", None)
119|        host = urlparse(getattr(router, "_server_url", "") or "").hostname
120|        if not host:
121|            raise RuntimeError("could not determine this container's command router host")
122|        return host

124|    async def _apply_parent_policy(self, force: bool = False) -> None:
125|        async with self._net_lock:
126|            own = {self._own_router_host} if self._own_router_host else set()
127|            domains = sorted(set(self._network.parent_domains) | own | set(self._router_hosts.values()))
128|            key = (tuple(domains), tuple(self._network.parent_cidrs))
129|            if key == self._applied and not force:
130|                return
131|            await self._parent._experimental_set_outbound_network_policy.aio(
132|                outbound_domain_allowlist=domains, outbound_cidr_allowlist=list(self._network.parent_cidrs))
133|            self._applied = key

135|    @staticmethod
136|    async def _wait_reachable(host: str, timeout_s: float = 20.0) -> float:
137|        """Wait until a TLS handshake with ``host`` succeeds (the new allowlist entry is in effect)."""
138|        t0 = time.monotonic()
139|        ctx = ssl.create_default_context()
140|        while True:
141|            try:
142|                _, w = await asyncio.wait_for(asyncio.open_connection(host, 443, ssl=ctx, server_hostname=host), 5)
143|                w.close()
144|                return time.monotonic() - t0
145|            except (OSError, asyncio.TimeoutError, ssl.SSLError):
146|                if time.monotonic() - t0 > timeout_s:
147|                    raise
148|                await asyncio.sleep(0.25)

150|    async def _allow_router(self, sandbox_id: str, sb) -> None:
151|        host = self._router_host(sb)
152|        if not host:
153|            log.warning("router host of %s is unknown; exec will fail until it is allowed", sandbox_id)
154|            return
155|        self._router_hosts[sandbox_id] = host
156|        await self._apply_parent_policy()
157|        try:
158|            waited = await self._wait_reachable(host)
159|            log.info("router %s reachable after %.2fs", host, waited)
160|        except Exception as exc:  # noqa: BLE001
161|            log.warning("router %s not reachable yet: %s", host, exc)

222|    async def exec(self, sandbox_id, argv, *, timeout_s, workdir, env, stdin, on_output: OutputCallback):
223|        sb = self._sb(sandbox_id)
224|        proc = await sb.exec.aio(*argv, timeout=int(timeout_s), workdir=workdir, env=env or None, text=False)
225|        if stdin is not None:
226|            proc.stdin.write(stdin)
227|            proc.stdin.write_eof()
228|            await proc.stdin.drain.aio()
229|
230|        async def pump(stream, fd):
231|            async for chunk in stream:
232|                if chunk:
233|                    await on_output(fd, chunk if isinstance(chunk, bytes) else chunk.encode())
234|
235|        await asyncio.gather(pump(proc.stdout, "stdout"), pump(proc.stderr, "stderr"))
236|        rc = await proc.wait.aio()
237|        rc = -1 if rc is None else int(rc)
238|        return ExecOutcome(returncode=rc, timed_out=rc == -1)
