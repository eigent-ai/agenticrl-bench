99|    async def _attempt_tmux_installation(self) -> None:
100|        """Install tmux and asciinema, bounded so a hung install cannot consume
101|        the agent-setup budget.
102|
103|        Recording tools are best-effort. On a closed-internet task apt/PyPI may
104|        be unreachable, so an install command can block producing no output;
105|        without a bound the only ceiling is harbor's agent-setup timeout, which
106|        fails the whole trial. Cap the entire step at ``_TOOL_INSTALL_BUDGET_SEC``
107|        and, on expiry, proceed without recording rather than dying.
108|        """
109|        try:
110|            await asyncio.wait_for(
111|                self._install_recording_tools(),
112|                timeout=self._TOOL_INSTALL_BUDGET_SEC,
113|            )
114|        except asyncio.TimeoutError:
115|            self._logger.warning(
116|                "Tool installation exceeded its %ss budget; proceeding without "
117|                "guaranteed tmux/asciinema (session recording may be "
118|                "unavailable).",
119|                self._TOOL_INSTALL_BUDGET_SEC,
120|            )

437|    def _batch_keys_for_send(self, keys: list[str]) -> list[list[str]]:
438|        """Group *keys* into batches whose send-keys command fits the limit.
439|
440|        Every key must individually fit in a single command; oversized keys
441|        (see ``_key_requires_paste``) are delivered via ``_paste_key``.
442|        """
443|        max_len = self._TMUX_SEND_KEYS_MAX_COMMAND_LENGTH
444|        prefix_len = self._utf8_len(self._send_keys_prefix)
445|
446|        batches: list[list[str]] = []
447|        batch: list[str] = []
448|        batch_len = prefix_len
449|
450|        for key in keys:
451|            escaped_len = self._utf8_len(shlex.quote(key))
452|            if escaped_len > self._max_escaped_key_length:
453|                raise ValueError(
454|                    "key exceeds the tmux send-keys command-length limit; "
455|                    "it must be sent via a paste buffer instead"
456|                )
457|            addition = 1 + escaped_len  # space + quoted key
458|            if batch and batch_len + addition > max_len:
459|                batches.append(batch)
460|                batch = []
461|                batch_len = prefix_len
462|            batch.append(key)
463|            batch_len += addition
464|
465|        if batch:
466|            batches.append(batch)
467|        return batches
