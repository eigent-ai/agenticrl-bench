118|# Modal creates an ephemeral Secret for each Secret.from_dict object. Reusing one
119|# object per env payload avoids workspace-wide create-rate spikes during startup.
120|#
121|# The cache is bounded as an LRU: a long-running process that sees many unique
122|# per-exec env payloads would otherwise retain a Secret object for every payload
123|# ever observed. Once the cache exceeds its capacity the least-recently-used
124|# entry is evicted, keeping memory bounded while still collapsing the bursts of
125|# identical payloads that motivate the cache.
126|_ENV_SECRET_CACHE_MAXSIZE = 256
127|_ENV_SECRET_CACHE: OrderedDict[_EnvSecretCacheKey, Any] = OrderedDict()
128|_ENV_SECRET_CACHE_LOCK = Lock()
129|
130|
131|def _env_secret_cache_key(env: Mapping[str, str | None]) -> _EnvSecretCacheKey:
132|    return tuple(sorted(env.items(), key=lambda item: item[0]))
133|
134|
135|def _cached_env_secret(env: Mapping[str, str | None]) -> Any:
136|    key = _env_secret_cache_key(env)
137|    with _ENV_SECRET_CACHE_LOCK:
138|        try:
139|            secret = _ENV_SECRET_CACHE[key]
140|        except KeyError:
141|            secret = Secret.from_dict(dict(env))
142|            _ENV_SECRET_CACHE[key] = secret
143|            if len(_ENV_SECRET_CACHE) > _ENV_SECRET_CACHE_MAXSIZE:
144|                _ENV_SECRET_CACHE.popitem(last=False)
145|        else:
146|            _ENV_SECRET_CACHE.move_to_end(key)

419|    @staticmethod
420|    def _build_host_network_overlay(
421|        environment_dir: Path,
422|        *,
423|        use_prebuilt: bool = False,
424|        extra_compose_paths: list[Path] | None = None,
425|    ) -> str:
426|        """Generate a compose overlay that sets host networking on all services.
427|
428|        Parses service names from the task's docker-compose.yaml and extra
429|        compose overlays so the overlay covers all services regardless of naming
430|        conventions.
431|        Only adds ``build.network: host`` for services that have a build
432|        context (not pure image-based services like redis).
433|
434|        Each service gets ``extra_hosts`` entries mapping every *other*
435|        service name to ``127.0.0.1`` so that Docker DNS hostnames
436|        (e.g. ``redis``, ``agent1``) resolve correctly under host networking.
437|
438|        Only used by the gVisor path; the VM runtime uses the default Docker
439|        bridge and needs no overlay.
440|        """
441|        import yaml
442|
443|        compose_path = environment_dir / "docker-compose.yaml"
444|        services: dict[str, bool] = {}  # name -> has_build
445|        compose_paths = [compose_path, *(extra_compose_paths or [])]
446|        for path in compose_paths:
447|            if not path.exists():
448|                continue
449|            doc = yaml.safe_load(path.read_text())
450|            if doc and "services" in doc:
451|                for name, cfg in doc["services"].items():
452|                    has_build = isinstance(cfg, dict) and "build" in cfg
453|                    services[name] = services.get(name, False) or has_build
454|
455|        # Fallback if parsing fails
456|        if not services:
457|            services = {"main": True, "sidecar": True, "redis": False}
458|
459|        # main always needs host networking.  In build mode it also needs
460|        # build.network: host; in prebuilt mode only network_mode: host.
461|        if use_prebuilt:
462|            services.setdefault("main", False)
463|        else:
464|            services["main"] = True
465|
466|        service_names = list(services.keys())
467|        lines = ["services:"]
468|        for svc, has_build in services.items():
469|            lines.append(f"  {svc}:")
470|            if has_build:
471|                lines.append("    build:")
472|                lines.append("      network: host")
473|            lines.append("    network_mode: host")
474|            # Map all other service names to localhost so Docker DNS
475|            # hostnames work under host networking.
476|            others = [s for s in service_names if s != svc]
477|            if others:
478|                lines.append("    extra_hosts:")
479|                for other in others:
480|                    lines.append(f'      - "{other}:127.0.0.1"')
481|            # NOTE: Do NOT add environment: here — it replaces (not merges)
482|            # the service's entire environment block from the base compose
483|            # file, wiping out AGENT_ID, API keys, etc.
484|        return "\n".join(lines)
