62|def _create_msg(image, build_context, cpu, memory_mb, timeout_s, idle_timeout_s, workdir, env, labels):
63|    msg: dict[str, Any] = {"op": "create"}
64|    if (image is None) == (build_context is None):
65|        raise ValueError("give exactly one of image or build_context")
66|    if image is not None:
67|        msg["image"] = image
68|    else:
69|        msg["build_context_b64"] = base64.b64encode(pack_build_context(build_context)).decode()
70|    for key, val in (("cpu", cpu), ("memory_mb", memory_mb), ("timeout_s", timeout_s),
71|                     ("idle_timeout_s", idle_timeout_s), ("workdir", workdir), ("env", env), ("labels", labels)):
72|        if val is not None:
73|            msg[key] = val
74|    return msg

77|def _exec_msg(sid, argv, timeout, workdir, env, stdin, binary, stream):
78|    if len(argv) == 1 and isinstance(argv[0], (list, tuple)):
79|        argv = tuple(argv[0])
80|    msg: dict[str, Any] = {"op": "exec", "id": sid, "argv": [str(a) for a in argv]}
81|    if timeout is not None:
82|        msg["timeout_s"] = int(timeout)
83|    if workdir is not None:
84|        msg["workdir"] = workdir
85|    if env:
86|        msg["env"] = {k: str(v) for k, v in env.items() if v is not None}
87|    if stdin is not None:
88|        data = stdin.encode() if isinstance(stdin, str) else bytes(stdin)
89|        msg["stdin_b64"] = base64.b64encode(data).decode()
90|    if binary:
91|        msg["binary"] = True
92|    if stream:
93|        msg["stream"] = True
94|    return msg

97|def _exec_result(reply: dict, binary: bool, streamed: dict[str, list] | None) -> ExecResult:
98|    if streamed is not None:
99|        if binary:
100|            out, err = b"".join(streamed["stdout"]), b"".join(streamed["stderr"])
101|        else:
102|            out, err = "".join(streamed["stdout"]), "".join(streamed["stderr"])
103|    elif binary:
104|        out = base64.b64decode(reply.get("stdout_b64", ""))
105|        err = base64.b64decode(reply.get("stderr_b64", ""))
106|    else:
107|        out, err = reply.get("stdout", ""), reply.get("stderr", "")
108|    return ExecResult(returncode=int(reply.get("rc", -1)), stdout=out, stderr=err,
109|                      timed_out=bool(reply.get("timed_out")), stdout_truncated=bool(reply.get("stdout_truncated")),
110|                      stderr_truncated=bool(reply.get("stderr_truncated")),
111|                      duration_s=float(reply.get("duration_s", 0.0)))

118|def pack_build_context(path: str | os.PathLike | bytes) -> bytes:
119|    """tar.gz of a Docker build context directory (the Dockerfile must be at its root)."""
120|    if isinstance(path, (bytes, bytearray)):
121|        return bytes(path)
122|    root = Path(path)
123|    if not (root / "Dockerfile").is_file():
124|        raise ValueError(f"{root} has no Dockerfile")
125|    buf = io.BytesIO()
126|    with tarfile.open(fileobj=buf, mode="w:gz") as tf:
127|        for p in sorted(root.rglob("*")):
128|            if p.is_symlink():
129|                raise ValueError(f"symlinks are not supported in build contexts: {p}")
130|            rel = p.relative_to(root).as_posix()
131|            info = tf.gettarinfo(str(p), arcname=rel)
132|            info.uid = info.gid = 0
133|            info.uname = info.gname = ""
134|            info.mtime = 0
135|            if p.is_file():
136|                with open(p, "rb") as f:
137|                    tf.addfile(info, f)
138|            elif p.is_dir():
139|                tf.addfile(info)
140|    return buf.getvalue()

143|def _tar_dir(local_dir: str | os.PathLike) -> bytes:
144|    buf = io.BytesIO()
145|    with tarfile.open(fileobj=buf, mode="w:gz") as tf:
146|        tf.add(str(local_dir), arcname=".")
147|    return buf.getvalue()

150|def _untar(data: bytes, dest: str | os.PathLike) -> None:
151|    dest = Path(dest)
152|    dest.mkdir(parents=True, exist_ok=True)
153|    with tarfile.open(fileobj=io.BytesIO(data), mode="r:*") as tf:
154|        try:
155|            tf.extractall(dest, filter="data")
156|        except TypeError:  # Python < 3.12 without extraction filters
157|            for m in tf.getmembers():
158|                target = (dest / m.name).resolve()
159|                if not str(target).startswith(str(dest.resolve())) or m.issym() or m.islnk() or m.isdev():
160|                    continue
161|                tf.extract(m, dest)

167|class BrokerClient:
168|    def __init__(self, socket_path: str | None = None, connect_timeout: float = 30.0):
169|        self.socket_path = socket_path or default_socket_path()
170|        self.connect_timeout = connect_timeout
171|
172|    def _connect(self) -> socket.socket:
173|        s = socket.socket(socket.AF_UNIX, socket.SOCK_STREAM)
174|        s.settimeout(self.connect_timeout)
175|        try:
176|            s.connect(self.socket_path)
177|        except (FileNotFoundError, ConnectionRefusedError) as exc:
178|            s.close()
179|            raise BrokerUnavailable(f"sandbox broker is not running at {self.socket_path}") from exc
180|        return s
181|
182|    def call(self, msg: dict, timeout: float | None = 300.0,
183|             on_event: Callable[[dict], None] | None = None) -> dict:
184|        s = self._connect()
185|        try:
186|            s.settimeout(timeout)
187|            s.sendall(encode(msg))
188|            f = s.makefile("rb")
189|            while True:
190|                line = f.readline()
191|                if not line:
192|                    raise BrokerUnavailable(f"broker closed the connection during {msg.get('op')}")
193|                reply = json.loads(line)
194|                if "event" in reply:
195|                    if on_event:
196|                        on_event(reply)
197|                    continue
198|                if not reply.get("ok", "error" not in reply):
199|                    raise error_from_reply(reply)
200|                return reply
201|        except socket.timeout as exc:
202|            raise TimeoutError(f"no reply from broker for {msg.get('op')} within {timeout}s") from exc
203|        finally:
204|            s.close()
205|
206|    def ping(self, timeout: float = 10.0) -> dict:
207|        return self.call({"op": "ping"}, timeout=timeout)
208|
209|    def usage(self) -> dict:
210|        return self.call({"op": "usage"})
211|
212|    def policy(self) -> dict:
213|        return self.call({"op": "policy"})["policy"]
214|
215|    def list(self, all: bool = False) -> list[dict]:
216|        return self.call({"op": "list", **({"all": True} if all else {})})["sandboxes"]
217|
218|    def shutdown(self, exit: bool = True) -> dict:
219|        return self.call({"op": "shutdown", "exit": exit}, timeout=600)["summary"]
220|
221|    def create(self, image: str | None = None, *, build_context: str | os.PathLike | bytes | None = None,
222|               cpu: float | None = None, memory_mb: int | None = None, timeout_s: int | None = None,

312|class AsyncBrokerClient:
313|    def __init__(self, socket_path: str | None = None, connect_timeout: float = 30.0):
314|        self.socket_path = socket_path or default_socket_path()
315|        self.connect_timeout = connect_timeout
316|
317|    async def call(self, msg: dict, timeout: float | None = 300.0,
318|                   on_event: Callable[[dict], Awaitable[None] | None] | None = None) -> dict:
319|        try:
320|            reader, writer = await asyncio.wait_for(
321|                asyncio.open_unix_connection(self.socket_path, limit=1 << 30), self.connect_timeout)
322|        except (FileNotFoundError, ConnectionRefusedError) as exc:
323|            raise BrokerUnavailable(f"sandbox broker is not running at {self.socket_path}") from exc
324|        try:
325|            writer.write(encode(msg))
326|            await writer.drain()
327|
328|            async def read_reply() -> dict:
329|                while True:
330|                    line = await reader.readline()
331|                    if not line:
332|                        raise BrokerUnavailable(f"broker closed the connection during {msg.get('op')}")
333|                    reply = json.loads(line)
334|                    if "event" in reply:
335|                        if on_event:
336|                            res = on_event(reply)
337|                            if asyncio.iscoroutine(res):
338|                                await res
339|                        continue
340|                    return reply
341|
342|            try:
343|                reply = await asyncio.wait_for(read_reply(), timeout)
344|            except asyncio.TimeoutError as exc:
345|                raise TimeoutError(f"no reply from broker for {msg.get('op')} within {timeout}s") from exc
346|        finally:
347|            writer.close()
348|        if not reply.get("ok", "error" not in reply):
349|            raise error_from_reply(reply)
350|        return reply
