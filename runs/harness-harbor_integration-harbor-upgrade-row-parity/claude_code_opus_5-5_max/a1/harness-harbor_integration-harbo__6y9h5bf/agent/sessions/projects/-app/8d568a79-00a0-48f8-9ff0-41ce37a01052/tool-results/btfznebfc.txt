   149	    @override
   150	    def get_model_context_limit(self) -> int:
   151	        """Get the context limit (max input tokens) for the current model.
   152	
   153	        Returns:
   154	            int: The maximum input tokens the model can accept, or a fallback value if unavailable.
   155	        """
   156	        fallback_context_limit = 1000000
   157	
   158	        try:
   159	            model_info = get_model_info(self._lookup_model_name)
   160	            max_input_tokens = model_info.get("max_input_tokens")
   161	
   162	            # Fallback to max_tokens if max_input_tokens not available
   163	            if max_input_tokens is None:
   164	                max_input_tokens = model_info.get("max_tokens")
   165	
   166	            if max_input_tokens is not None:
   167	                return max_input_tokens
   168	
   169	            # Model info exists but doesn't have context limit info
   170	            self._logger.warning(
   171	                f"Model '{self._display_name}' info found but missing context limit fields. "
   172	                f"Using fallback context limit: {fallback_context_limit}"
   173	            )
   174	        except Exception as e:
   175	            self._logger.debug(
   176	                f"Failed to retrieve model info for '{self._display_name}': {e}. "
   177	                f"Using fallback context limit: {fallback_context_limit}"
   178	            )
   179	
   180	        return fallback_context_limit

   206	    def _clean_value(self, value):
   207	        match value:
   208	            case _ if callable(value):
   209	                return None
   210	            case dict():
   211	                return {
   212	                    k: v
   213	                    for k, v in {
   214	                        k: self._clean_value(v) for k, v in value.items()
   215	                    }.items()
   216	                    if v is not None
   217	                }
   218	            case list():
   219	                return [
   220	                    self._clean_value(v)
   221	                    for v in value
   222	                    if self._clean_value(v) is not None
   223	                ]
   224	            case str() | int() | float() | bool():
   225	                return value
   226	            case _:
   227	                return str(value)
   228	
   229	    def _init_logger_fn(self, logging_path: Path):
   230	        def logger_fn(model_call_dict: dict[str, Any]):
   231	            clean_dict = self._clean_value(model_call_dict)
   232	            if isinstance(clean_dict, dict) and "api_key" in clean_dict:
   233	                hash_key = hashlib.sha256(clean_dict["api_key"].encode()).hexdigest()
   234	                clean_dict["api_key_sha256"] = hash_key
   235	                del clean_dict["api_key"]
   236	
   237	            if isinstance(clean_dict, dict) and "x-api-key" in clean_dict:
   238	                hash_key = hashlib.sha256(clean_dict["x-api-key"].encode()).hexdigest()
   239	                clean_dict["x-api-key_sha256"] = hash_key
   240	                del clean_dict["x-api-key"]
   241	
   242	            # Only save post_api_call (response) to preserve logprobs
   243	            # logger_fn is called multiple times with different event types, and
   244	            # write_text() overwrites the file each time. By filtering for
   245	            # post_api_call only, we ensure the response with logprobs is saved.
   246	            log_event_type = clean_dict.get("log_event_type", "unknown")
   247	            if log_event_type == "post_api_call":
   248	                logging_path.write_text(
   249	                    json.dumps(
   250	                        clean_dict,
   251	                        indent=4,
   252	                    )
   253	                )
   254	
   255	        return logger_fn

   257	    @retry(
   258	        stop=stop_after_attempt(3),
   259	        wait=wait_exponential(multiplier=1, min=4, max=15),
   260	        retry=(
   261	            # To avoid asyncio.CancelledError retries which inherits from BaseException
   262	            # rather than Exception
   263	            retry_if_exception_type(Exception)
   264	            & retry_if_not_exception_type(
   265	                (
   266	                    ContextLengthExceededError,
   267	                    OutputLengthExceededError,
   268	                    LiteLLMAuthenticationError,
   269	                )
   270	            )
   271	        ),
   272	        reraise=True,

   274	    @override
   275	    async def call(
   276	        self,
   277	        prompt: str,
   278	        message_history: list[dict[str, Any] | Message] = [],
   279	        response_format: dict[str, Any] | type[BaseModel] | None = None,
   280	        logging_path: Path | None = None,
   281	        **kwargs,
   282	    ) -> LLMResponse:
   283	        if self._use_responses_api:
   284	            return await self._call_responses(
   285	                prompt, message_history, response_format, logging_path, **kwargs
   286	            )
   287	
   288	        if response_format is not None and not self._supports_response_format:
   289	            if isinstance(response_format, dict):
   290	                schema = json.dumps(response_format, indent=2)
   291	            else:
   292	                schema = json.dumps(response_format.model_json_schema(), indent=2)
   293	            prompt = self._structured_response_prompt_template.format(
   294	                schema=schema, prompt=prompt
   295	            )
   296	            response_format = None
   297	
   298	        # Prepare messages with caching for Anthropic models
   299	        messages = message_history + [{"role": "user", "content": prompt}]
   300	        messages = add_anthropic_caching(messages, self._model_name)
   301	
   302	        try:
   303	            # Build completion_kwargs with all parameters
   304	            completion_kwargs = {
   305	                **self._build_base_kwargs(logging_path),
   306	                "messages": messages,
   307	                "response_format": response_format,
   308	                "reasoning_effort": self._reasoning_effort,
   309	            }
   310	            if self._temperature is not None:
   311	                completion_kwargs["temperature"] = self._temperature
   312	
   313	            # Add logprobs and return_token_ids if rollout details collection is enabled
   314	            if self._collect_rollout_details:
   315	                completion_kwargs["logprobs"] = True
   316	                # Request token IDs from provider (supported by vLLM)
   317	                # Note: Some providers (e.g., OpenAI) will reject this parameter, but we'll catch and retry without it
   318	                if "extra_body" not in completion_kwargs:
   319	                    completion_kwargs["extra_body"] = {}
   320	                extra_body: dict[str, Any] = completion_kwargs["extra_body"]  # ty: ignore[invalid-assignment]
   321	                extra_body["return_token_ids"] = True
   322	
   323	            # Add any additional kwargs, deep-merging extra_body to preserve
   324	            # internally-set fields (e.g., return_token_ids) when caller also
   325	            # passes extra_body via llm_call_kwargs.
   326	            if "extra_body" in completion_kwargs and "extra_body" in kwargs:
   327	                existing_extra_body: dict[str, Any] = completion_kwargs["extra_body"]  # ty: ignore[invalid-assignment]
   328	                new_extra_body: dict[str, Any] = kwargs.pop("extra_body")
   329	                completion_kwargs["extra_body"] = {
   330	                    **existing_extra_body,
   331	                    **new_extra_body,
   332	                }
   333	            elif "extra_body" in kwargs:
   334	                kwargs["extra_body"] = {**kwargs["extra_body"]}
   335	            completion_kwargs.update(kwargs)
   336	
   337	            # Add thinking parameter for Anthropic models if max_thinking_tokens is set
   338	            if self._max_thinking_tokens is not None and (
   339	                "anthropic" in self._model_name.lower()
   340	                or "claude" in self._model_name.lower()
   341	            ):
   342	                budget = self._max_thinking_tokens
   343	                if budget < 1024:
   344	                    self._logger.warning(
   345	                        f"max_thinking_tokens={budget} is below minimum of 1024. "
   346	                        "Using minimum value of 1024."
   347	                    )
   348	                    budget = 1024
   349	                completion_kwargs["thinking"] = {
   350	                    "type": "enabled",
   351	                    "budget_tokens": budget,
   352	                }
   353	
   354	            # Add session_id to extra_body if available
   355	            if self._session_id is not None:
   356	                if "extra_body" not in completion_kwargs:
   357	                    completion_kwargs["extra_body"] = {}
   358	                extra_body: dict[str, Any] = completion_kwargs["extra_body"]  # ty: ignore[invalid-assignment]
   359	                extra_body["session_id"] = self._session_id
   360	
   361	                # Also send the session id as the X-Session-ID HTTP header. Some routers like
   362	                # vLLM router (consistent_hash policy) key session affinity on this
   363	                # header
   364	                if "extra_headers" not in completion_kwargs:
   365	                    completion_kwargs["extra_headers"] = {}
   366	                extra_headers: dict[str, Any] = completion_kwargs["extra_headers"]  # ty: ignore[invalid-assignment]
   367	                extra_headers["X-Session-ID"] = self._session_id
   368	
   369	            try:
   370	                response = await litellm.acompletion(**completion_kwargs)
   371	            except LiteLLMBadRequestError as e:
   372	                # If provider (e.g., OpenAI) rejects extra_body parameters, retry without them
   373	                # Some providers reject custom parameters like: return_token_ids, session_id, etc.
   374	                error_msg = str(e)
   375	                if (
   376	                    "Unrecognized request argument" in error_msg
   377	                    and "extra_body" in completion_kwargs
   378	                ):
   379	                    rejected_params = []
   380	                    if "return_token_ids" in error_msg:
   381	                        rejected_params.append("return_token_ids")
   382	                    if "session_id" in error_msg:
   383	                        rejected_params.append("session_id")
   384	
   385	                    if rejected_params:
   386	                        self._logger.warning(
   387	                            f"Provider {self._model_name} rejected extra_body parameters: {', '.join(rejected_params)}. "
   388	                            f"Retrying without them. Token IDs will not be available."
   389	                        )
   390	                        extra_body_val = completion_kwargs.get("extra_body")
   391	                        if extra_body_val and isinstance(extra_body_val, dict):
   392	                            extra_body: dict[str, Any] = extra_body_val
   393	                            for param in rejected_params:
   394	                                if param in extra_body:
   395	                                    del extra_body[param]
   396	
   397	                            if not extra_body:
   398	                                del completion_kwargs["extra_body"]
   399	
   400	                        response = await litellm.acompletion(**completion_kwargs)
   401	                    else:
   402	                        raise e
   403	                else:
   404	                    raise e
   405	        except Exception as e:
   406	            self._handle_litellm_error(e)
   407	
   408	        if isinstance(response, CustomStreamWrapper):
   409	            raise NotImplementedError("Streaming is not supported for T bench yet")
   410	
   411	        # Extract usage information
   412	        usage_info = self._extract_usage_info(response)
   413	
   414	        # Extract and process token IDs if rollout details collection is enabled
   415	        prompt_token_ids = None
   416	        completion_token_ids = None
   417	        logprobs = None
   418	        extra = None
   419	
   420	        if self._collect_rollout_details:
   421	            prompt_token_ids, completion_token_ids = self._extract_token_ids(response)
   422	            logprobs = self._extract_logprobs(response)
   423	            extra = self._extract_provider_extra(response)
   424	
   425	        choice = response["choices"][0]
   426	        message = choice["message"]
   427	        content = message.get("content") or ""
   428	        reasoning_content = message.get("reasoning_content")
   429	
   430	        # Sometimes the LLM returns a response with a finish reason of "length"
   431	        # This typically means we hit the max_tokens limit, not the context window
   432	        if choice.get("finish_reason") == "length":
   433	            # Create exception with truncated response attached
   434	            exc = OutputLengthExceededError(
   435	                f"Model {self._model_name} hit max_tokens limit. "
   436	                f"Response was truncated. Consider increasing max_tokens if possible.",
   437	                truncated_response=content,
   438	            )
   439	            raise exc
   440	
   441	        return LLMResponse(
   442	            content=content,
   443	            reasoning_content=reasoning_content,
   444	            model_name=response.get("model"),
   445	            usage=usage_info,
   446	            prompt_token_ids=prompt_token_ids,
   447	            completion_token_ids=completion_token_ids,
   448	            logprobs=logprobs,
   449	            extra=extra,
   450	        )

   452	    def _extract_token_ids(self, response) -> tuple[list[int] | None, list[int] | None]:
   453	        """Extract token IDs from a response.
   454	
   455	        Returns:
   456	            Tuple of (prompt_token_ids, completion_token_ids) where:
   457	            - prompt_token_ids: Full prompt token IDs including conversation history
   458	            - completion_token_ids: Token IDs for the generated completion as provided by the provider
   459	        """
   460	        try:
   461	            # Extract completion token IDs
   462	            completion_token_ids = None
   463	            choices = response.get("choices", [])
   464	            if choices:
   465	                choice = choices[0]
   466	                if (
   467	                    hasattr(choice, "provider_specific_fields")
   468	                    and choice.provider_specific_fields
   469	                ):
   470	                    provider_token_ids = choice.provider_specific_fields.get(
   471	                        "token_ids"
   472	                    )
   473	                    if provider_token_ids and isinstance(provider_token_ids, list):
   474	                        completion_token_ids = provider_token_ids
   475	
   476	            # Extract full prompt token IDs (including conversation history)
   477	            prompt_token_ids = getattr(response, "prompt_token_ids", None)
   478	
   479	            return prompt_token_ids, completion_token_ids
   480	
   481	        except Exception as e:
   482	            self._logger.debug(f"Error extracting token IDs: {e}")
   483	            return None, None

   485	    def _extract_logprobs(self, response) -> list[float] | None:
   486	        """Extract logprobs from a response.
   487	
   488	        Args:
   489	            response: The LLM response object
   490	
   491	        Returns:
   492	            List of log probabilities for each token, or None if not available.
   493	        """
   494	        try:
   495	            choices = response.get("choices", [])
   496	            if not choices:
   497	                return None
   498	
   499	            logprobs_data = choices[0].get("logprobs")
   500	            if not logprobs_data:
   501	                return None
   502	
   503	            content = logprobs_data.get("content", [])
   504	            return [
   505	                token_data["logprob"]
   506	                for token_data in content
   507	                if "logprob" in token_data
   508	            ]
   509	        except (KeyError, TypeError, IndexError):
   510	            return None

   512	    def _extract_provider_extra(self, response) -> dict[str, Any] | None:
   513	        """Extract non-token_ids fields from provider_specific_fields.
   514	
   515	        Returns all fields from provider_specific_fields except 'token_ids'
   516	        (which is handled separately by _extract_token_ids). This captures
   517	        provider-specific data such as router expert indices for MoE models.
   518	
   519	        Returns:
   520	            Dictionary of extra provider-specific data, or None if not available.
   521	        """
   522	        try:
   523	            choices = response.get("choices", [])
   524	            if not choices:
   525	                return None
   526	
   527	            choice = choices[0]
   528	            if (
   529	                not hasattr(choice, "provider_specific_fields")
   530	                or not choice.provider_specific_fields
   531	            ):
   532	                return None
   533	
   534	            extra = {
   535	                k: v
   536	                for k, v in choice.provider_specific_fields.items()
   537	                if k != "token_ids"
   538	            }
   539	            return extra if extra else None
   540	
   541	        except Exception as e:
   542	            self._logger.debug(f"Error extracting provider extra fields: {e}")
   543	            return None

   545	    def _extract_cost(self, response) -> float:
   546	        """Extract cost from a response's _hidden_params or compute via litellm.
   547	
   548	        Args:
   549	            response: The LLM response object
   550	
   551	        Returns:
   552	            The cost in USD, or 0.0 if unavailable.
   553	        """
   554	        cost = 0.0
   555	        if hasattr(response, "_hidden_params"):
   556	            hidden_params = response._hidden_params
   557	            if isinstance(hidden_params, dict):
   558	                cost = hidden_params.get("response_cost", 0.0) or 0.0
   559	
   560	        if cost == 0.0:
   561	            try:
   562	                cost = litellm.completion_cost(completion_response=response) or 0.0
   563	            except Exception:
   564	                cost = 0.0
   565	
   566	        return float(cost)

   568	    def _extract_usage_info(self, response) -> UsageInfo | None:
   569	        """Extract token usage and cost from a completion API response.
   570	
   571	        Args:
   572	            response: The LLM response object
   573	
   574	        Returns:
   575	            UsageInfo with token counts and cost, or None if not available.
   576	        """
   577	        try:
   578	            # LiteLLM returns a ModelResponse object with a usage attribute
   579	            # See: https://docs.litellm.ai/docs/completion/prompt_caching
   580	            usage = response.usage
   581	
   582	            prompt_tokens = getattr(usage, "prompt_tokens", 0)
   583	            completion_tokens = getattr(usage, "completion_tokens", 0)
   584	
   585	            # Get cache tokens from prompt_tokens_details.cached_tokens
   586	            cache_tokens = 0
   587	            if hasattr(usage, "prompt_tokens_details"):
   588	                prompt_tokens_details = usage.prompt_tokens_details
   589	                if prompt_tokens_details is not None:
   590	                    cache_tokens = (
   591	                        getattr(prompt_tokens_details, "cached_tokens", 0) or 0
   592	                    )
   593	
   594	            cost = self._extract_cost(response)
   595	
   596	            return UsageInfo(
   597	                prompt_tokens=prompt_tokens,
   598	                completion_tokens=completion_tokens,
   599	                cache_tokens=cache_tokens,
   600	                cost_usd=cost,
   601	            )
   602	        except (AttributeError, TypeError):
   603	            return None
   604	
   605	    def _extract_responses_usage_info(self, response) -> UsageInfo | None:
   606	        """Extract token usage and cost from a responses API response.
   607	
   608	        Args:
   609	            response: The responses API response object
   610	
   611	        Returns:
   612	            UsageInfo with token counts and cost, or None if not available.
   613	        """
   614	        if not hasattr(response, "usage") or response.usage is None:
   615	            return None
   616	
   617	        usage = response.usage
   618	        prompt_tokens = getattr(usage, "input_tokens", 0) or 0
   619	        completion_tokens = getattr(usage, "output_tokens", 0) or 0
   620	        cost = self._extract_cost(response)
   621	
   622	        return UsageInfo(
   623	            prompt_tokens=prompt_tokens,
   624	            completion_tokens=completion_tokens,
   625	            cache_tokens=0,
   626	            cost_usd=cost,
   627	        )

   629	    def _handle_litellm_error(self, e: Exception) -> NoReturn:
   630	        """Translate litellm exceptions into harbor exceptions.
   631	
   632	        Always re-raises; never returns normally.
   633	        """
   634	        if isinstance(e, LiteLLMContextWindowExceededError):
   635	            raise ContextLengthExceededError
   636	        if isinstance(e, LiteLLMAuthenticationError):
   637	            raise e
   638	        if isinstance(e, LiteLLMBadRequestError):
   639	            if self._is_context_length_error(e):
   640	                raise ContextLengthExceededError from e
   641	        raise e
   642	
   643	    def _build_base_kwargs(self, logging_path: Path | None = None) -> dict[str, Any]:
   644	        """Build the base kwargs shared by both completion and responses API calls."""
   645	        logger_fn = (
   646	            self._init_logger_fn(logging_path) if logging_path is not None else None
   647	        )
   648	        return {
   649	            **self._llm_kwargs,
   650	            "model": self._model_name,
   651	            "drop_params": True,
   652	            "logger_fn": logger_fn,
   653	            "api_base": self._api_base,
   654	        }
   655	
   656	    def _is_context_length_error(self, error: LiteLLMBadRequestError) -> bool:
   657	        """Check provider error payloads for context-length overflow signals."""
   658	
   659	        parts = [
   660	            str(error),
   661	            str(getattr(error, "body", "")),
   662	            str(getattr(error, "message", "")),
   663	            str(getattr(error, "error", "")),
   664	        ]
   665	
   666	        combined = " ".join(part.lower() for part in parts if part)
   667	        phrases = (
   668	            "context length exceeded",
   669	            "context_length_exceeded",
   670	            "maximum context length",
   671	            "`inputs` tokens + `max_new_tokens`",
   672	            "model's context length",  # vllm 0.16.0 error message
   673	            "prompt is too long",  # Anthropic via OpenAI-compatible proxies
   674	            "input is too long for requested model",  # Bedrock via proxy
   675	        )
   676	        return any(phrase in combined for phrase in phrases)

   678	    async def _call_responses(
   679	        self,
   680	        prompt: str,
   681	        message_history: list[dict[str, Any] | Message] = [],
   682	        response_format: dict[str, Any] | type[BaseModel] | None = None,
   683	        logging_path: Path | None = None,
   684	        **kwargs,
   685	    ) -> LLMResponse:
   686	        """Call the OpenAI Responses API via litellm.aresponses().
   687	
   688	        When previous_response_id is provided (via kwargs), uses server-side
   689	        state chaining — only the new user prompt is sent. Otherwise, builds
   690	        the full input from message_history + prompt.
   691	        """
   692	        previous_response_id = kwargs.pop("previous_response_id", None)
   693	
   694	        try:
   695	            # Build responses_kwargs
   696	            responses_kwargs: dict[str, Any] = self._build_base_kwargs(logging_path)
   697	
   698	            if self._reasoning_effort is not None:
   699	                responses_kwargs["reasoning"] = {
   700	                    "effort": self._reasoning_effort,
   701	                }
   702	
   703	            # Get max_output_tokens from model info
   704	            max_output_tokens = self.get_model_output_limit()
   705	            if max_output_tokens is not None:
   706	                responses_kwargs["max_output_tokens"] = max_output_tokens
   707	
   708	            if response_format is not None:
   709	                responses_kwargs["response_format"] = response_format
   710	
   711	            if self._temperature is not None:
   712	                responses_kwargs["temperature"] = self._temperature
   713	            responses_kwargs.update(kwargs)
   714	
   715	            if previous_response_id is not None:
   716	                # Server-side state chaining: only send the new prompt
   717	                responses_kwargs["previous_response_id"] = previous_response_id
   718	                responses_kwargs["input"] = prompt
   719	            else:
   720	                # Build full input from message history + new prompt
   721	                input_items = []
   722	                for msg in message_history:
   723	                    role = (
   724	                        msg.get("role", "user")
   725	                        if isinstance(msg, dict)
   726	                        else getattr(msg, "role", "user")
   727	                    )
   728	                    content = (
   729	                        msg.get("content", "")
   730	                        if isinstance(msg, dict)
   731	                        else getattr(msg, "content", "")
   732	                    )
   733	                    input_items.append({"role": role, "content": content})
   734	                input_items.append({"role": "user", "content": prompt})
   735	                responses_kwargs["input"] = input_items
   736	
   737	            response = await litellm.aresponses(**responses_kwargs)
   738	
   739	        except Exception as e:
   740	            self._handle_litellm_error(e)
   741	
   742	        # Extract text content from response.output
   743	        content = ""
   744	        reasoning_content = None
   745	        for output_item in response.output:
   746	            if getattr(output_item, "type", None) == "message":
   747	                for content_part in getattr(output_item, "content", []):
   748	                    if getattr(content_part, "type", None) == "output_text":
   749	                        content += getattr(content_part, "text", "")
   750	
   751	        # Extract usage information
   752	        usage_info = self._extract_responses_usage_info(response)
   753	
   754	        # Check for truncation via response status
   755	        response_status = getattr(response, "status", None)
   756	        if response_status == "incomplete":
   757	            incomplete_details = getattr(response, "incomplete_details", None)
   758	            reason = (
   759	                getattr(incomplete_details, "reason", "unknown")
   760	                if incomplete_details
   761	                else "unknown"
   762	            )
   763	            if reason == "max_output_tokens":
   764	                raise OutputLengthExceededError(
   765	                    f"Model {self._model_name} hit max_tokens limit. "
   766	                    f"Response was truncated.",
   767	                    truncated_response=content,
   768	                )
   769	
   770	        response_id = getattr(response, "id", None)
   771	
   772	        return LLMResponse(
   773	            content=content,
   774	            reasoning_content=reasoning_content,
   775	            model_name=getattr(response, "model", None),
   776	            usage=usage_info,
   777	            response_id=response_id,
   778	        )
