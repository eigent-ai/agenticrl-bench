   123	def pack_build_context(path: str | os.PathLike | bytes) -> bytes:
   124	    """tar.gz of a Docker build context directory (the Dockerfile must be at its root)."""
   125	    if isinstance(path, (bytes, bytearray)):
   126	        return bytes(path)
   127	    root = Path(path)
   128	    if not (root / "Dockerfile").is_file():
   129	        raise ValueError(f"{root} has no Dockerfile")
   130	    buf = io.BytesIO()
   131	    with tarfile.open(fileobj=buf, mode="w:gz") as tf:
   132	        for p in sorted(root.rglob("*")):
   133	            if p.is_symlink():
   134	                raise ValueError(f"symlinks are not supported in build contexts: {p}")
   135	            rel = p.relative_to(root).as_posix()
   136	            info = tf.gettarinfo(str(p), arcname=rel)
   137	            info.uid = info.gid = 0
   138	            info.uname = info.gname = ""
   139	            info.mtime = 0
   140	            if p.is_file():
   141	                with open(p, "rb") as f:
   142	                    tf.addfile(info, f)
   143	            elif p.is_dir():
   144	                tf.addfile(info)
   145	    return buf.getvalue()

   157	    dest.mkdir(parents=True, exist_ok=True)
   158	    with tarfile.open(fileobj=io.BytesIO(data), mode="r:*") as tf:
   159	        try:
   160	            tf.extractall(dest, filter="data")
   161	        except TypeError:  # Python < 3.12 without extraction filters
   162	            for m in tf.getmembers():
   163	                target = (dest / m.name).resolve()
   164	                if not str(target).startswith(str(dest.resolve())) or m.issym() or m.islnk() or m.isdev():
   165	                    continue
   166	                tf.extract(m, dest)

   172	class BrokerClient:
   173	    def __init__(self, socket_path: str | None = None, connect_timeout: float = 30.0):
   174	        self.socket_path = socket_path or default_socket_path()
   175	        self.connect_timeout = connect_timeout
   176	
   177	    def _connect(self) -> socket.socket:
   178	        s = socket.socket(socket.AF_UNIX, socket.SOCK_STREAM)
   179	        s.settimeout(self.connect_timeout)
   180	        try:
   181	            s.connect(self.socket_path)
   182	        except (FileNotFoundError, ConnectionRefusedError) as exc:
   183	            s.close()
   184	            raise BrokerUnavailable(f"sandbox broker is not running at {self.socket_path}") from exc
   185	        return s
   186	
   187	    def call(self, msg: dict, timeout: float | None = 300.0,
   188	             on_event: Callable[[dict], None] | None = None) -> dict:
   189	        s = self._connect()
   190	        try:
   191	            s.settimeout(timeout)
   192	            s.sendall(encode(msg))
   193	            f = s.makefile("rb")
   194	            while True:
   195	                line = f.readline()
   196	                if not line:
   197	                    raise BrokerUnavailable(f"broker closed the connection during {msg.get('op')}")
   198	                reply = json.loads(line)
   199	                if "event" in reply:
   200	                    if on_event:
   201	                        on_event(reply)
   202	                    continue
   203	                if not reply.get("ok", "error" not in reply):
   204	                    raise error_from_reply(reply)
   205	                return reply
   206	        except socket.timeout as exc:
   207	            raise TimeoutError(f"no reply from broker for {msg.get('op')} within {timeout}s") from exc
   208	        finally:
   209	            s.close()
   210	
   211	    def ping(self, timeout: float = 10.0) -> dict:
   212	        return self.call({"op": "ping"}, timeout=timeout)
   213	
   214	    def usage(self) -> dict:
   215	        return self.call({"op": "usage"})
   216	
   217	    def policy(self) -> dict:
   218	        return self.call({"op": "policy"})["policy"]
   219	
   220	    def list(self, all: bool = False) -> list[dict]:
   221	        return self.call({"op": "list", **({"all": True} if all else {})})["sandboxes"]
   222	
   223	    def shutdown(self, exit: bool = True) -> dict:
   224	        return self.call({"op": "shutdown", "exit": exit}, timeout=600)["summary"]
   225	
   226	    def create(self, image: str | None = None, *, build_context: str | os.PathLike | bytes | None = None,
   227	               cpu: float | None = None, memory_mb: int | None = None, timeout_s: int | None = None,
   228	               idle_timeout_s: int | None = None, workdir: str | None = None, env: dict | None = None,
   229	               labels: dict | None = None) -> "Sandbox":
   230	        msg = _create_msg(image, build_context, cpu, memory_mb, timeout_s, idle_timeout_s, workdir, env, labels)
   231	        reply = self.call(msg, timeout=CREATE_REPLY_TIMEOUT)
   232	        return Sandbox(self, reply["id"], reply.get("sandbox", {}))

   317	class AsyncBrokerClient:
   318	    def __init__(self, socket_path: str | None = None, connect_timeout: float = 30.0):
   319	        self.socket_path = socket_path or default_socket_path()
   320	        self.connect_timeout = connect_timeout
   321	
   322	    async def call(self, msg: dict, timeout: float | None = 300.0,
   323	                   on_event: Callable[[dict], Awaitable[None] | None] | None = None) -> dict:
   324	        try:
   325	            reader, writer = await asyncio.wait_for(
   326	                asyncio.open_unix_connection(self.socket_path, limit=1 << 30), self.connect_timeout)
   327	        except (FileNotFoundError, ConnectionRefusedError) as exc:
   328	            raise BrokerUnavailable(f"sandbox broker is not running at {self.socket_path}") from exc
   329	        try:
   330	            writer.write(encode(msg))
   331	            await writer.drain()
   332	
   333	            async def read_reply() -> dict:
   334	                while True:
   335	                    line = await reader.readline()
   336	                    if not line:
   337	                        raise BrokerUnavailable(f"broker closed the connection during {msg.get('op')}")
   338	                    reply = json.loads(line)
   339	                    if "event" in reply:
   340	                        if on_event:
   341	                            res = on_event(reply)
   342	                            if asyncio.iscoroutine(res):
   343	                                await res
   344	                        continue
   345	                    return reply
   346	
   347	            try:
   348	                reply = await asyncio.wait_for(read_reply(), timeout)
   349	            except asyncio.TimeoutError as exc:
   350	                raise TimeoutError(f"no reply from broker for {msg.get('op')} within {timeout}s") from exc
   351	        finally:
   352	            writer.close()
   353	        if not reply.get("ok", "error" not in reply):
   354	            raise error_from_reply(reply)
   355	        return reply
   356	
   357	    async def ping(self, timeout: float = 10.0) -> dict:
   358	        return await self.call({"op": "ping"}, timeout=timeout)

   360	    async def usage(self) -> dict:
   361	        return await self.call({"op": "usage"})

   369	    async def shutdown(self, exit: bool = True) -> dict:
   370	        return (await self.call({"op": "shutdown", "exit": exit}, timeout=600))["summary"]

   374	                     idle_timeout_s: int | None = None, workdir: str | None = None, env: dict | None = None,

   290	def normalize_ref(ref: str) -> ImageRef:
   291	    ref = ref.strip()
   292	    if not _REF_RE.match(ref) or ".." in ref or "//" in ref:
   293	        raise ValueError(f"malformed image reference: {ref!r}")
   294	    digest = None
   295	    if "@" in ref:
   296	        ref, digest = ref.split("@", 1)
   297	        _check_digest(digest)
   298	    tag = None
   299	    slash = ref.rfind("/")
   300	    colon = ref.rfind(":")
   301	    if colon > slash:
   302	        ref, tag = ref[:colon], ref[colon + 1:]
   303	        if not re.fullmatch(r"[A-Za-z0-9_][A-Za-z0-9_.\-]{0,127}", tag):
   304	            raise ValueError(f"malformed image tag: {tag!r}")
   305	    parts = ref.split("/")
   306	    if len(parts) > 1 and ("." in parts[0] or ":" in parts[0] or parts[0] == "localhost"):
   307	        registry, repo = parts[0], "/".join(parts[1:])
   308	    else:
   309	        registry, repo = "docker.io", ref
   310	    if registry in ("index.docker.io", "registry-1.docker.io"):
   311	        registry = "docker.io"
   312	    if registry == "docker.io" and "/" not in repo:
   313	        repo = f"library/{repo}"
   314	    if repo != repo.lower():
   315	        raise ValueError(f"repository names must be lowercase: {repo!r}")
   316	    if tag is None and digest is None:
   317	        tag = "latest"

   371	            entries.append((rel, full))
   372	    for rel, full in sorted(entries):
   373	        st = os.lstat(full)
   374	        if stat.S_ISLNK(st.st_mode):
   375	            h.update(f"L {rel} {os.readlink(full)}\n".encode())
   376	        elif stat.S_ISREG(st.st_mode):
   377	            fh = hashlib.sha256()
   378	            with open(full, "rb") as f:

   380	                    fh.update(chunk)
   381	            x = "x" if st.st_mode & 0o111 else "-"
   382	            h.update(f"F {rel} {x} {fh.hexdigest()}\n".encode())

   119	        raise BadRequest(f"unsupported field(s) for {op}: {', '.join(sorted(unknown))}")
   120	    return op
