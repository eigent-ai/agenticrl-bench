60|class LinearTrajectory:
61|    """State for a linear trajectory.
62|
63|    Tracks the full message history and accumulated token IDs for one session.
64|
65|    Session-generated assistant responses create checkpoints; client-injected assistant messages remain prompt history.
66|
67|    Rollback uses ``generated_checkpoint_message_ends`` instead of inferring checkpoints from message roles.
68|
69|    The typical message sequence is: [system?, user, assistant, tool, assistant, tool, …],
70|    but the agent may retry from an earlier point (e.g. re-running a tool call),
71|    in which case the session is rolled back at most one assistant step.
72|
73|    ``turn_args_history`` stores each successful generation's full resolved request.
74|    Rollback slices it with the token checkpoints. Model rules decide which of
75|    these fields a continuation inherits or must keep unchanged.
76|
77|    ``prepare_token_ids_and_request_args`` resolves arguments and renders from
78|    the selected checkpoint before applying rollback.
79|
80|    Concurrency contract: all mutating methods must be called under ``self.lock``.
81|    """

108|    def prepare_token_ids_and_request_args(
109|        self,
110|        client_args: dict[str, Any],
111|        *,
112|        config: SessionServerConfig,
113|        tito_tokenizer: TITOTokenizer,
114|        message_matcher: SessionMessageMatcher | None = None,
115|    ) -> PreparedChatRequest:
116|        """Return the prepared request with prompt token IDs in `body["input_ids"]`.
117|
118|        In order: select the checkpoint *client_args*'s messages continue
119|        (``_find_rollback_checkpoint``); decide the request args
120|        (``request_args.prepare_chat_request``, which may raise a 400); render
121|        the prompt with those template args; then roll back to the checkpoint
122|        (``_rollback_to_checkpoint``). The args come before the render because
123|        they change the token ids. Both steps precede rollback so a validation
124|        or rendering error leaves session state unchanged.
125|
126|        The caller must hold `self.lock`.
127|        """
128|        matcher = message_matcher if message_matcher is not None else strict_message_matches
129|        request_messages = client_args.get("messages", [])
130|        checkpoint_index = self._find_rollback_checkpoint(request_messages, matcher)
131|        turn_args = self.turn_args_history[checkpoint_index] if checkpoint_index >= 0 else None
132|        prepared = prepare_chat_request(
133|            client_args,
134|            tito_tokenizer,
135|            config=config,
136|            turn_args=turn_args,
137|            evaluation=self.evaluation,
138|            sampling_defaults=self.sampling_defaults,
139|            sampling_support_replay=self.sampling_support_replay,
140|        )
141|        prepared.body["input_ids"] = self._render_token_ids(
142|            request_messages,
143|            checkpoint_index=checkpoint_index,
144|            template_args=prepared.template_args,
145|            tito_tokenizer=tito_tokenizer,
146|            message_matcher=matcher,
147|        )
148|        self._rollback_to_checkpoint(checkpoint_index)
149|        return prepared

151|    def _render_token_ids(
152|        self,
153|        request_messages: list[dict[str, Any]],
154|        *,
155|        checkpoint_index: int,
156|        template_args: dict[str, Any],
157|        tito_tokenizer: TITOTokenizer,
158|        message_matcher: SessionMessageMatcher | None = None,
159|    ) -> list[int]:
160|        """Build prompt token IDs while reusing the selected checkpoint's token prefix.
161|
162|        Use `template_args` for newly rendered tokens, without modifying session state.
163|
164|        Validates that *request_messages* extends the selected checkpoint's history
165|        under *message_matcher* (defaults to the strict matcher) and reuses its stored
166|        token_ids as the pretokenized prefix. When no stored checkpoint is selected
167|        to build on — the first turn, or a retry of the first turn that selects
168|        the empty checkpoint — renders *request_messages* from scratch via
169|        the chat template instead.
170|
171|        Must be called under ``self.lock``.
172|        """
173|        matcher = message_matcher if message_matcher is not None else strict_message_matches
174|
175|        token_ids = self.trajectory_token_ids[checkpoint_index] if checkpoint_index >= 0 else []
176|        if not token_ids:
177|            return tito_tokenizer.apply_chat_template(
178|                request_messages,
179|                add_generation_prompt=True,
180|                tokenize=True,
181|                template_args=template_args,
182|            )
183|
184|        message_end = self.generated_checkpoint_message_ends[checkpoint_index]
185|        stored_messages = self.messages[:message_end]

205|    def update_pretokenized_state(
206|        self,
207|        request_messages: list[dict[str, Any]],
208|        assistant_message: dict[str, Any],
209|        prompt_token_ids: list[int],
210|        completion_token_ids: list[int],
211|        max_trim_tokens: int,
212|        turn_args: dict[str, Any] | None = None,
213|    ) -> None:
214|        """Store raw token IDs after a successful response.
215|
216|        Appends ``prompt_token_ids + completion_token_ids`` as a new checkpoint,
217|        recording the full resolved request in ``turn_args`` alongside.
218|        Validates that the previously stored token_ids are a prefix of the new
219|        checkpoint (tolerating up to ``max_trim_tokens`` trailing differences).
220|        Must be called under ``self.lock``.
221|        """
222|        all_token_ids = prompt_token_ids + completion_token_ids
223|        assert_pretokenized_prefix(
224|            self.token_ids,
225|            all_token_ids,
226|            max_trim_tokens=max_trim_tokens,
227|            request_messages=request_messages,
228|            assistant_message=assistant_message,
229|        )
230|
231|        # Commit the same effective history the tokens were built from (stored
232|        # spellings for the reused prefix, the replay only for the new tail):
233|        # committing the raw replay would let an accepted-but-reserialized
234|        # prefix rewrite stored history, so the next canonical replay could
235|        # no longer match its own session.
236|        self.messages = self.messages + request_messages[len(self.messages) :] + [assistant_message]
237|        self.trajectory_token_ids.append(all_token_ids)
238|        self.turn_args_history.append(deepcopy(turn_args or {}))
239|        self.generated_checkpoint_message_ends.append(len(request_messages) + 1)
240|        self.num_assistant = len(self.generated_checkpoint_message_ends)

242|    def _find_rollback_checkpoint(
243|        self,
244|        request_messages: list[dict[str, Any]],
245|        message_matcher: SessionMessageMatcher,
246|    ) -> int:
247|        """Find the last generated checkpoint in the matching prefix, without mutation.
248|
249|        In agentic workflows the agent may retry from an earlier point — for
250|        example, re-running a tool call with different arguments. When that
251|        happens the new request shares a common prefix with the stored messages
252|        but diverges before the end. This method selects the last generated
253|        assistant checkpoint within the matching prefix, or the empty checkpoint
254|        when the matching prefix holds no generated checkpoint at all. The caller
255|        applies rollback only after argument resolution and rendering succeed.
256|
257|        Only a single-step rollback is allowed (controlled by
258|        ``MAX_ASSISTANT_ROLLBACK_STEPS``). Discarding exactly one generated
259|        checkpoint means the agent is retrying from the preceding checkpoint —
260|        the request shares the stored prefix up to that generated response and
261|        then continues with whatever the agent chooses (same or different tool
262|        result, additional messages, etc.). Any request that would need to
263|        discard more than one generated checkpoint (i.e. jump back across
264|        multiple turns) is rejected with ``MessageValidationError`` and no
265|        state is modified.
266|
267|        Example — agent retries after the first tool call::
268|
269|            stored:  [sys, user, assistant₁, tool₁, assistant₂]
270|                      ───────────────────── ▲
271|                      checkpoint 0 (assistant₁)   checkpoint 1 (assistant₂)
272|
273|            request: [sys, user, assistant₁, tool₁_different, ...]
274|                                             ↑ diverges here (index 3)
275|
276|            match_len = 3  (sys, user, assistant₁ all match)
277|            Last generated checkpoint in matched prefix → assistant₁ (checkpoint 0)
278|            discard_count = 2 - 1 = 1  (≤ MAX_ASSISTANT_ROLLBACK_STEPS)
279|
280|            After preparation succeeds and the caller applies rollback:
281|              messages           = [sys, user, assistant₁]
282|              trajectory_token_ids = [checkpoint_0_ids]
283|              records              = [record_0]
284|              num_assistant        = 1
285|
286|        Example — agent retries the very first turn::
287|
288|            stored:  [user, assistant₁]
289|            request: [user]
290|                           ↑ stored continues past the request (index 1)
291|
292|            match_len = 1  (user matches), no generated checkpoint in the matched prefix
293|            Rollback target → the empty checkpoint (index -1)
294|            discard_count = 1 - 0 = 1  (≤ MAX_ASSISTANT_ROLLBACK_STEPS)
295|
296|            The caller re-renders the prompt from scratch before applying rollback.
297|            After successful preparation and rollback, the session is empty,
298|            so turn 1 regenerates like any later turn.
299|
300|        No rollback is needed when:
301|        - The stored history is empty (return -1).
302|        - *request_messages* is a strict extension of stored messages
303|          (``match_len >= len(stored)``; return the current checkpoint index).
304|        """
305|        stored = self.messages
306|        if not stored or not self.trajectory_token_ids:
307|            return -1
308|
309|        match_len = 0
310|        for i in range(min(len(request_messages), len(stored))):
311|            if message_matcher(stored[i], request_messages[i]):
312|                match_len = i + 1
313|            else:
314|                break
315|
316|        if match_len >= len(stored):
317|            return self.num_assistant - 1
318|
319|        # Only responses generated by this session create checkpoints.
320|        # Client-injected assistant messages won't create new checkpoints.
321|        checkpoint_index = -1
322|        for i in reversed(range(len(self.generated_checkpoint_message_ends))):
323|            if self.generated_checkpoint_message_ends[i] <= match_len:
324|                checkpoint_index = i
325|                break
326|
327|        discard_count = self.num_assistant - (checkpoint_index + 1)
328|        if discard_count > MAX_ASSISTANT_ROLLBACK_STEPS:
329|            raise MessageValidationError(
330|                f"rollback failed: discard_count={discard_count} exceeds "
331|                f"max_assistant_rollback_steps={MAX_ASSISTANT_ROLLBACK_STEPS} "
332|                f"(stored has {len(stored)} messages, "
333|                f"request has {len(request_messages)} messages)"
334|            )
335|
336|        return checkpoint_index
337|
338|    def _rollback_to_checkpoint(self, checkpoint_index: int) -> None:
339|        """Discard history after the selected checkpoint once preparation succeeds."""
340|        discard_count = self.num_assistant - (checkpoint_index + 1)
341|        if discard_count == 0:
342|            return
343|        # No generated checkpoint in the matched prefix means the agent is retrying the
344|        # first turn, so roll back to the empty checkpoint and retain no messages.
345|        rollback_msg_end = self.generated_checkpoint_message_ends[checkpoint_index] if checkpoint_index >= 0 else 0
346|        logger.info(
347|            "Rolling back session: stored %d messages / %d checkpoints -> "
348|            "checkpoint %d (messages[:%d]), discarding %d generated checkpoint(s)",
349|            len(self.messages),
350|            self.num_assistant,
351|            checkpoint_index,
352|            rollback_msg_end,
353|            discard_count,
354|        )
355|
356|        self.messages = self.messages[:rollback_msg_end]
357|        self.trajectory_token_ids = self.trajectory_token_ids[: checkpoint_index + 1]
358|        self.turn_args_history = self.turn_args_history[: checkpoint_index + 1]
359|        self.records = self.records[: checkpoint_index + 1]
360|        self.generated_checkpoint_message_ends = self.generated_checkpoint_message_ends[: checkpoint_index + 1]
361|        self.num_assistant = len(self.generated_checkpoint_message_ends)
