22|def _write_trajectory(agent: BaseInstalledAgent, directory: Path, data: bytes) -> None:
23|    # Each download is independent; replaced/deleted logs need no bookkeeping.
24|    with tempfile.TemporaryDirectory(prefix="harbor-stream-") as temporary:

27|            archive.extractall(logs_dir / "sessions", filter="data")

29|        if trajectory is not None:
30|            directory.mkdir(parents=True, exist_ok=True)
31|            temporary_path = directory / "trajectory.tmp"
32|            temporary_path.write_text(
