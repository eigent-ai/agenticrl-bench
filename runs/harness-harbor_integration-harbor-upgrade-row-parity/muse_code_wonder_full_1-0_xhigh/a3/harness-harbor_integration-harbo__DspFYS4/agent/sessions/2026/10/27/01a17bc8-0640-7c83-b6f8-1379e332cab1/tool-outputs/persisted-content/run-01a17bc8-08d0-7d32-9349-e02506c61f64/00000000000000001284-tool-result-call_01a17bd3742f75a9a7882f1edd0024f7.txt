44|class ArtifactHandler:
45|    """Collects artifacts from agent environments and re-materializes them in
46|    separate verifier environments.
47|
48|    Host layout (under the trial's ``artifacts/`` dir) — a single flat base dir
49|    shared by every compose service:
50|
51|    - ``<abs source path>`` — source-derived entries from ANY service, mirrored
52|      directly under ``artifacts/`` (no per-service subtree). The agent's
53|      conventional publish dir lands at ``artifacts/logs/artifacts/``.
54|    - ``<destination>`` — entries with an explicit (relative) destination.
55|    - ``manifest.json`` — record of every collection attempt.
56|
57|    Because services share the base dir, two services that export the same
58|    source path collide on the host; collection keeps the first and logs a
59|    warning for the rest (it never overwrites).
60|
61|    Verifier-side placement never depends on ``destination``: every entry
62|    re-materializes at its original ``source`` path ("no translation"), and
63|    the convention dir maps to the verifier's convention dir.
64|    """

91|    def _find_conflicting_claim(self, target: Path, source: str) -> str | None:
92|        """Return the source that already claimed a host path overlapping
93|        *target* (equal or nested either way), or None. Re-collection of the
94|        same *source* is never a conflict (multi-step re-runs overwrite their
95|        own files)."""
96|        for claimed_target, claimed_source in self._claimed_targets:
97|            if claimed_source == source:
98|                continue
99|            if (
100|                claimed_target == target
101|                or claimed_target in target.parents
102|                or target in claimed_target.parents
103|            ):
104|                return claimed_source
105|        return None
