37|def _get_agent_server_client() -> httpx.AsyncClient:
38|    """Return a client whose long-running requests survive idle network paths."""
39|    global _agent_server_client
40|    if _agent_server_client is None:
41|        socket_options = [
42|            (socket.SOL_SOCKET, socket.SO_KEEPALIVE, 1),
43|            (socket.IPPROTO_TCP, getattr(socket, "TCP_KEEPIDLE", 4), 60),
44|            (socket.IPPROTO_TCP, getattr(socket, "TCP_KEEPINTVL", 5), 30),
45|            (socket.IPPROTO_TCP, getattr(socket, "TCP_KEEPCNT", 6), 5),
46|        ]
47|        transport = httpx.AsyncHTTPTransport(socket_options=socket_options)
48|        _agent_server_client = httpx.AsyncClient(
49|            transport=transport,
50|            limits=httpx.Limits(max_connections=64, max_keepalive_connections=32),
51|            timeout=None,
52|        )
53|    return _agent_server_client

128|async def abort(args) -> None:
129|    """Teardown hook for oversampling abort (called by sglang_rollout.abort).
130|
131|    When Miles has enough samples and aborts SGLang, the in-flight Harbor trials
132|    keep looping and hitting SGLang until they hit their own max_seq_len/timeout.
133|    Flush the agent server so it cancels those ``/run`` tasks and releases their
134|    containers. No-op unless AGENT_SERVER_URL and session_server_instance_id are
135|    available.
136|    """
137|    agent_server_url = os.getenv("AGENT_SERVER_URL", os.getenv("SWE_AGENT_URL"))
138|
139|    instances = getattr(args, "session_server_instances", None) or []
140|    instance_ids = {instance.instance_id for instance in instances if instance.instance_id}
141|    singular = getattr(args, "session_server_instance_id", None)  # back-compat / child path
142|    if singular:
143|        instance_ids.add(singular)
144|
145|    if not agent_server_url or not instance_ids:
146|        return
147|
148|    headers = None
149|    admin_secret = os.getenv("HARBOR_ADMIN_SECRET")
150|    if admin_secret:
151|        headers = {"Authorization": f"Bearer {admin_secret}"}
152|
153|    for instance_id in instance_ids:
154|        try:
155|            result = await post(
156|                f"{agent_server_url.rstrip('/')}/flush",
157|                {"session_server_instance_id": instance_id},
158|                max_retries=3,
159|                headers=headers,
160|            )
161|            logger.info(f"Flushed agent server {agent_server_url}: {result}")
162|        except Exception as e:
163|            logger.warning(f"Failed to flush agent server {agent_server_url}: {e}")
