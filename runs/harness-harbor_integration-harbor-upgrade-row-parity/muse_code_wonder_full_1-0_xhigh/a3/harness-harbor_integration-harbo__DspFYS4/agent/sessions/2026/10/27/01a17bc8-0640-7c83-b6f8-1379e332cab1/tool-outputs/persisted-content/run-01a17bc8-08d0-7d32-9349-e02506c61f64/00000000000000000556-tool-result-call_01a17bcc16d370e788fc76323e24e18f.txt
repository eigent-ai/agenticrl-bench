1|"""
2|Generic agentic generate function for agent-environment RL training.
3|
4|The agent logic is fully encapsulated in a user-provided async function
5|(--custom-agent-function-path). This generate function only handles:
6|  1. TITO session tracing (OpenAIEndpointTracer)
7|  2. Collecting the worker-assembled training samples (the session server
8|     converts records to samples, truncates and merges in the owning worker)
9|  3. Driver-side metadata application (agent_metadata, session_metadata)
10|
11|Agent function contract:
12|  async def my_agent(
13|      base_url: str,
14|      prompt: ...,
15|      request_kwargs: dict,
16|      metadata: dict,       # sample.metadata — env-specific fields
17|      **kwargs,
18|  ) -> dict | None:
19|      ...
20|
21|  Returning None means no extra metadata to attach.
22|  Returning a dict merges it into every sample's metadata, so downstream
23|  reward models (--custom-rm-path) can read whatever the agent left there.
24|"""

188|def build_chat_request_kwargs(sampling_params: dict[str, Any]) -> dict[str, Any]:
189|    request_kwargs = dict(sampling_params)
190|    key_map = {
191|        "max_new_tokens": "max_tokens",
192|        "min_new_tokens": "min_tokens",
193|        "sampling_seed": "seed",
194|    }
195|    for src, dst in key_map.items():
196|        if src in request_kwargs:
197|            if dst not in request_kwargs:
198|                request_kwargs[dst] = request_kwargs[src]
199|            request_kwargs.pop(src, None)
200|
201|    reserved_keys = {"model", "messages"}
202|    allowed_keys = set(ChatCompletionRequest.model_fields) - reserved_keys
203|    return {key: value for key, value in request_kwargs.items() if key in allowed_keys and value is not None}
