+def _effective_sampling(body):
+    """Manual SGLang ChatCompletionRequest.to_sampling_params + SamplingParams defaults."""
+    def gp(name, default):
+        v = body.get(name)
+        return default if v is None else v
+    ctk = body.get("chat_template_kwargs") or {}
+    max_new = body.get("max_completion_tokens") or body.get("max_tokens")
+    stop = body.get("stop")
+    stop_list = [] if stop is None else ([stop] if isinstance(stop, str) else list(stop))
+    top_k = gp("top_k", -1)
+    temp = gp("temperature", 1.0)
+    if isinstance(temp, (int, float)) and 0 <= temp < 1e-6:
+        temp, top_k = 1.0, 1
+    rf = body.get("response_format")
+    constraint = None
+    if isinstance(rf, dict) and rf.get("type") == "json_object":
+        constraint = ["json_schema", '{"type": "object"}']
+    elif isinstance(rf, dict) and rf.get("type") == "json_schema":
+        constraint = ["json_schema", json.dumps((rf.get("json_schema") or {}).get("schema"), sort_keys=True)]
+    elif isinstance(rf, dict) and rf.get("type") == "structural_tag":
+        constraint = ["structural_tag", json.dumps(rf, sort_keys=True)]
+    elif rf is not None:
+        constraint = ["response_format", json.dumps(rf, sort_keys=True, default=str)]
+    for key in ("regex", "ebnf"):
+        if body.get(key) and constraint is None:
+            constraint = [key, body.get(key)]
+    logit = body.get("logit_bias")
+    sampling = {
+        "temperature": temp, "top_p": gp("top_p", 1.0),
+        "top_k": "all" if top_k == -1 else top_k,
+        "min_p": gp("min_p", 0.0), "repetition_penalty": gp("repetition_penalty", 1.0),
+        "presence_penalty": gp("presence_penalty", 0.0), "frequency_penalty": gp("frequency_penalty", 0.0),
+        "max_new_tokens": max_new, "min_new_tokens": gp("min_tokens", 0),
+        "stop": stop_list,
+        "stop_token_ids": sorted(body.get("stop_token_ids") or []),
+        "stop_regex": body.get("stop_regex"),
+        "n": gp("n", 1), "seed": body.get("seed"), "logit_bias": logit,
+        "ignore_eos": bool(body.get("ignore_eos", False)),
+        "skip_special_tokens": bool(body.get("skip_special_tokens", True)),
+        "spaces_between_special_tokens": bool(ctk.get("spaces_between_special_tokens", True)),
+        "no_stop_trim": bool(body.get("no_stop_trim", False)),
+        "constraint": constraint,
+        "custom_params": body.get("custom_params"),
+        "custom_logit_processor": body.get("custom_logit_processor"),
+        "watermark": body.get("watermark"),
+    }
+    other = {
+        "model": body.get("model", "default"),
+        "stream": bool(body.get("stream", False)),
+        "tools": body.get("tools"), "tool_choice": body.get("tool_choice", "auto"),
+        "parallel_tool_calls": bool(body.get("parallel_tool_calls", True)),
+        "continue_final_message": bool(body.get("continue_final_message", False)),
+        "separate_reasoning": bool(body.get("separate_reasoning", True)),
+        "reasoning_effort": body.get("reasoning_effort") or ctk.get("reasoning_effort"),
+    }

+        other["input_ids"] = body.get("input_ids")

+def render_requests(scen_path, req_path, out_path):
+    """Run under engine python. req_path: [{key, server, body}] -> {key: {...}}"""
+    doc = json.loads(Path(scen_path).read_text())
+    reqs = json.loads(Path(req_path).read_text())
+    from transformers import AutoTokenizer
+    toks = {}
+    for name, spec in doc["servers"].items():
+        tok = AutoTokenizer.from_pretrained(spec["tokenizer_dir"])
+        tok.chat_template = _served_template_text(spec)
+        toks[name] = tok
+    try:
+        from sglang.srt.parser.jinja_template_utils import process_content_for_template_format
+        from sglang.srt.jinja_template_utils import process_content_for_template_format as _alt  # noqa
+    except Exception:
+        process_content_for_template_format = None
+    results = {}
+    for item in reqs:
+        body = item["body"]
+        tok = toks[item["server"]]
+        try:
+            ids = _render_one(tok, body, process_content_for_template_format)
+            results[item["key"]] = {
+                "sha256": hashlib.sha256(",".join(map(str, ids)).encode()).hexdigest(),
+                "len": len(ids)}
+        except BaseException as e:  # noqa: BLE001
+            results[item["key"]] = {"render_error": f"{type(e).__name__}: {e}"[:300]}
+    Path(out_path).write_text(json.dumps(results))
+

+def _render_one(tok, body, process_content):
+    if body.get("input_ids") is not None:
+        return list(body["input_ids"])
+    import copy as _copy
+    messages = []
+    for m in body.get("messages") or []:
+        mm = _copy.deepcopy(m)
+        if mm.get("content") is None:
+            mm["content"] = ""
+        messages.append(mm)
+    # _handle_last_assistant_message equivalent
+    prefix = None
+    if messages and messages[-1].get("role") == "assistant" and isinstance(messages[-1].get("content"), str):
+        if body.get("continue_final_message"):
+            prefix = messages[-1]["content"]
+            messages = messages[:-1]
+        else:
+            messages[-1] = {"role": "user", "content": messages[-1]["content"]}
+    ctk = dict(body.get("chat_template_kwargs") or {})
+    extra = {}
+    if body.get("reasoning_effort") is not None:
+        extra["reasoning_effort"] = body["reasoning_effort"]
+    extra.update(ctk)
+    tools = None
+    if body.get("tools") and body.get("tool_choice", "auto") != "none":
+        tools = body["tools"]
+    rendered = tok.apply_chat_template(
+        messages, tokenize=False, add_generation_prompt=True, tools=tools, **extra)
+    ids = tok.encode(rendered)
+    if prefix:
+        enc = tok.encode(prefix)
+        if enc and enc[0] == tok.bos_token_id:
+            enc = enc[1:]
+        ids = list(ids) + list(enc)
+    return list(ids)
