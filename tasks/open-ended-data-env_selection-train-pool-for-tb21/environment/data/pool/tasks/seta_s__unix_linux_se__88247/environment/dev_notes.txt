# === VIM COMMAND REFERENCE ===
# Quitting and saving
:q              Quit (only if no unsaved changes exist)
:q!             Force quit, discarding unsaved changes
:qa!            Force quit all open windows
:wq             Write changes and quit
:wq!            Force write and quit (overrides vim readonly flag)
:x              Write only if changed, then quit
ZQ              Normal mode: equivalent to :q!
ZZ              Normal mode: equivalent to :wq
# Writing to files
:w              Write current buffer to disk
:w filename     Write buffer to a new file
:w!             Force write (overrides vim readonly flag)
:N,Mw file      Write lines N through M to file
:1,$w all.txt   Write all lines to all.txt
# Read-only mode
vim -R file     Open file in vim readonly mode
view file       Same as vim -R
# Help system
:help           Open built-in help
:help Q_wq      Quick reference for quit/write
# Edit modes
i               Insert before cursor
a               Append after cursor
Esc             Return to normal mode
v               Start visual selection
:               Enter command-line mode
/pattern        Search forward
?pattern        Search backward
# === BASH READLINE REFERENCE ===
# Editing mode
set -o vi       Enable vi-style line editing
set -o emacs    Enable emacs-style editing (default)
# Vi command mode (press Esc to enter)
k               Previous command from history
j               Next command from history
/string         Search history backward
v               Open command in $EDITOR
Esc+.           Insert last argument of previous command
# History expansion
!!              Repeat last command
!$              Last argument of previous command
!*              All arguments of previous command
!string         Last command starting with string
^old^new        Quick substitution in last command
# Job control
Ctrl+Z          Suspend current job
bg %N           Resume job N in background
fg %N           Bring job N to foreground
jobs -l         List all jobs with process IDs
disown %N       Remove job from shell job table
# Shell options
shopt -s histappend      Append history, do not overwrite
shopt -s checkwinsize    Update LINES/COLUMNS after commands
shopt -s globstar        Enable ** pattern for recursive matching
set -e                   Exit on first error
set -u                   Error on unset variables
set -o pipefail          Propagate pipe exit status
# PROMPT_COMMAND         Run before each prompt is displayed

# Navigation
Space / f       Scroll forward one page
b               Scroll backward one page
g               Go to first line
G               Go to last line
Ng              Go to line N
/pattern        Search forward
?pattern        Search backward
n               Jump to next match
N               Jump to previous match
# less options
-N              Show line numbers
-M              Show verbose status line with filename
-R              Process ANSI color sequences (needed for highlighting)
-r              Output raw control characters (less safe than -R)
-F              Quit if content fits on one screen
-X              Do not clear screen on exit
-S              Truncate long lines, no wrapping
# Useful alias
alias lessh='LESSOPEN="| /path/to/src-hilite-lesspipe.sh %s" less -M'
