#!/usr/bin/env python3
"""Module for handling data transformations in the ETL pipeline.""" 

import json
import logging
from pathlib import Path
from typing import Optional, Dict, List	




class DataTransformer:
	"""Handles transformation of raw data records."""

	def __init__(self, config_path: str, output_dir: str): 
		self.config = self._load_config(config_path)
		self.output_dir = Path(output_dir)
		self.logger = logging.getLogger(__name__)
		self.stats = {"processed": 0, "errors": 0, "skipped": 0}
	 
	def _load_config(self, path: str) -> Dict:
		"""Load transformation configuration from JSON file."""
		with open(path, 'r') as f:
			return json.load(f)

	def transform_record(self, record: Dict) -> Optional[Dict]:
		"""Apply transformation rules to a single record."""
		if not record.get("id"): 
			self.stats["skipped"] += 1
			self.logger.warning("Skipping record with no ID")
			return None



  
	def process_batch(self, records: List[Dict]) -> List[Dict]:
		"""Process a batch of records through the transformation pipeline."""
		results = []
		for record in records:
			try:
				transformed = self.transform_record(record)
				if transformed is not None:	
					results.append(transformed)
					self.stats["processed"] += 1
			except Exception as e:
				self.stats["errors"] += 1
				self.logger.error(f"Error processing record {record.get('id')}: {e}")
		return results 	




	def write_output(self, records: List[Dict], filename: str) -> str:
		"""Write transformed records to output file."""
		output_path = self.output_dir / filename 
		self.output_dir.mkdir(parents=True, exist_ok=True)
		with open(output_path, 'w') as f:
			json.dump(records, f, indent=2)
		self.logger.info(f"Wrote {len(records)} records to {output_path}")
		return str(output_path)	

	def get_stats(self) -> Dict:
		"""Return processing statistics."""
		return dict(self.stats)



 
def main():
	"""Entry point for running transformations from CLI."""
	import argparse
	parser = argparse.ArgumentParser(description="Data transformation tool")
	parser.add_argument("--config", required=True, help="Path to config file")
	parser.add_argument("--input", required=True, help="Path to input JSON")
	parser.add_argument("--output-dir", default="./output", help="Output directory")
	args = parser.parse_args()

	transformer = DataTransformer(args.config, args.output_dir)
	with open(args.input) as f:
		records = json.load(f)
	results = transformer.process_batch(records)
	transformer.write_output(results, "transformed.json")
	print(f"Stats: {transformer.get_stats()}")


if __name__ == "__main__":
	main()
